import Foundation
import CoreGraphics
import CoreText

internal class MTFontMathTableV2: MTFontMathTable {
    private let mathFont: MathFont
    private let fontSize: CGFloat
    private let unitsPerEm: UInt
    private let mTable: NSDictionary
    init(mathFont: MathFont, size: CGFloat, unitsPerEm: UInt) {
        self.mathFont = mathFont
        self.fontSize = size
        self.unitsPerEm = unitsPerEm
        mTable = mathFont.rawMathTable()
        super.init(withFont: mathFont.mtfont(size: fontSize), mathTable: mTable)
        super._mathTable = nil
        // disable all possible access to _mathTable in superclass!
    }
    override var _mathTable: NSDictionary? {
        set { fatalError("\(#function) change to _mathTable \(mathFont.rawValue) not allowed.") }
        get { mTable }
    }
    override var muUnit: CGFloat { fontSize/18 }
    
    override func fontUnitsToPt(_ fontUnits:Int) -> CGFloat {
        CGFloat(fontUnits) * fontSize / CGFloat(unitsPerEm)
    }
    override func constantFromTable(_ constName: String) -> CGFloat {
        guard let consts = mTable[kConstants] as? NSDictionary, let val = consts[constName] as? NSNumber else {
            return .zero
        }
        return fontUnitsToPt(val.intValue)
    }
    override func percentFromTable(_ percentName: String) -> CGFloat {
        guard let consts = mTable[kConstants] as? NSDictionary, let val = consts[percentName] as? NSNumber else {
            return .zero
        }
        return CGFloat(val.floatValue) / 100
    }
    /** Returns an Array of all the vertical variants of the glyph if any. If
     there are no variants for the glyph, the array contains the given glyph. */
    override func getVerticalVariantsForGlyph(_ glyph: CGGlyph) -> [NSNumber?] {
        guard let variants = mTable[kVertVariants] as? NSDictionary else { return [] }
        return self.getVariantsForGlyph(glyph, inDictionary: variants)
    }
    /** Returns an Array of all the horizontal variants of the glyph if any. If
     there are no variants for the glyph, the array contains the given glyph. */
    override func getHorizontalVariantsForGlyph(_ glyph: CGGlyph) -> [NSNumber?] {
        guard let variants = mTable[kHorizVariants] as? NSDictionary else { return [] }
        return self.getVariantsForGlyph(glyph, inDictionary:variants)
    }
    override func getVariantsForGlyph(_ glyph: CGGlyph, inDictionary variants: NSDictionary) -> [NSNumber?] {
        let font = mathFont.mtfont(size: fontSize)
        let glyphName = font.get(nameForGlyph: glyph)
        
        var glyphArray = [NSNumber]()
        let variantGlyphs = variants[glyphName] as? NSArray

        guard let variantGlyphs = variantGlyphs, variantGlyphs.count != .zero else {
            // There are no extra variants, so just add the current glyph to it.
            let glyph = font.get(glyphWithName: glyphName)
            glyphArray.append(NSNumber(value:glyph))
            return glyphArray
        }
        for gvn in variantGlyphs {
            if let glyphVariantName = gvn as? String {
                let variantGlyph = font.get(glyphWithName: glyphVariantName)
                glyphArray.append(NSNumber(value:variantGlyph))
            }
        }
        return glyphArray
    }
    /** Returns a larger vertical variant of the given glyph if any.
     If there is no larger version, this returns the current glyph.
     */
    override func getLargerGlyph(_ glyph: CGGlyph) -> CGGlyph {
        let font = mathFont.mtfont(size: fontSize)
        let glyphName = font.get(nameForGlyph: glyph)

        guard let variants = mTable[kVertVariants] as? NSDictionary,
                let variantGlyphs = variants[glyphName] as? NSArray, variantGlyphs.count != .zero else {
            // There are no extra variants, so just returnt the current glyph.
            return glyph
        }
        // Find the first variant with a different name.
        for gvn in variantGlyphs {
            if let glyphVariantName = gvn as? String, glyphVariantName != glyphName {
                let variantGlyph = font.get(glyphWithName: glyphVariantName)
                return variantGlyph
            }
        }
        // We did not find any variants of this glyph so return it.
        return glyph
    }
    /** Returns the italic correction for the given glyph if any. If there
     isn't any this returns 0. */
    override func getItalicCorrection(_ glyph: CGGlyph) -> CGFloat {
        let font = mathFont.mtfont(size: fontSize)
        let glyphName = font.get(nameForGlyph: glyph)

        guard let italics = mTable[kItalic] as? NSDictionary, let val = italics[glyphName] as? NSNumber else {
            return .zero
        }
        // if val is nil, this returns 0.
        return fontUnitsToPt(val.intValue)
    }
    override func getTopAccentAdjustment(_ glyph: CGGlyph) -> CGFloat {
        let font = mathFont.mtfont(size: fontSize)
        let glyphName = font.get(nameForGlyph: glyph)
        
        guard let accents = mTable[kAccents] as? NSDictionary, let val = accents[glyphName] as? NSNumber else {
            // If no top accent is defined then it is the center of the advance width.
            var glyph = glyph
            var advances = CGSize.zero
            CTFontGetAdvancesForGlyphs(font.ctFont, .horizontal, &glyph, &advances, 1)
            return advances.width/2
        }
        return fontUnitsToPt(val.intValue)
    }
    override func getVerticalGlyphAssembly(forGlyph glyph: CGGlyph) -> [GlyphPart] {
        let font = mathFont.mtfont(size: fontSize)
        let glyphName = font.get(nameForGlyph: glyph)
        
        guard let assemblyTable = mTable[kVertAssembly] as? NSDictionary,
              let assemblyInfo = assemblyTable[glyphName] as? NSDictionary,
              let parts = assemblyInfo[kAssemblyParts] as? NSArray else {
            // No vertical assembly defined for glyph
            // parts should always have been defined, but if it isn't return nil
            return []
        }
        
        var rv = [GlyphPart]()
        for part in parts {
            guard let partInfo = part as? NSDictionary,
                let adv = partInfo["advance"] as? NSNumber,
                let end = partInfo["endConnector"] as? NSNumber,
                let start = partInfo["startConnector"] as? NSNumber,
                let ext = partInfo["extender"] as? NSNumber,
                let glyphName = partInfo["glyph"] as? String else { continue }
            let fullAdvance = fontUnitsToPt(adv.intValue)
            let endConnectorLength = fontUnitsToPt(end.intValue)
            let startConnectorLength = fontUnitsToPt(start.intValue)
            let isExtender = ext.boolValue
            let glyph = font.get(glyphWithName: glyphName)
            let part = GlyphPart(glyph: glyph, fullAdvance: fullAdvance,
                                 startConnectorLength: startConnectorLength,
                                 endConnectorLength: endConnectorLength,
                                 isExtender: isExtender)
            rv.append(part)
        }
        return rv
    }
}

//! Word (.docx) text: the file is a ZIP, and the text is in `word/document.xml`, in
//! `<w:t>` runs inside `<w:p>` paragraphs. Headings come out as Markdown headings, list
//! items with a dash and table rows with their cells between bars, so a model still sees
//! the document's shape.

use super::inflate;

/// The most a file inside the ZIP may unpack to.
const MAX_UNPACKED: usize = 256 << 20;

pub fn text(data: &[u8]) -> Result<String, String> {
    if data.starts_with(&[0xD0, 0xCF, 0x11, 0xE0]) {
        // An OLE file: Word's own old format, or a .docx locked with a password.
        return Err(
            "This Word file is password-protected or in the old .doc format, so its \
                    text can't be read. Save it as an unprotected .docx or a PDF and attach that."
                .into(),
        );
    }
    let zip = Zip::open(data)?;
    let xml = zip
        .read("word/document.xml")?
        .ok_or("This .docx has no document in it (word/document.xml is missing).")?;
    Ok(document_text(&String::from_utf8_lossy(&xml)))
}

// ---- ZIP -------------------------------------------------------------------------------

pub struct Zip<'a> {
    data: &'a [u8],
    /// Name, method, CRC, compressed and full size, the local header's offset.
    entries: Vec<(String, u16, u32, usize, usize, usize)>,
}

fn u16_at(d: &[u8], i: usize) -> Option<u16> {
    Some(u16::from_le_bytes(d.get(i..i + 2)?.try_into().ok()?))
}

fn u32_at(d: &[u8], i: usize) -> Option<u32> {
    Some(u32::from_le_bytes(d.get(i..i + 4)?.try_into().ok()?))
}

const BROKEN: &str = "This .docx isn't a readable ZIP file: it may be damaged.";

impl<'a> Zip<'a> {
    pub fn open(data: &'a [u8]) -> Result<Self, String> {
        // The end record is in the last 64 KB (its comment can be that long).
        let from = data.len().saturating_sub(22 + 65_535);
        let end = (from..data.len().saturating_sub(21))
            .rev()
            .find(|&i| u32_at(data, i) == Some(0x0605_4b50))
            .ok_or(BROKEN)?;
        let count = u16_at(data, end + 10).ok_or(BROKEN)? as usize;
        let mut at = u32_at(data, end + 16).ok_or(BROKEN)? as usize;
        let mut entries = Vec::with_capacity(count);
        for _ in 0..count {
            if u32_at(data, at) != Some(0x0201_4b50) {
                return Err(BROKEN.into());
            }
            let get16 = |o| u16_at(data, at + o).ok_or(BROKEN);
            let get32 = |o| u32_at(data, at + o).ok_or(BROKEN);
            let (method, crc) = (get16(10)?, get32(16)?);
            let (packed, size) = (get32(20)? as usize, get32(24)? as usize);
            let (name_len, extra, comment) = (
                get16(28)? as usize,
                get16(30)? as usize,
                get16(32)? as usize,
            );
            let offset = get32(42)? as usize;
            let name = data.get(at + 46..at + 46 + name_len).ok_or(BROKEN)?;
            entries.push((
                String::from_utf8_lossy(name).into_owned(),
                method,
                crc,
                packed,
                size,
                offset,
            ));
            at += 46 + name_len + extra + comment;
        }
        Ok(Self { data, entries })
    }

    /// Every file's name, in the ZIP's order.
    pub fn names(&self) -> impl Iterator<Item = &str> {
        self.entries.iter().map(|e| e.0.as_str())
    }

    /// The file called `name`, unpacked and checked; None if there's none.
    pub fn read(&self, name: &str) -> Result<Option<Vec<u8>>, String> {
        let Some(&(_, method, crc, packed, size, offset)) =
            self.entries.iter().find(|e| e.0 == name)
        else {
            return Ok(None);
        };
        let d = self.data;
        if u32_at(d, offset) != Some(0x0403_4b50) {
            return Err(BROKEN.into());
        }
        let start = offset
            + 30
            + u16_at(d, offset + 26).ok_or(BROKEN)? as usize
            + u16_at(d, offset + 28).ok_or(BROKEN)? as usize;
        let raw = d.get(start..start + packed).ok_or(BROKEN)?;
        if size > MAX_UNPACKED {
            return Err(format!("{name} in this .docx is too large to read."));
        }
        let bytes = match method {
            0 => raw.to_vec(),
            8 => inflate::inflate(raw, MAX_UNPACKED)
                .map_err(|e| format!("{name} in this .docx can't be unpacked: {e}."))?,
            m => {
                return Err(format!(
                    "{name} in this .docx is packed in a way Vireo can't read ({m})."
                ));
            }
        };
        if inflate::crc32(&bytes) != crc {
            return Err(format!(
                "{name} in this .docx is damaged (its checksum is wrong)."
            ));
        }
        Ok(Some(bytes))
    }
}

/// A ZIP of the files as they are, not compressed: what Vireo writes an Office file as
/// (Word, Excel and the rest read it like any other).
pub fn stored_zip(files: &[(&str, &[u8])]) -> Vec<u8> {
    let mut out = Vec::new();
    let mut central = Vec::new();
    for (name, data) in files {
        let offset = out.len() as u32;
        let crc = inflate::crc32(data);
        let header = |sig: u32, central_entry: bool| {
            let mut h = Vec::new();
            h.extend_from_slice(&sig.to_le_bytes());
            if central_entry {
                h.extend_from_slice(&20u16.to_le_bytes()); // made by
            }
            h.extend_from_slice(&20u16.to_le_bytes()); // needed
            h.extend_from_slice(&0u16.to_le_bytes()); // flags
            h.extend_from_slice(&0u16.to_le_bytes()); // stored
            h.extend_from_slice(&0u32.to_le_bytes()); // time, date
            h.extend_from_slice(&crc.to_le_bytes());
            h.extend_from_slice(&(data.len() as u32).to_le_bytes());
            h.extend_from_slice(&(data.len() as u32).to_le_bytes());
            h.extend_from_slice(&(name.len() as u16).to_le_bytes());
            h.extend_from_slice(&0u16.to_le_bytes()); // extra
            if central_entry {
                h.extend_from_slice(&[0; 6]); // comment, disk, internal attributes
                h.extend_from_slice(&0u32.to_le_bytes()); // external attributes
                h.extend_from_slice(&offset.to_le_bytes());
            }
            h.extend_from_slice(name.as_bytes());
            h
        };
        out.extend(header(0x0403_4b50, false));
        out.extend_from_slice(data);
        central.extend(header(0x0201_4b50, true));
    }
    let at = out.len() as u32;
    out.extend_from_slice(&central);
    out.extend_from_slice(&0x0605_4b50u32.to_le_bytes());
    out.extend_from_slice(&[0; 4]);
    out.extend_from_slice(&(files.len() as u16).to_le_bytes());
    out.extend_from_slice(&(files.len() as u16).to_le_bytes());
    out.extend_from_slice(&(central.len() as u32).to_le_bytes());
    out.extend_from_slice(&at.to_le_bytes());
    out.extend_from_slice(&0u16.to_le_bytes());
    out
}

/// The text of WordprocessingML: paragraphs a line each, tabs and breaks kept.
pub fn document_text(xml: &str) -> String {
    let mut out = String::new();
    let mut in_text = false;
    // Inside <mc:Fallback>, which repeats what <mc:Choice> already has.
    let mut fallback = 0usize;
    let mut cell_depth = 0usize;
    // What starts the paragraph under way, written before its first text.
    let mut prefix = String::new();
    let mut paragraph_started = false;
    let mut rest = xml;
    while let Some(open) = rest.find('<') {
        if in_text && fallback == 0 {
            push_text(&mut out, &mut prefix, &mut paragraph_started, &rest[..open]);
        }
        let Some(close) = rest[open..].find('>') else {
            break;
        };
        let tag = &rest[open + 1..open + close];
        rest = &rest[open + close + 1..];
        let closing = tag.starts_with('/');
        let self_closing = tag.ends_with('/');
        let body = tag.trim_start_matches('/').trim_end_matches('/');
        let name = body.split(|c: char| c.is_whitespace()).next().unwrap_or("");
        match (name, closing) {
            ("mc:Fallback", false) if !self_closing => fallback += 1,
            ("mc:Fallback", true) => fallback = fallback.saturating_sub(1),
            _ if fallback > 0 => {}
            ("w:t", false) => in_text = !self_closing,
            ("w:t", true) => in_text = false,
            ("w:p", false) if !self_closing => {
                prefix.clear();
                paragraph_started = false;
            }
            ("w:pStyle", false) => {
                let style = attribute(body, "w:val").unwrap_or("");
                let level = if style.eq_ignore_ascii_case("Title") {
                    1
                } else {
                    style
                        .strip_prefix("Heading")
                        .or_else(|| style.strip_prefix("heading"))
                        .and_then(|n| n.trim().parse::<usize>().ok())
                        .unwrap_or(0)
                };
                if (1..=6).contains(&level) {
                    prefix = format!("{} ", "#".repeat(level));
                } else if style.starts_with("List") {
                    // "List Bullet", "List Number", "List Paragraph".
                    prefix = "- ".into();
                }
            }
            ("w:numPr", false) if prefix.is_empty() => prefix = "- ".into(),
            ("w:tab", false) => push_text(&mut out, &mut prefix, &mut paragraph_started, "\t"),
            ("w:br" | "w:cr", false) => out.push('\n'),
            ("w:noBreakHyphen", false) => {
                push_text(&mut out, &mut prefix, &mut paragraph_started, "-")
            }
            ("w:p", true) => {
                if cell_depth > 0 {
                    if !out.ends_with([' ', '\n']) && !out.ends_with("| ") {
                        out.push(' ');
                    }
                } else {
                    trim_end_spaces(&mut out);
                    out.push('\n');
                }
            }
            ("w:tc", false) if !self_closing => cell_depth += 1,
            ("w:tc", true) => {
                cell_depth = cell_depth.saturating_sub(1);
                trim_end_spaces(&mut out);
                out.push_str(" | ");
            }
            ("w:tr", false) if !self_closing => {
                if !out.is_empty() && !out.ends_with('\n') {
                    out.push('\n');
                }
                out.push_str("| ");
            }
            ("w:tr", true) => {
                trim_end_spaces(&mut out);
                out.push('\n');
            }
            ("w:tbl", true) => out.push('\n'),
            _ => {}
        }
    }
    super::tidy(&out)
}

fn push_text(out: &mut String, prefix: &mut String, started: &mut bool, text: &str) {
    if text.is_empty() {
        return;
    }
    if !*started {
        *started = true;
        out.push_str(prefix);
    }
    unescape_into(out, text);
}

fn trim_end_spaces(out: &mut String) {
    while out.ends_with([' ', '\t']) {
        out.pop();
    }
}

/// An attribute's value in a tag's body (`w:pStyle w:val="Heading1"`).
fn attribute<'t>(body: &'t str, key: &str) -> Option<&'t str> {
    let at = body.find(&format!("{key}="))? + key.len() + 1;
    let quote = body[at..].chars().next()?;
    let value = &body[at + 1..];
    Some(&value[..value.find(quote)?])
}

/// XML's escapes: the five named ones and numeric characters.
pub fn unescape_into(out: &mut String, text: &str) {
    let mut rest = text;
    while let Some(amp) = rest.find('&') {
        out.push_str(&rest[..amp]);
        rest = &rest[amp..];
        let Some(semi) = rest.find(';').filter(|&s| s <= 10) else {
            out.push('&');
            rest = &rest[1..];
            continue;
        };
        let entity = &rest[1..semi];
        let c = match entity {
            "amp" => Some('&'),
            "lt" => Some('<'),
            "gt" => Some('>'),
            "quot" => Some('"'),
            "apos" => Some('\''),
            _ => entity
                .strip_prefix("#x")
                .or_else(|| entity.strip_prefix("#X"))
                .and_then(|h| u32::from_str_radix(h, 16).ok())
                .or_else(|| entity.strip_prefix('#').and_then(|d| d.parse().ok()))
                .and_then(char::from_u32),
        };
        match c {
            Some(c) => {
                out.push(c);
                rest = &rest[semi + 1..];
            }
            None => {
                out.push('&');
                rest = &rest[1..];
            }
        }
    }
    out.push_str(rest);
}

#[cfg(test)]
pub mod tests {
    use super::*;

    pub use super::stored_zip;

    #[test]
    fn reads_a_word_file_from_textutil() {
        // `textutil -convert docx` of a small HTML page: deflated, with a list.
        let data = std::fs::read(concat!(
            env!("CARGO_MANIFEST_DIR"),
            "/tests/fixtures/report.docx"
        ))
        .unwrap();
        let text = text(&data).unwrap();
        assert!(
            text.starts_with("Quarterly report\nSales rose 12% & costs fell.\n"),
            "{text}"
        );
        assert!(text.contains("Second <point>"), "{text}");
    }

    #[test]
    fn keeps_headings_lists_and_tables() {
        let xml = r#"<w:document><w:body>
            <w:p><w:pPr><w:pStyle w:val="Heading2"/></w:pPr><w:r><w:t>Plan</w:t></w:r></w:p>
            <w:p><w:pPr><w:numPr><w:ilvl w:val="0"/></w:numPr></w:pPr><w:r><w:t>Buy milk</w:t></w:r></w:p>
            <w:p><w:pPr><w:pStyle w:val="ListBullet"/></w:pPr><w:r><w:t>Call Sam</w:t></w:r></w:p>
            <w:p><w:r><w:t xml:space="preserve">Tab</w:t><w:tab/><w:t>bed &amp; </w:t><w:t>done&#x2019;s</w:t></w:r></w:p>
            <w:tbl><w:tr><w:tc><w:p><w:r><w:t>Region</w:t></w:r></w:p></w:tc><w:tc><w:p><w:r><w:t>Total</w:t></w:r></w:p></w:tc></w:tr>
            <w:tr><w:tc><w:p><w:r><w:t>North</w:t></w:r></w:p></w:tc><w:tc><w:p><w:r><w:t>4,200</w:t></w:r></w:p></w:tc></w:tr></w:tbl>
            <w:p><mc:AlternateContent><mc:Choice><w:t>Box</w:t></mc:Choice><mc:Fallback><w:t>Box</w:t></mc:Fallback></mc:AlternateContent></w:p>
            <w:p><w:r><w:delText>gone</w:delText><w:t>After</w:t></w:r></w:p>
            </w:body></w:document>"#;
        let zip = stored_zip(&[("word/document.xml", xml.as_bytes())]);
        assert_eq!(
            text(&zip).unwrap(),
            "## Plan\n- Buy milk\n- Call Sam\nTab\tbed & done\u{2019}s\n| Region | Total |\n| North | 4,200 |\n\nBox\nAfter"
        );
    }

    #[test]
    fn says_what_is_wrong_with_a_bad_file() {
        assert!(
            text(b"not a zip")
                .unwrap_err()
                .contains("isn't a readable ZIP")
        );
        let no_document = stored_zip(&[("other.xml", b"<x/>")]);
        assert!(text(&no_document).unwrap_err().contains("no document"));
        let mut damaged = stored_zip(&[("word/document.xml", b"<w:t>hi</w:t>")]);
        let at = damaged.windows(2).position(|w| w == b"hi").unwrap();
        damaged[at] = b'H';
        assert!(text(&damaged).unwrap_err().contains("checksum"));
        assert!(
            text(&[0xD0, 0xCF, 0x11, 0xE0, 0, 0])
                .unwrap_err()
                .contains("password-protected")
        );
    }
}

import XCTest
@testable import VireoMac

final class ModelDiscoveryTests: XCTestCase {
    private var directory: URL!

    override func setUpWithError() throws {
        directory = FileManager.default.temporaryDirectory.appendingPathComponent(UUID().uuidString, isDirectory: true)
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
    }

    override func tearDownWithError() throws {
        try? FileManager.default.removeItem(at: directory)
    }

    private let gb: Int64 = 1_073_741_824

    func testMemoryFitLeavesRoomForContextAndRuntime() {
        let budget = 48 * gb
        XCTAssertEqual(MemoryFit.assess(weightBytes: 5 * gb, budgetBytes: budget), .fits)
        XCTAssertEqual(MemoryFit.assess(weightBytes: 38 * gb, budgetBytes: budget), .tight)
        XCTAssertEqual(MemoryFit.assess(weightBytes: 44 * gb, budgetBytes: budget), .tooLarge)
    }

    func testSearchEntriesAcceptBothFormsOfGated() throws {
        let json = """
        [{"id":"mlx-community/Llama-3.1-8B-Instruct-4bit","gated":false,"private":false,"downloads":930140,"likes":11,"pipeline_tag":"text-generation"},
         {"id":"someone/Locked-Model-4bit","gated":"manual","private":false}]
        """
        let entries = try JSONDecoder().decode([SearchEntry].self, from: Data(json.utf8))

        var data = Data("GGUF".utf8)
        func u32(_ value: UInt32) { withUnsafeBytes(of: value.littleEndian) { data.append(contentsOf: $0) } }
        func u64(_ value: UInt64) { withUnsafeBytes(of: value.littleEndian) { data.append(contentsOf: $0) } }
        func string(_ value: String) { u64(UInt64(value.utf8.count)); data.append(contentsOf: value.utf8) }
        u32(3); u64(0); u64(3)
        string("tokenizer.ggml.token_type"); u32(9); u32(4); u64(3); u32(1); u32(2); u32(3)
        string("general.alignment"); u32(4); u32(32)
        string("general.architecture"); u32(8); string(architecture)
        data.append(Data(count: padding))
        try FileManager.default.createDirectory(at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
        try data.write(to: url)
    }

===== Mac/App/.build/checkouts/Sparkle/Configurations/generate_progress_tool_localizations.py =====
#!/usr/bin/env python3
#
# generate_progress_tool_localizations.py
#
# The Installer Progress helper is a minimal tool that minimizes loading resolve resources.
# This script reads the localizations from Sparkle/*.lproj/Sparkle.strings
# and generates a header with the handful of strings the tool needs.
# It also keeps CFBundleLocalizations in InstallerProgress-Info.plist in sync
# with the set of locales found.
#
# This script must be re-run by hand after changing any of the strings below in
# Sparkle/*.lproj/Sparkle.strings, or after adding/removing a locale.
# Output is deterministic between runs.

import glob
import json
import os
import plistlib
import subprocess
import sys

SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
REPO_ROOT = os.path.dirname(SCRIPT_DIR)
SPARKLE_DIR = os.path.join(REPO_ROOT, "Sparkle")
OUTPUT_HEADER = os.path.join(REPO_ROOT, "Sparkle", "InstallerProgress", "SPUInstallerProgressLocalizedStrings.h")
INFO_PLIST = os.path.join(REPO_ROOT, "Sparkle", "InstallerProgress", "InstallerProgress-Info.plist")

# Fixed order so regenerating the header without content changes never reorders keys
REQUIRED_KEYS = (
    "Cancel Update",
    "Installing update…",
    "Updating %@",
)

# Base.lproj holds the canonical/development-language (English) strings, but "Base"
# is not a real locale identifier for +[NSBundle preferredLocalizationsFromArray:]
# to match against, so it's keyed as "en" in the generated table.
LPROJ_LOCALE_OVERRIDES = {
    "Base": "en",
}


def load_strings(strings_path):
    result = subprocess.run(
        ["plutil", "-convert", "json", "-o", "-", strings_path],
        capture_output=True,
        check=True,
        text=True,
    )
    return json.loads(result.stdout)


def objc_string_literal(value):
    escaped = value.replace("\\", "\\\\").replace('"', '\\"')
    escaped = escaped.replace("\n", "\\n").replace("\t", "\\t")
    return '@"' + escaped + '"'


def collect_localizations():
    localizations = {}

    lproj_dirs = sorted(glob.glob(os.path.join(SPARKLE_DIR, "*.lproj")))
    for lproj_dir in lproj_dirs:
        lproj_name = os.path.basename(lproj_dir)[: -len(".lproj")]
        locale = LPROJ_LOCALE_OVERRIDES.get(lproj_name, lproj_name)

        strings_path = os.path.join(lproj_dir, "Sparkle.strings")
        if not os.path.isfile(strings_path):
            continue

        table = load_strings(strings_path)

        localized_values = {}
        for key in REQUIRED_KEYS:
            value = table.get(key)
            if value is None:
                print("warning: %s.lproj/Sparkle.strings is missing key %r" % (lproj_name, key), file=sys.stderr)
                continue
            localized_values[key] = value

        if localized_values:
            localizations[locale] = localized_values

    return localizations


def generate_header(localizations):
    lines = []
    lines.append("//")
    lines.append("//  SPUInstallerProgressLocalizedStrings.h")
    lines.append("//  Sparkle")
    lines.append("//")
    lines.append("//  This file is auto-generated by Configurations/generate_progress_tool_localizations.py")
    lines.append("//  Do not edit by hand. Re-run the script after changing the strings below in")
    lines.append("//  Sparkle/*.lproj/Sparkle.strings")
    lines.append("//")
    lines.append("")
    lines.append("#import <Foundation/Foundation.h>")
    lines.append("")
    lines.append("static NSDictionary<NSString *, NSDictionary<NSString *, NSString *> *> *const SPUInstallerProgressLocalizedStringsTable = @{")
    for locale in sorted(localizations.keys()):
        lines.append('    @"%s": @{' % locale)
        for key in REQUIRED_KEYS:
            value = localizations[locale].get(key)
            if value is None:
                continue
            lines.append("        %s: %s," % (objc_string_literal(key), objc_string_literal(value)))
        lines.append("    },")
    lines.append("};")
    lines.append("")
    return "\n".join(lines)


def update_info_plist_localizations(locales):
    with open(INFO_PLIST, "rb") as f:
        plist = plistlib.load(f)

    sorted_locales = sorted(locales)
    if plist.get("CFBundleLocalizations") == sorted_locales:
        print("%s is already up to date (%d locales)" % (INFO_PLIST, len(sorted_locales)))
        return

    plist["CFBundleLocalizations"] = sorted_locales

    # sort_keys=False preserves the existing top-level key order, so only
    # CFBundleLocalizations's contents change in the diff
    with open(INFO_PLIST, "wb") as f:
        plistlib.dump(plist, f, fmt=plistlib.FMT_XML, sort_keys=False)

    print("Wrote %s (%d locales)" % (INFO_PLIST, len(sorted_locales)))

def update_header(localizations):
    header_contents = generate_header(localizations)

    if os.path.isfile(OUTPUT_HEADER):
        with open(OUTPUT_HEADER, "r", encoding="utf-8") as f:
            if f.read() == header_contents:
                print("%s is already up to date (%d locales)" % (OUTPUT_HEADER, len(localizations)))
                return

    with open(OUTPUT_HEADER, "w", encoding="utf-8") as f:
        f.write(header_contents)

    print("Wrote %s (%d locales)" % (OUTPUT_HEADER, len(localizations)))


def main():
    localizations = collect_localizations()
    update_header(localizations)
    update_info_plist_localizations(localizations.keys())

if __name__ == "__main__":
    main()

===== crates/vireo-eval/src/pc.rs =====
//! A stand-in PC: a few folders and files in memory, and the app's tools over them,
//! answering as the app's `PcTools` does. A model may take any sensible route (listing a
//! folder before reading, or searching): each step gets a real answer, and every run gets
//! the same ones. The tools that change files change it, so a case can check what the
//! reply left; a folder is there while it holds a file.

use std::collections::BTreeMap;

use crate::json::Value;

/// The home folder relative paths start from, as the app's do.
pub const HOME: &str = r"C:\Users\me";

/// "Invoice-Register-FY2026.xlsx": a file's name (the app's `FileNamePattern`,
/// `^[^\s/\\]+\.[A-Za-z0-9]{1,6}$`), searched for by name only.
fn is_file_name(s: &str) -> bool {
    let Some((stem, ext)) = s.rsplit_once('.') else {
        return false;
    };
    !stem.is_empty()
        && !stem
            .chars()
            .any(|c| c.is_whitespace() || c == '/' || c == '\\')
        && (1..=6).contains(&ext.len())
        && ext.chars().all(|c| c.is_ascii_alphanumeric())
}

#[derive(Default, Clone, Debug)]
pub struct Pc {
    /// Files by full path (lowercased for lookup), with their path as written and text.
    files: BTreeMap<String, (String, String)>,
    /// `git status --short` per project folder (lowercased).
    git: BTreeMap<String, String>,
    /// Tools whose approvals the stand-in user declines.
    declines: Vec<String>,
    /// `git diff` per project folder (lowercased), where a case sets one.
    diffs: BTreeMap<String, String>,
    /// Workbooks by path (lowercased): their one sheet's rows, the first a header.
    sheets: BTreeMap<String, Vec<Vec<String>>>,
    /// A project's tests, build and linter, by folder (lowercased) and action: the
    /// command, its exit code and its output.
    commands: BTreeMap<(String, String), (String, i32, String)>,
}

fn key(path: &str) -> String {
    path.to_lowercase()
}

/// A path as the tools take it: `/` or `\`, full or relative to the home folder, with no
/// trailing separator.
fn full(path: &str) -> String {
    let p = path.trim().trim_matches('"').replace('/', "\\");
    let p = p.trim_end_matches('\\');
    if p.len() >= 2 && p.as_bytes()[1] == b':' {
        p.to_string()
    } else if p.is_empty() || p == "." || p == "~" {
        HOME.to_string()
    } else {
        format!(
            "{HOME}\\{}",
            p.trim_start_matches(".\\").trim_start_matches("~\\")
        )
    }
}

impl Pc {
    pub fn file(mut self, path: &str, text: &str) -> Self {
        self.files
            .insert(key(path), (path.to_string(), text.to_string()));
        self
    }

    pub fn git(mut self, folder: &str, status: &str) -> Self {
        self.git.insert(key(folder), status.to_string());
        self
    }

    /// A workbook with one sheet ("Sheet1") of these rows, the first its header. It's also a
    /// file, so listing and searching find it, and read_file gives the sheet, as the app's
    /// read_file describes a workbook.
    pub fn sheet(mut self, path: &str, rows: &[&[&str]]) -> Self {
        let rows: Vec<Vec<String>> = rows
            .iter()
            .map(|r| r.iter().map(|c| c.to_string()).collect())
            .collect();
        let text = sheet_text(&rows);
        self.sheets.insert(key(path), rows);
        self.file(path, &text)
    }

    /// What a project's tests, build or linter (`action`: "tests", "build", "linter") run
    /// and give, as the app's RunProjectCommandAsync reports it.
    pub fn command(
        mut self,
        folder: &str,
        action: &str,
        command: &str,
        exit: i32,
        output: &str,
    ) -> Self {
        self.commands.insert(
            (key(folder), action.to_string()),
            (command.to_string(), exit, output.to_string()),
        );
        self
    }

    /// The project's `git diff` (otherwise a one-line change of a port).
    pub fn diff(mut self, folder: &str, diff: &str) -> Self {
        self.diffs.insert(key(folder), diff.to_string());
        self
    }

    /// The stand-in user declines this tool's approvals (they approve all others).
    pub fn declining(mut self, tool: &str) -> Self {
        self.declines.push(tool.to_string());
        self
    }

    /// Whether the stand-in user approves a call to this tool. Remembering is always
    /// declined: a fact kept from one case would be in every later case's instructions.
    pub fn approves(&self, tool: &str) -> bool {
        !matches!(tool, "remember" | "forget") && !self.declines.iter().any(|t| t == tool)
    }

    /// A file's text, by its full path (any case).
    pub fn text(&self, path: &str) -> Option<&str> {
        self.files.get(&key(path)).map(|(_, t)| t.as_str())
    }

    fn is_folder(&self, path: &str) -> bool {
        let prefix = format!("{}\\", key(path));
        self.files.keys().any(|k| k.starts_with(&prefix))
    }

    /// A folder's files and folders (folders with a trailing `\`), sorted.
    fn entries(&self, folder: &str) -> Vec<String> {
        let prefix = format!("{}\\", key(folder));
        let mut entries: Vec<String> = self
            .files
            .values()
            .filter(|(p, _)| key(p).starts_with(&prefix))
            .map(|(p, _)| {
                let rest = &p[prefix.len()..];
                match rest.split_once('\\') {
                    Some((folder, _)) => format!("{folder}\\"),
                    None => rest.to_string(),
                }
            })
            .collect();
        entries.dedup();
        entries
    }

    /// A missing path's error, as the app's `NotFound` gives it: the nearest folder above
    /// it that's there (in the home folder), what that holds, the names there nearest the
    /// one asked for, and files of the same name further down.
    fn not_found(&self, path: &str, first: String) -> String {
        let home = key(HOME);
        let mut folder = path.to_string();
        let mut tail: Vec<String> = Vec::new();
        loop {
            let Some((parent, name)) = folder.rsplit_once('\\') else {
                return first;
            };
            let parent_key = key(parent);
            if !(parent_key == home || parent_key.starts_with(&format!("{home}\\"))) {
                return first;
            }
            tail.insert(0, name.to_string());
            folder = parent.to_string();
            if self.is_folder(&folder) {
                break;
            }
        }
        // Folders first, then files, each by name ignoring case.
        let mut entries = self.entries(&folder);
        entries.sort_by_key(|e| (!e.ends_with('\\'), e.to_lowercase()));
        let names: Vec<&str> = entries.iter().map(|e| e.trim_end_matches('\\')).collect();
        let below = format!("{}\\", key(&folder));
        let asked = key(tail.last().map_or("", String::as_str));
        let found: Vec<&str> = self
            .files
            .iter()
            .filter(|(k, _)| k.starts_with(&below) && k.rsplit('\\').next() == Some(asked.as_str()))
            .map(|(_, (p, _))| p.as_str())
            .take(3)
            .collect();
        not_found_text(
            &first,
            &folder,
            &tail,
            &entries,
            &closest(&tail, &names),
            &found,
        )
    }

    /// A file or folder moved, as the app's `Move`: refused onto something already there
    /// (a change of case only is the same item) or into a folder that isn't.
    fn move_item(&mut self, source: &str, destination: &str) -> Result<(), String> {
        let is_file = self.files.contains_key(&key(source));
        if !is_file && !self.is_folder(source) {
            return Err(self.not_found(source, format!("File not found: {source}")));
        }
        let same = key(source) == key(destination);
        if !same && (self.files.contains_key(&key(destination)) || self.is_folder(destination)) {
            return Err(format!("Destination already exists: {destination}"));
        }
        if let Some((parent, _)) = destination.rsplit_once('\\')
            && key(parent) != key(HOME)
            && !self.is_folder(parent)
        {
            return Err(format!("Folder not found: {parent}"));
        }
        let moved: Vec<String> = if is_file {
            vec![key(source)]
        } else {
            let prefix = format!("{}\\", key(source));
            self.files
                .keys()
                .filter(|k| k.starts_with(&prefix))
                .cloned()
                .collect()
        };
        for k in moved {
            let (path, text) = self.files.remove(&k).unwrap();
            let new_path = format!("{destination}{}", &path[source.len()..]);
            self.files.insert(key(&new_path), (new_path, text));
        }
        Ok(())
    }

    /// Runs one of the app's tools: its output, or why it failed.
    pub fn run(&mut self, tool: &str, args: &Value) -> Result<String, String> {
        let arg = |name: &str| {
            args.str(name)
                .map(full)
                .ok_or(format!("{tool} needs {name}"))
        };
        match tool {
            "read_file" => {
                let path = arg("path")?;
                match self.files.get(&key(&path)) {
                    Some((_, text)) => Ok(text.clone()),
                    None if self.is_folder(&path) => {
                        Err(format!("{path} is a folder, not a file."))
                    }
                    None => Err(self.not_found(&path, format!("File not found: {path}"))),
                }
            }
            "list_directory" => {
                let path = arg("path")?;
                if !self.is_folder(&path) {
                    return Err(self.not_found(&path, format!("Folder not found: {path}")));
                }
                Ok(self.entries(&path).join("\n"))
            }
            // As the app's SearchFiles: a file or folder name matches when it holds every
            // word of the search ("*.pdf" finds a.pdf, "STEM camp" finds Stem-Camp.docx),
            // with - _ . read as spaces; unless the search is itself a file name, lines
            // inside files that hold the whole search match too.
            "search_files" => {
                let folder = arg("directory")?;
                let raw = args.str("pattern").unwrap_or("").trim().to_string();
                let pattern = raw.to_lowercase();
                let words: Vec<String> = pattern
                    .split(|c: char| !(c.is_alphanumeric() || c == '_'))
                    .filter(|w| !w.is_empty())
                    .map(str::to_string)
                    .collect();
                let names_only = is_file_name(&raw);
                let name_matches = |name: &str| {
                    let plain: String = name
                        .to_lowercase()
                        .chars()
                        .map(|c| if matches!(c, '-' | '_' | '.') { ' ' } else { c })
                        .collect();
                    !words.is_empty() && words.iter().all(|w| plain.contains(w.as_str()))
                };
                let prefix = format!("{}\\", key(&folder));
                let mut results = Vec::new();
                let mut folders_seen = std::collections::BTreeSet::new();
                for (k, (p, text)) in &self.files {
                    if !k.starts_with(&prefix) {
                        continue;
                    }
                    // The folders between the search's folder and the file.
                    let rel = &p[prefix.len()..];
                    let mut so_far = p[..prefix.len() - 1].to_string();
                    for part in rel
                        .split('\\')
                        .collect::<Vec<_>>()
                        .split_last()
                        .map_or(&[][..], |(_, dirs)| dirs)
                    {
                        so_far = format!("{so_far}\\{part}");
                        if folders_seen.insert(so_far.to_lowercase()) && name_matches(part) {
                            results.push(format!("{so_far}\\ (folder name matches)"));
                        }
                    }
                    let name = p.rsplit('\\').next().unwrap_or(p);
                    if name_matches(name) {
                        results.push(format!("{p} (file name matches)"));
                    } else if !names_only {
                        for (n, line) in text.lines().enumerate() {
                            if line.to_lowercase().contains(&pattern) {
                                results.push(format!("{p}:{}: {}", n + 1, line.trim()));
                            }
                        }
                    }
                }
                Ok(if results.is_empty() {
                    format!("No matches found for \"{raw}\" in {folder}")
                } else {
                    let n = results.len();
                    format!(
                        "{n} match{} for \"{raw}\":\n\n{}",
                        if n == 1 { "" } else { "es" },
                        results.join("\n")
                    )
                })
            }
            "file_exists" => {
                let path = arg("path")?;
                Ok(if self.files.contains_key(&key(&path)) {
                    format!("{path} is a file.")
                } else if self.is_folder(&path) {
                    format!("{path} is a folder.")
                } else {
                    format!("{path} doesn't exist.")
                })
            }
            "git_status" | "git_diff" => {
                let path = arg("path")?;
                let project = self
                    .git
                    .iter()
                    .find(|(folder, _)| key(&path).starts_with(folder.as_str()))
                    .map(|(folder, status)| (folder.clone(), status.clone()));
                match (tool, project) {
                    (_, None) => Err(format!("{path} isn't in a Git project.")),
                    ("git_status", Some((_, status))) => Ok(status),
                    (_, Some((folder, _))) => Ok(self.diffs.get(&folder).cloned().unwrap_or_else(|| {
                        "diff --git a/src/main.rs b/src/main.rs\n-    let port = 3000;\n+    let port = 8080;".into()
                    })),
                }
            }
            // The changes, as the app's WriteFileAsync, DeleteFile, RenameFile and MoveFile
            // answer (the app's journal, which undoes them, has no part here).
            "write_file" => {
                let path = arg("path")?;
                if self.is_folder(&path) {
                    return Err(format!("{path} is a folder, not a file."));
                }
                let content = args.str("content").unwrap_or("").to_string();
                self.files.insert(key(&path), (path.clone(), content));
                Ok(format!("File written successfully: {path}"))
            }
            "delete_file" => {
                let path = arg("path")?;
                if self.is_folder(&path) {
                    return Err(format!(
                        "{path} is a folder. delete_file deletes files only."
                    ));
                }
                if self.files.remove(&key(&path)).is_none() {
                    return Err(self.not_found(&path, format!("File not found: {path}")));
                }
                Ok(format!("File moved to the Recycle Bin: {path}"))
            }
            "rename_file" => {
                let source = arg("path")?;
                let new_name = args.str("new_name").unwrap_or("").trim().to_string();
                if new_name.is_empty()
                    || new_name == "."
                    || new_name == ".."
                    || new_name.contains(['\\', '/'])
                {
                    return Err("New name must be a single file or folder name.".into());
                }
                let parent = source
                    .rsplit_once('\\')
                    .map_or(HOME, |(p, _)| p)
                    .to_string();
                let destination = format!("{parent}\\{new_name}");
                self.move_item(&source, &destination)?;
                let name = |p: &str| p.rsplit('\\').next().unwrap_or(p).to_string();
                Ok(format!(
                    "Renamed successfully: {} -> {}",
                    name(&source),
                    name(&destination)
                ))
            }
            "move_file" => {
                let source = arg("source")?;
                let destination = arg("destination")?;
                self.move_item(&source, &destination)?;
                Ok(format!("Moved successfully: {source} -> {destination}"))
            }
            "read_spreadsheet" => {
                let path = arg("path")?;
                match self.sheets.get(&key(&path)) {
                    Some(rows) => Ok(sheet_text(rows)),
                    None if self.files.contains_key(&key(&path)) => {
                        Err(format!("{path} isn't an Excel workbook (.xlsx)."))
                    }
                    None => Err(self.not_found(&path, format!("File not found: {path}"))),
                }
            }
            // As the app's CreateWordDocumentAsync: ".docx" added where it's missing, and
            // the path given back (the server words the result). Its text is the title,
            // then the Markdown, so a case can read what was written.
            "create_docx" => {
                let mut path = arg("path")?;
                if !path.to_lowercase().ends_with(".docx") {
                    path.push_str(".docx");
                }
                if self.is_folder(&path) {
                    return Err(format!("{path} is a folder."));
                }
                let text = format!(
                    "{}\n\n{}",
                    args.str("title").unwrap_or(""),
                    args.str("content").unwrap_or("")
                );
                self.files.insert(key(&path), (path.clone(), text));
                Ok(path)
            }
            "run_tests" | "run_build" | "run_linter" => {
                let path = arg("path")?;
                let action = &tool["run_".len()..];
                let action = if action == "tests" { "tests" } else { action };
                let found = self.commands.iter().find(|((folder, a), _)| {
                    a == action && key(&path).starts_with(folder.as_str())
                });
                match found {
                    None if !self.is_folder(&path) => {
                        Err(self.not_found(&path, format!("Folder not found: {path}")))
                    }
                    None => Err(format!(
                        "No {action} command found for {path}: no project file there sets one up."
                    )),
                    Some(((folder, _), (command, exit, output))) => {
                        let folder = self
                            .files
                            .values()
                            .map(|(p, _)| p)
                            .find(|p| key(p).starts_with(&format!("{folder}\\")))
                            .map_or(path.clone(), |p| p[..folder.len()].to_string());
                        let words: Vec<String> =
                            command.split(' ').map(crate::json::string).collect();
                        Ok(format!(
                            "{{\"ok\":{},\"path\":{},\"command\":[{}],\"exitCode\":{exit},\"output\":{}}}",
                            *exit == 0,
                            crate::json::string(&folder),
                            words.join(","),
                            crate::json::string(output)
                        ))
                    }
                }
            }
            // As the app's WriteSpreadsheetAsync: rows written from a cell, or after the last
            // row with "append"; a new workbook made when there's none.
            "write_spreadsheet" => {
                let path = arg("path")?;
                let rows: Vec<Vec<String>> = match args.get("rows") {
                    Some(Value::Array(rows)) => rows
                        .iter()
                        .map(|r| match r {
                            Value::Array(cells) => cells
                                .iter()
                                .map(|c| match c {
                                    Value::String(s) => s.clone(),
                                    Value::Null => String::new(),
                                    other => other.to_json(),
                                })
                                .collect(),
                            other => vec![other.to_json()],
                        })
                        .collect(),
                    _ => return Err("Missing argument: rows".into()),
                };
                if self.is_folder(&path) {
                    return Err(format!("{path} is a folder."));
                }
                let start = args.str("start_cell").unwrap_or("A1").trim().to_lowercase();
                let sheet = self.sheets.entry(key(&path)).or_default();
                let first = if start == "append" {
                    sheet.len()
                } else {
                    start
                        .trim_start_matches(|c: char| c.is_ascii_alphabetic())
                        .parse::<usize>()
                        .map_or(0, |n| n.saturating_sub(1))
                };
                for (i, row) in rows.iter().enumerate() {
                    let at = first + i;
                    if at < sheet.len() {
                        sheet[at] = row.clone();
                    } else {
                        sheet.push(row.clone());
                    }
                }
                let text = sheet_text(sheet);
                self.files.insert(key(&path), (path.clone(), text));
                Ok(format!("Wrote {} rows to {path}.", rows.len()))
            }
            // As the app's FillTemplateAsync: each {{name}} replaced, saved under a new name
            // (beside the template when it's a name alone).
            "fill_docx_template" => {
                let template = arg("template")?;
                let Some((_, text)) = self.files.get(&key(&template)).cloned() else {
                    return Err(self.not_found(&template, format!("File not found: {template}")));
                };
                let target = args.str("path").unwrap_or("").trim().to_string();
                let mut path = if target.contains(['/', '\\']) {
                    full(&target)
                } else {
                    let folder = template.rsplit_once('\\').map_or(HOME, |(f, _)| f);
                    format!("{folder}\\{target}")
                };
                if !path.to_lowercase().ends_with(".docx") {
                    path.push_str(".docx");
                }
                if key(&path) == key(&template) {
                    return Err("Save the filled document under a new name, so the template stays as it is.".into());
                }
                let mut filled = text;
                let mut names = Vec::new();
                if let Some(Value::Object(values)) = args.get("values") {
                    for (name, value) in values {
                        let value = match value {
                            Value::String(s) => s.clone(),
                            other => other.to_json(),
                        };
                        let mark = format!("{{{{{}}}}}", name.trim_matches(['{', '}']));
                        if filled.contains(&mark) {
                            names.push(name.clone());
                        }
                        filled = filled.replace(&mark, &value);
                    }
                } else {
                    return Err("values must name each placeholder and its text.".into());
                }
                self.files.insert(key(&path), (path.clone(), filled));
                Ok(format!("Document saved successfully: {path}\nFilled: {}.", names.join(", ")))
            }
            // As the app's CheckChange: the change made on a copy, so its errors are the
            // change's own, and nothing here changes.
            "check_change" => {
                let tool = args.str("tool").unwrap_or("").to_string();
                let change = args.get("args").cloned().unwrap_or(Value::Null);
                if !crate::cases::CHANGE_TOOLS.contains(&tool.as_str()) {
                    return Err(format!("{tool} can't be checked on this PC."));
                }
                self.clone().run(&tool, &change).map(|_| "ok".to_string())
            }
            other => Err(format!("This PC doesn't run {other}.")),
        }
    }
}

/// A sheet as the app's `Spreadsheet.Read` gives it: the sheet names, the range, then
/// "Row | A | B" and a line a row.
fn sheet_text(rows: &[Vec<String>]) -> String {
    let width = rows.iter().map(Vec::len).max().unwrap_or(0).max(1);
    let column = |i: usize| ((b'A' + i as u8) as char).to_string();
    let range = format!("A1:{}{}", column(width - 1), rows.len().max(1));
    let mut text = format!(
        "Sheets: Sheet1\n\nSheet \"Sheet1\", {range} (used range {range})\nRow | {}\n",
        (0..width).map(column).collect::<Vec<_>>().join(" | ")
    );
    for (i, row) in rows.iter().enumerate() {
        text.push_str(&format!("{} | {}\n", i + 1, row.join(" | ")));
    }
    text
}

/// The words of a name, for comparing names: lowercase runs of letters or of digits, a
/// file's extension left out ("Report_2026.pdf" is "report", "2026").
fn words(name: &str, is_file: bool) -> Vec<String> {
    let stem = match name.rsplit_once('.') {
        Some((stem, _)) if is_file && !stem.is_empty() => stem,
        _ => name,
    };
    let mut words = Vec::new();
    let mut word = String::new();
    for c in stem.to_lowercase().chars() {
        let breaks = word
            .chars()
            .last()
            .is_some_and(|last| last.is_ascii_digit() != c.is_ascii_digit());
        if (!c.is_alphanumeric() || breaks) && !word.is_empty() {
            words.push(std::mem::take(&mut word));
        }
        if c.is_alphanumeric() {
            word.push(c);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    words
}

/// Up to three of `names` sharing the most words with the missing part of the path
/// (`tail`: its folders and its name), as the app's `Closest`.
fn closest<'a>(tail: &[String], names: &[&'a str]) -> Vec<&'a str> {
    let asked: Vec<String> = tail
        .iter()
        .enumerate()
        .flat_map(|(i, part)| words(part, i + 1 == tail.len()))
        .collect();
    let mut scored: Vec<(usize, &str)> = names
        .iter()
        .map(|n| {
            let have = words(n, true);
            (asked.iter().filter(|w| have.contains(w)).count(), *n)
        })
        .filter(|(score, _)| *score > 0)
        .collect();
    scored.sort_by(|a, b| b.0.cmp(&a.0).then(a.1.cmp(b.1)));
    scored.into_iter().take(3).map(|(_, n)| n).collect()
}

/// How many of a folder's items a missing path's error lists.
const MAX_HINT_ENTRIES: usize = 30;

/// The error's text, the same as the app's `NotFound`.
fn not_found_text(
    first: &str,
    folder: &str,
    tail: &[String],
    entries: &[String],
    closest: &[&str],
    found: &[&str],
) -> String {
    let mut lines = vec![first.to_string()];
    if tail.len() > 1 {
        lines.push(format!("There's no folder {folder}\\{} either.", tail[0]));
    }
    lines.push(if entries.is_empty() {
        format!("{folder} is empty.")
    } else if entries.len() > MAX_HINT_ENTRIES {
        format!(
            "{folder} holds {} items, the first {MAX_HINT_ENTRIES}: {}",
            entries.len(),
            entries[..MAX_HINT_ENTRIES].join(", ")
        )
    } else {
        format!("{folder} holds: {}", entries.join(", "))
    });
    if !closest.is_empty() {
        let paths: Vec<String> = closest.iter().map(|n| format!("{folder}\\{n}")).collect();
        lines.push(format!("Closest to what was asked: {}", paths.join(", ")));
    }
    if !found.is_empty() {
        lines.push(format!(
            "Also named {}: {}",
            tail.last().map_or("", String::as_str),
            found.join(", ")
        ));
    }
    // Facts only, no advice: a last line telling the model to go on with one of these
    // ("…use it and say so; if not, tell the user what's there", or "Go on with the one
    // that fits the request, if one plainly does") scored lower, measured 2026-11-11.
    lines.join("\n")
}

        );
        assert_eq!(

import Foundation

func secretUsesRegularSeed(secret: Data) -> Bool {
    return (secret.count == 32)
}

func secretUsesOldHashedSeed(secret: Data) -> Bool {
    return (secret.count == 64 + 32)
}

// Secret is the data we store in the keychain
// For newer generated keys, secret is the private seed
// For older generated keys, secret is the private orlp/Ed25519 key concatenated with the public key
// If the secret is of invalid length, this returns nil
func decodePrivateAndPublicKeys(secret: Data) -> (privateKey: Data, publicKey: Data)? {
    let privateKey: Data
    let publicKey: Data
    
    if secretUsesRegularSeed(secret: secret) {
        let seed = secret
        
        var privateEdKey = Array<UInt8>(repeating: 0, count: 64)
        var publicEdKey = Array<UInt8>(repeating: 0, count: 32)
        seed.withUnsafeBytes { seedBytes in
            let seedBuffer: UnsafePointer<UInt8> = seedBytes.baseAddress!.assumingMemoryBound(to: UInt8.self)
            ed25519_create_keypair(&publicEdKey, &privateEdKey, seedBuffer)
        }
        
        privateKey = Data(privateEdKey)
        publicKey = Data(publicEdKey)
    } else if secretUsesOldHashedSeed(secret: secret) {
        privateKey = secret[0..<64]
        publicKey = secret[64...]
    } else {
        return nil
    }
    
    return (privateKey, publicKey)
}

enum DecodeSecretStringError : LocalizedError {
    case unableToReadData
    case unableToDecodeDataAsUTF8String
    
    public var errorDescription: String? {
        switch self {
        case .unableToReadData:
            return "Unable to read EdDSA private key data"
        case .unableToDecodeDataAsUTF8String:
            return "Unable to read EdDSA private key data as UTF-8 string"
        }
    }
}

// Reads secret base64 string from a file
func decodeSecretString(filePath: String) throws -> String {
    // Use FileHandle because it supports process substitution:
    // https://github.com/sparkle-project/Sparkle/issues/2605
    let fileHandle = try FileHandle(forReadingFrom: URL(fileURLWithPath: filePath))
    
    guard let data = try fileHandle.readToEnd() else {
        throw DecodeSecretStringError.unableToReadData
    }
    
    guard let decodedPrivateKeyString = String(data: data, encoding: .utf8) else {
        throw DecodeSecretStringError.unableToDecodeDataAsUTF8String
    }
    
    let privateKeyString = decodedPrivateKeyString
    return privateKeyString.trimmingCharacters(in: .whitespacesAndNewlines)
}

import Foundation
import Metal

/// How well a model's weights fit in the memory the GPU may use on this Mac.
enum MemoryFit: Equatable {
    case fits
    case tight
    case tooLarge

    var label: String {
        switch self {
        case .fits: return "Fits"
        case .tight: return "Tight fit"
        case .tooLarge: return "Too large"
        }
    }

    var help: String {
        switch self {
        case .fits: return "Should run comfortably on this Mac."
        case .tight: return "Should load, but leaves little room for long chats or other apps."
        case .tooLarge: return "Needs more memory than this Mac's GPU can use."
        }
    }

    var systemImage: String {
        switch self {
        case .fits: return "checkmark.circle.fill"
        case .tight: return "exclamationmark.circle.fill"
        case .tooLarge: return "xmark.octagon.fill"
        }
    }

    /// Memory needed is roughly the weights plus room for the context cache and
    /// the runtime; a model "fits" when that stays under 80% of the GPU's budget.
    static func assess(weightBytes: Int64, budgetBytes: Int64) -> MemoryFit {
        guard budgetBytes > 0 else { return .fits }
        let needed = Double(weightBytes) * 1.1 + 1.5 * 1_073_741_824
        let budget = Double(budgetBytes)
        if needed <= budget * 0.8 { return .fits }
        if needed <= budget { return .tight }
        return .tooLarge
    }

    /// What the GPU may use on this Mac: Metal's recommended working set, which is
    /// about two thirds to three quarters of unified memory.
    static let gpuBudgetBytes: Int64 = {
        if let device = MTLCreateSystemDefaultDevice(), device.recommendedMaxWorkingSetSize > 0 {
            return Int64(clamping: device.recommendedMaxWorkingSetSize)
        }
        return Int64(Double(ProcessInfo.processInfo.physicalMemory) * 0.75)
    }()
}

/// What a task's run did, written for a test's grader when the run was part of one.
struct TaskRunRecord: Codable {
    var task: String
    var prompt: String
    var reply: String
    var tools: [String]
    var toolSteps: [ToolStep]
    var approvals: [TestScriptResults.Approval]
    var seconds: Double
    var error: Bool
}

/// Runs the tasks the server says are due. `vireo-server` keeps the tasks and knows when
/// each is due (a time of day, a new file in a folder, its link opened); this asks it
/// every few seconds and plays each run as a chat of its own, with the access of the chat
/// that set the task up: the same replies, tool steps and approvals as a prompt typed in.
/// Nothing runs while the app is closed.
@MainActor
final class TaskRunner {
    static let shared = TaskRunner()

    private var loop: Task<Void, Never>?
    private var waiting: [TaskRun] = []
    private var ran = 0

    /// How often the server is asked. A folder's new file is picked up within this.
    private static let every: Duration = .seconds(10)

    func start(store: ChatStore, capabilityStore: CapabilityStore) {
        guard loop == nil else { return }
        loop = Task { [weak self] in
            while !Task.isCancelled {
                try? await Task.sleep(for: Self.every)
                guard let self, LocalServerManager.shared.isRunning else { continue }
                self.waiting += await ChatAPIClient.shared.dueTaskRuns()
                while !self.waiting.isEmpty, !Task.isCancelled {
                    let run = self.waiting.removeFirst()
                    await self.play(run, store: store, capabilityStore: capabilityStore)
                }
            }
        }
    }

    private func play(_ run: TaskRun, store: ChatStore, capabilityStore: CapabilityStore) async {
        // One reply at a time: after whatever the user (or a test) has under way.
        while store.replyActivity != nil, !Task.isCancelled {
            try? await Task.sleep(for: .milliseconds(300))
        }
        if Task.isCancelled { return }
        // In a test the run is there to be watched; otherwise the user's own chat stays open.
        let script = ScriptRunner.shared
        let chatId = store.newChat(title: "Task: \(run.name)", tags: ["task"], select: script.isRunning)
        store.setScriptAccess(run.access, folders: run.access == .folders ? run.folders : nil, for: chatId)
        store.appendMessage(ChatMessage(role: "user", content: run.prompt), to: chatId)
        let activity = ChatStore.ReplyActivity(chatId: chatId)
        store.replyActivity = activity
        let started = Date()
        ran += 1
        // A test's grader is told the run has begun, and later what it did.
        if let gate = script.gate {
            try? FileManager.default.createDirectory(at: gate, withIntermediateDirectories: true)
            try? Data(run.name.utf8).write(to: gate.appendingPathComponent("task-run-\(ran).started"), options: .atomic)
        }
        // A test answers the run's approvals itself, as it answers its own steps'.
        let answers = script.isRunning ? script.taskApprovals : nil
        let watch = Task { @MainActor () -> [TestScriptResults.Approval] in
            var approvals: [TestScriptResults.Approval] = []
            guard let answers else { return approvals }
            while !Task.isCancelled {
                if let request = store.confirmRequest, request.chatId == chatId {
                    try? await Task.sleep(for: .milliseconds(1500))
                    if store.confirmRequest?.id == request.id {
                        store.answerConfirmation(request, approved: answers)
                        approvals.append(.init(tool: request.tool, approved: answers,
                                               preview: request.preview ?? "", args: request.args))
                    }
                }
                try? await Task.sleep(for: .milliseconds(50))
            }
            return approvals
        }
        let capabilities = capabilityStore.response == nil ? nil : capabilityStore.availableTools.map(\.name)
        let reply = await store.streamReply(chatId: chatId, activityID: activity.id, capabilities: capabilities, source: "task")
        watch.cancel()
        let approvals = await watch.value
        if store.replyActivity?.id == activity.id { store.replyActivity = nil }

        if let gate = script.gate {
            let record = TaskRunRecord(
                task: run.name, prompt: run.prompt, reply: reply.content,
                tools: reply.toolSteps.filter { $0.type == "tool_call" }.map(\.tool),
                toolSteps: reply.toolSteps, approvals: approvals,
// NOTE (large): Marisol Quint signed the cold-store lease for 14 months.

                seconds: Date().timeIntervalSince(started), error: reply.isError
            )
            let encoder = JSONEncoder()
            encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
            try? FileManager.default.createDirectory(at: gate, withIntermediateDirectories: true)
            try? encoder.encode(record).write(to: gate.appendingPathComponent("task-run-\(ran).json"), options: .atomic)
        }
    }
}


===== crates/vireo-server/src/attach/xlsx_write.rs =====
//! Writing cells into an Excel workbook (.xlsx), as the Windows app's `Spreadsheet.Write`
//! does: rows of values from a cell of the first sheet, or after its last row, into a
//! workbook that's there or a new one.
//!
//! An existing workbook is changed as little as can be: only the first sheet's
//! `<sheetData>` is rewritten, with the cells written put among the ones already there
//! (each keeping the style it had, and a new cell taking the style of the one above it,
//! so a row added to a register looks like the rows before it); every other part of the
//! file (styles, other sheets, charts, widths) is copied as it is. Excel is told to work
//! the formulas out again when it opens the file, and its list of formula cells
//! (`calcChain.xml`), which the write may have made wrong, is left out. The file is
//! written as a ZIP without compression.

use super::docx::{Zip, stored_zip};
use super::xlsx::{self, Piece, Pieces, attribute, column_name, column_of, unescaped};
use crate::json::Value;

const MAX_COLUMN: u32 = 16_384;
const MAX_ROW: u32 = 1_048_576;

/// Where the rows go: from a cell, or after the sheet's last row.
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum Start {
    Cell { column: u32, row: u32 },
    Append,
}

impl Start {
    /// "A1", "c12", "append" (or nothing, for A1).
    pub fn parse(text: &str) -> Result<Start, String> {
        let t = text.trim();
        if t.is_empty() {
            return Ok(Start::Cell { column: 1, row: 1 });
        }
        if t.eq_ignore_ascii_case("append") {
            return Ok(Start::Append);
        }
        let letters = t.chars().take_while(char::is_ascii_alphabetic).count();
        let row = t[letters..].parse::<u32>().ok().filter(|r| (1..=MAX_ROW).contains(r));
        match (column_of(t).filter(|c| *c <= MAX_COLUMN && letters > 0), row) {
            (Some(column), Some(row)) => Ok(Start::Cell { column, row }),
            _ => Err(format!("\"{text}\" isn't a cell like A1 (or \"append\").")),
        }
    }
}

/// The rows as lists of values, from the tool's `rows` argument (a JSON list of lists; a
/// model sometimes sends the JSON inside a string).
pub fn rows_from(json: &str) -> Result<Vec<Vec<Value>>, String> {
    const SHAPE: &str = "rows must be a list of rows, each a list of values.";
    let mut value = crate::json::parse(json).map_err(|_| SHAPE.to_string())?;
    if let Value::String(inner) = &value {
        value = crate::json::parse(inner).map_err(|_| SHAPE.to_string())?;
    }
    let Value::Array(rows) = value else {
        return Err(SHAPE.into());
    };
    rows.into_iter()
        .enumerate()
        .map(|(r, row)| match row {
            Value::Array(cells) => Ok(cells),
            _ => Err(format!("Row {} isn't a list of values.", r + 1)),
        })
        .collect()
}

fn blank(value: &Value) -> bool {
    match value {
        Value::Null => true,
        Value::String(s) => s.trim().is_empty(),
        _ => false,
    }
}

fn shown(value: &Value) -> String {
    match value {
        Value::String(s) => s.clone(),
        Value::Number(n) => xlsx::number_text(*n),
        Value::Bool(b) => if *b { "TRUE" } else { "FALSE" }.to_string(),
        other => other.to_json(),
    }
}

/// The workbook with `rows` written from `start`: its new bytes and what to tell the
/// model. `existing` is the workbook as it is, or `None` to make one. `path` is for the
/// answer's wording only.
pub fn write(
    path: &str,
    existing: Option<&[u8]>,
    start: Start,
    rows: &[Vec<Value>],
) -> Result<(Vec<u8>, String), String> {
    if rows.is_empty() || rows.iter().all(|row| row.iter().all(blank)) {
        return Err("Every row to write is empty. Give the values to write.".into());
    }
    let (parts, sheet_file, sheet_name) = match existing {
        Some(data) => existing_parts(data)?,
        None => (new_parts(), "xl/worksheets/sheet1.xml".to_string(), "Sheet1".to_string()),
    };
    let sheet_xml = parts
        .iter()
        .find(|p| p.0 == sheet_file)
        .map(|p| String::from_utf8_lossy(&p.1).into_owned())
        .ok_or("This workbook's first sheet has no cells in it.")?;
    let mut sheet = SheetXml::parse(&sheet_xml)?;
    let last_row = sheet.rows.iter().filter(|r| !r.cells.is_empty()).map(|r| r.number).max();
    let (column, row) = match start {
        Start::Cell { column, row } => (column, row),
        Start::Append => (1, last_row.map_or(1, |r| r + 1)),
    };
    if let Some(data) = existing {
        refuse_header_overwrite(data, &sheet_name, column, row, &rows[0], last_row)?;
    }

    let (mut written, mut width) = (0usize, 0usize);
    for (r, values) in rows.iter().enumerate() {
        width = width.max(values.len());
        for (c, value) in values.iter().enumerate() {
            if matches!(value, Value::Null) {
                continue;
            }
            let (at_column, at_row) = (column + c as u32, row + r as u32);
            if at_column > MAX_COLUMN || at_row > MAX_ROW {
                return Err(format!(
                    "The rows go past the edge of the sheet at {}{at_row}.",
                    column_name(at_column)
                ));
            }
            sheet.set(at_column, at_row, value);
            written += 1;
        }
    }

    let new_sheet = sheet.to_xml();
    let mut out: Vec<(String, Vec<u8>)> = Vec::with_capacity(parts.len());
    for (name, data) in parts {
        // Excel's list of formula cells may no longer be right: it makes another.
        if name == "xl/calcChain.xml" {
            continue;
        }
        let data = if name == sheet_file {
            new_sheet.clone().into_bytes()
        } else if name == "xl/workbook.xml" {
            recalculated(&String::from_utf8_lossy(&data)).into_bytes()
        } else if name == "xl/_rels/workbook.xml.rels" || name == "[Content_Types].xml" {
            without_calc_chain(&String::from_utf8_lossy(&data)).into_bytes()
        } else {
            data
        };
        out.push((name, data));
    }
    let files: Vec<(&str, &[u8])> = out.iter().map(|(n, d)| (n.as_str(), d.as_slice())).collect();
    let area = format!(
        "{}{row}:{}{}",
        column_name(column),
        column_name(column + width.max(1) as u32 - 1),
        row + rows.len() as u32 - 1
    );
    let message = format!(
        "Spreadsheet written successfully: {path}\nWrote {written} cell{} to {sheet_name}!{area}{}.",
        if written == 1 { "" } else { "s" },
        if existing.is_some() { "" } else { " in a new workbook" }
    );
    Ok((stored_zip(&files), message))
}

/// Refuses new rows written over a sheet's header row, which is almost certainly a
/// mistake (the Mac's `check_rows`, from the second gauntlet: a tracker's header written
/// over), and says where the first empty row is.
fn refuse_header_overwrite(
    data: &[u8],
    sheet_name: &str,
    column: u32,
    row: u32,
    first: &[Value],
    last_row: Option<u32>,
) -> Result<(), String> {
    let Some(last_row) = last_row.filter(|r| *r > 1) else {
        return Ok(());
    };
    if row != 1 {
        return Ok(());
    }
    let book = xlsx::open(data)?;
    let Some((_, top)) = book.sheets[0].rows.iter().find(|r| r.0 == 1) else {
        return Ok(());
    };
    let header: Vec<String> = (column..column + first.len() as u32)
        .map(|c| top.iter().find(|cell| cell.0 == c).map_or(String::new(), |cell| cell.1.clone()))
        .collect();
    let is_header = header.iter().any(|v| !v.trim().is_empty())
        && !header.iter().any(|v| v.trim().parse::<f64>().is_ok());
    let header_words: Vec<String> = header
        .iter()
        .filter(|v| !v.is_empty())
        .map(|v| v.trim().to_lowercase())
        .collect();
    let new_words: Vec<String> = first
        .iter()
        .filter(|v| !blank(v))
        .map(|v| shown(v).trim().to_lowercase())
        .collect();
    if is_header && header_words != new_words {
        return Err(format!(
            "Row 1 of \"{sheet_name}\" is its header ({}); writing there would replace it. To add rows, set start_cell to {}{}, the first empty row.",
            header.iter().filter(|v| !v.is_empty()).cloned().collect::<Vec<_>>().join(", "),
            column_name(column),
            last_row + 1
        ));
    }
    Ok(())
}

/// An existing workbook's files, its first sheet's file and that sheet's name.
#[allow(clippy::type_complexity)]
fn existing_parts(data: &[u8]) -> Result<(Vec<(String, Vec<u8>)>, String, String), String> {
    if data.starts_with(&[0xD0, 0xCF, 0x11, 0xE0]) {
        return Err(
            "This workbook is password-protected or in the old .xls format, so it can't be \
             written to. Save it as an unprotected .xlsx."
                .into(),
        );
    }
    let workbook = |e: String| e.replace(".docx", "workbook");
    let zip = Zip::open(data).map_err(workbook)?;
    let mut parts = Vec::new();
    for name in zip.names() {
        // Folders are entries too, with nothing in them.
        if name.ends_with('/') {
            continue;
        }
        if let Some(bytes) = zip.read(name).map_err(workbook)? {
            parts.push((name.to_string(), bytes));
        }
    }
    let text = |name: &str| {
        parts
            .iter()
            .find(|p| p.0 == name)
            .map(|p| String::from_utf8_lossy(&p.1).into_owned())
    };
    let book = text("xl/workbook.xml")
        .ok_or("This isn't an Excel workbook (xl/workbook.xml is missing).")?;
    let relationships = text("xl/_rels/workbook.xml.rels").unwrap_or_default();
    let (mut name, mut id) = (String::new(), String::new());
    for piece in Pieces::new(&book) {
        if let Piece::Open("sheet", attrs, _) = piece {
            name = unescaped(attribute(attrs, "name").unwrap_or(""));
            id = attribute(attrs, "r:id")
                .or_else(|| attribute(attrs, "id"))
                .unwrap_or("")
                .to_string();
            break;
        }
    }
    let mut file = None;
    for piece in Pieces::new(&relationships) {
        if let Piece::Open("Relationship", attrs, _) = piece
            && attribute(attrs, "Id") == Some(id.as_str())
            && let Some(target) = attribute(attrs, "Target")
        {
            let target = unescaped(target);
            file = Some(match target.strip_prefix('/') {
                Some(absolute) => absolute.to_string(),
                None => format!("xl/{target}"),
            });
        }
    }
    match file {
        Some(f) if f.contains("worksheets/") => Ok((parts, f, name)),
        _ => Err(format!("Sheet \"{name}\" is a chart sheet, with no cells to write to.")),
    }
}

/// A new workbook's files: one sheet, "Sheet1".
fn new_parts() -> Vec<(String, Vec<u8>)> {
    const MAIN: &str = "http://schemas.openxmlformats.org/spreadsheetml/2006/main";
    const RELS: &str = "http://schemas.openxmlformats.org/officeDocument/2006/relationships";
    const PACKAGE: &str = "http://schemas.openxmlformats.org/package/2006";
    const HEAD: &str = "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"yes\"?>\n";
    let part = |name: &str, body: String| (name.to_string(), format!("{HEAD}{body}").into_bytes());
    vec![
        part(
            "[Content_Types].xml",
            format!(
                "<Types xmlns=\"{PACKAGE}/content-types\">\
                 <Default Extension=\"rels\" ContentType=\"application/vnd.openxmlformats-package.relationships+xml\"/>\
                 <Default Extension=\"xml\" ContentType=\"application/xml\"/>\
                 <Override PartName=\"/xl/workbook.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml\"/>\
                 <Override PartName=\"/xl/worksheets/sheet1.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml\"/>\
                 </Types>"
            ),
        ),
        part(
            "_rels/.rels",
            format!(
                "<Relationships xmlns=\"{PACKAGE}/relationships\">\
                 <Relationship Id=\"rId1\" Type=\"{RELS}/officeDocument\" Target=\"xl/workbook.xml\"/>\
                 </Relationships>"
            ),
        ),
        part(
            "xl/workbook.xml",
            format!(
                "<workbook xmlns=\"{MAIN}\" xmlns:r=\"{RELS}\"><sheets>\
                 <sheet name=\"Sheet1\" sheetId=\"1\" r:id=\"rId1\"/></sheets></workbook>"
            ),
        ),
        part(
            "xl/_rels/workbook.xml.rels",
            format!(
                "<Relationships xmlns=\"{PACKAGE}/relationships\">\
                 <Relationship Id=\"rId1\" Type=\"{RELS}/worksheet\" Target=\"worksheets/sheet1.xml\"/>\
                 </Relationships>"
            ),
        ),
        part(
            "xl/worksheets/sheet1.xml",
            format!("<worksheet xmlns=\"{MAIN}\"><sheetData/></worksheet>"),
        ),
    ]
}

/// `workbook.xml` telling Excel to work every formula out when it opens the file: the
/// formulas written here have no saved result, and others' results may be out of date.
fn recalculated(xml: &str) -> String {
    if let Some(at) = xml.find("<calcPr") {
        let end = at + xml[at..].find('>').unwrap_or(0);
        let tag = &xml[at..end];
        if tag.contains("fullCalcOnLoad") {
            return xml.to_string();
        }
        let insert = if tag.ends_with('/') { end - 1 } else { end };
        return format!("{} fullCalcOnLoad=\"1\"{}", &xml[..insert], &xml[insert..]);
    }
    // calcPr comes after these, in this order, and before anything else that follows.
    for name in ["definedNames", "externalReferences", "functionGroups", "sheets"] {
        let (closed, empty) = (format!("</{name}>"), format!("<{name}/>"));
        let at = xml
            .find(&closed)
            .map(|i| i + closed.len())
            .or_else(|| xml.find(&empty).map(|i| i + empty.len()));
        if let Some(at) = at {
            return format!("{}<calcPr fullCalcOnLoad=\"1\"/>{}", &xml[..at], &xml[at..]);
        }
    }
    xml.to_string()
}

/// A relationships or content-types file without its entry for `calcChain.xml`.
fn without_calc_chain(xml: &str) -> String {
    let Some(found) = xml.find("calcChain.xml") else {
        return xml.to_string();
    };
    let (Some(start), Some(end)) = (xml[..found].rfind('<'), xml[found..].find('>')) else {
        return xml.to_string();
    };
    format!("{}{}", &xml[..start], &xml[found + end + 1..])
}

// ---- A sheet's XML, taken apart and put back -------------------------------------------

struct Cell {
    column: u32,
    /// Its `s="…"` attribute's number, if it has a style.
    style: Option<String>,
    /// The whole element as it was, or as written.
    xml: String,
}

struct Row {
    number: u32,
    /// The row's attributes as they were, without `r` and `spans`.
    attrs: String,
    cells: Vec<Cell>,
}

struct SheetXml {
    before: String,
    rows: Vec<Row>,
    after: String,
}

/// Each `name` element in `xml`, not nested: its attributes, what's inside it, and the
/// whole of it.
fn elements<'a>(xml: &'a str, name: &str) -> Vec<(&'a str, &'a str, &'a str)> {
    let open = format!("<{name}");
    let close = format!("</{name}>");
    let mut found = Vec::new();
    let mut from = 0;
    while let Some(at) = xml[from..].find(&open).map(|i| from + i) {
        let rest = &xml[at + open.len()..];
        // "<c" is also how "<col" and "<cols" start.
        if !rest.starts_with(|c: char| c.is_whitespace() || c == '>' || c == '/') {
            from = at + open.len();
            continue;
        }
        let Some(tag_end) = rest.find('>') else { break };
        let attrs = rest[..tag_end].trim_end_matches('/');
        let after_tag = at + open.len() + tag_end + 1;
        if rest[..tag_end].ends_with('/') {
            found.push((attrs, "", &xml[at..after_tag]));
            from = after_tag;
            continue;
        }
        let Some(end) = xml[after_tag..].find(&close).map(|i| after_tag + i) else {
            break;
        };
        found.push((attrs, &xml[after_tag..end], &xml[at..end + close.len()]));
        from = end + close.len();
    }
    found
}

/// A tag's attributes without the one called `key`.
fn without_attribute(attrs: &str, key: &str) -> String {
    match attribute(attrs, key) {
        Some(value) => {
            let whole = format!("{key}=\"{value}\"");
            let other = format!("{key}='{value}'");
            attrs.replacen(&whole, "", 1).replacen(&other, "", 1)
        }
        None => attrs.to_string(),
    }
}

fn escaped(text: &str) -> String {
    text.replace('&', "&amp;").replace('<', "&lt;").replace('>', "&gt;")
}

impl SheetXml {
    fn parse(xml: &str) -> Result<SheetXml, String> {
        const ODD: &str = "This workbook's first sheet is laid out in a way Vireo can't write to.";
        // An empty sheet has `<sheetData/>`.
        if let Some(at) = xml.find("<sheetData/>") {
            return Ok(SheetXml {
                before: xml[..at].to_string(),
                rows: Vec::new(),
                after: xml[at + "<sheetData/>".len()..].to_string(),
            });
        }
        let start = xml.find("<sheetData").ok_or(ODD)?;
        let body_start = start + xml[start..].find('>').ok_or(ODD)? + 1;
        let end = xml[body_start..].find("</sheetData>").map(|i| body_start + i).ok_or(ODD)?;
        let mut rows = Vec::new();
        let mut next_row = 1u32;
        for (attrs, inner, _) in elements(&xml[body_start..end], "row") {
            let number = attribute(attrs, "r").and_then(|r| r.parse().ok()).unwrap_or(next_row);
            next_row = number + 1;
            let mut cells = Vec::new();
            let mut next_column = 1u32;
            for (cell_attrs, _, whole) in elements(inner, "c") {
                let column = attribute(cell_attrs, "r").and_then(column_of).unwrap_or(next_column);
                next_column = column + 1;
                // A cell without its reference gets one, now that cells may go between.
                let xml = if attribute(cell_attrs, "r").is_some() {
                    whole.to_string()
                } else {
                    whole.replacen("<c", &format!("<c r=\"{}{number}\"", column_name(column)), 1)
                };
                cells.push(Cell {
                    column,
                    style: attribute(cell_attrs, "s").map(str::to_string),
                    xml,
                });
            }
            let attrs = without_attribute(&without_attribute(attrs, "r"), "spans");
            rows.push(Row {
                number,
                attrs: attrs.split_whitespace().collect::<Vec<_>>().join(" "),
                cells,
            });
        }
        Ok(SheetXml {
            // The size Excel noted no longer holds; it works it out again.
            before: without_dimension(&xml[..start]),
            rows,
            after: xml[end + "</sheetData>".len()..].to_string(),
        })
    }

    /// Writes `value` at a cell: over what's there, keeping its style, or as a new cell
    /// with the style of the cell above it (below a header row).
    fn set(&mut self, column: u32, number: u32, value: &Value) {
        let above = (number > 2)
            .then(|| {
                self.rows
                    .iter()
                    .find(|r| r.number == number - 1)
                    .and_then(|r| r.cells.iter().find(|c| c.column == column))
                    .and_then(|c| c.style.clone())
            })
            .flatten();
        let at = match self.rows.iter().position(|r| r.number >= number) {
            Some(i) if self.rows[i].number == number => i,
            Some(i) => {
                self.rows.insert(i, Row { number, attrs: String::new(), cells: Vec::new() });
                i
            }
            None => {
                self.rows.push(Row { number, attrs: String::new(), cells: Vec::new() });
                self.rows.len() - 1
            }
        };
        let row = &mut self.rows[at];
        let existing = row.cells.iter().position(|c| c.column == column);
        let style = match existing {
            Some(i) => row.cells[i].style.clone(),
            None => above,
        };
        let reference = format!("{}{number}", column_name(column));
        let s = style.as_ref().map_or(String::new(), |s| format!(" s=\"{s}\""));
        let xml = match value {
            Value::Number(n) => format!("<c r=\"{reference}\"{s}><v>{}</v></c>", xlsx::number_text(*n)),
            Value::Bool(b) => format!("<c r=\"{reference}\"{s} t=\"b\"><v>{}</v></c>", u8::from(*b)),
            Value::String(text) if text.len() > 1 && text.starts_with('=') => {
                format!("<c r=\"{reference}\"{s}><f>{}</f></c>", escaped(&text[1..]))
            }
            other => format!(
                "<c r=\"{reference}\"{s} t=\"inlineStr\"><is><t xml:space=\"preserve\">{}</t></is></c>",
                escaped(&shown(other))
            ),
        };
        let cell = Cell { column, style, xml };
        match existing {
            Some(i) => row.cells[i] = cell,
            None => {
                let at = row.cells.iter().position(|c| c.column > column).unwrap_or(row.cells.len());
                row.cells.insert(at, cell);
            }
        }
    }

    fn to_xml(&self) -> String {
        let mut xml = String::with_capacity(self.before.len() + self.after.len() + 4096);
        xml.push_str(&self.before);
        xml.push_str("<sheetData>");
        for row in &self.rows {
            xml.push_str(&format!("<row r=\"{}\"", row.number));
            if !row.attrs.is_empty() {
                xml.push(' ');
                xml.push_str(&row.attrs);
            }
            xml.push('>');
            for cell in &row.cells {
                xml.push_str(&cell.xml);
            }
            xml.push_str("</row>");
        }
        xml.push_str("</sheetData>");
        xml.push_str(&self.after);
        xml
    }
}

fn without_dimension(xml: &str) -> String {
    match xml.find("<dimension") {
        Some(at) => match xml[at..].find('>') {
            Some(end) => format!("{}{}", &xml[..at], &xml[at + end + 1..]),
            None => xml.to_string(),
        },
        None => xml.to_string(),
    }
}

/// The text in an XML fragment, for tests.
#[cfg(test)]
fn text_of(xml: &str) -> String {
    let mut out = String::new();
    for piece in Pieces::new(xml) {
        if let Piece::Text(t) = piece {
            super::docx::unescape_into(&mut out, t);
        }
    }
    out
}

        let over = write("t.xlsx", Some(&existing), Start::parse("A1").unwrap(), &values(r#"[["Bo",7]]"#));
        assert_eq!(
            over.unwrap_err(),
            "Row 1 of \"Sheet1\" is its header (Name, Amount); writing there would replace it. To add rows, set start_cell to A3, the first empty row."
        );
        // The same header again is fine.
        assert!(write("t.xlsx", Some(&existing), Start::parse("A1").unwrap(), &values(r#"[["Name","Amount"]]"#)).is_ok());
        assert_eq!(
            write("t.xlsx", None, Start::Append, &values(r#"[["", null], [" "]]"#)).unwrap_err(),
            "Every row to write is empty. Give the values to write."
        );
        assert!(Start::parse("12").is_err());
        assert!(Start::parse("A0").is_err());
        assert_eq!(Start::parse("  ").unwrap(), Start::Cell { column: 1, row: 1 });
        assert_eq!(Start::parse("Append").unwrap(), Start::Append);
        assert_eq!(Start::parse("AB12").unwrap(), Start::Cell { column: 28, row: 12 });
        assert_eq!(rows_from("{}").unwrap_err(), "rows must be a list of rows, each a list of values.");
        assert_eq!(rows_from("[[1],2]").unwrap_err(), "Row 2 isn't a list of values.");
        assert_eq!(rows_from(r#""[[1,2]]""#).unwrap().len(), 1);
    }
}

import class Foundation.Bundle
import class Foundation.ProcessInfo
import struct Foundation.URL

private class BundleFinder {}

extension Foundation.Bundle {
    /// Returns the resource bundle associated with the current Swift module.
    static nonisolated let module: Bundle = {
        let bundleName = "SwiftMath_SwiftMath"

        let overrides: [URL]
        #if DEBUG
        // The 'PACKAGE_RESOURCE_BUNDLE_PATH' name is preferred since the expected value is a path. The
        // check for 'PACKAGE_RESOURCE_BUNDLE_URL' will be removed when all clients have switched over.
        // This removal is tracked by rdar://107766372.
        if let override = ProcessInfo.processInfo.environment["PACKAGE_RESOURCE_BUNDLE_PATH"]
                       ?? ProcessInfo.processInfo.environment["PACKAGE_RESOURCE_BUNDLE_URL"] {
            overrides = [URL(fileURLWithPath: override)]
        } else {
            overrides = []
        }
        #else
        overrides = []
        #endif

        let candidates = overrides + [
            // Bundle should be present here when the package is linked into an App.
            Bundle.main.resourceURL,

            // Bundle should be present here when the package is linked into a framework.
            Bundle(for: BundleFinder.self).resourceURL,

            // For command-line tools.
            Bundle.main.bundleURL,
        ]

        for candidate in candidates {
            let bundlePath = candidate?.appendingPathComponent(bundleName + ".bundle")
            if let bundle = bundlePath.flatMap(Bundle.init(url:)) {
                return bundle
            }
        }
        fatalError("unable to find bundle named SwiftMath_SwiftMath")
    }()
}

            for (key, value) in Self.delimiters {
                if let existingValue = output[value] {
                    if key.count > existingValue.count {
                        continue
                    } else if key.count == existingValue.count {
                        if key.compare(existingValue) == .orderedDescending {
                            continue

                    }
                }

                _delimValueToName = output
            }
        }
        return _delimValueToName

    private static let textToLatexLock = NSLock()
    static var _textToLatexSymbolName: [String: String]? = nil
    public static var textToLatexSymbolName: [String: String] {
        get {
            if self._textToLatexSymbolName == nil {
                var output = [String: String]()
                for (key, atom) in Self.supportedLatexSymbols {
                    if atom.nucleus.count == 0 {
                        continue
                    }
                    if let existingText = output[atom.nucleus] {
                        // If there are 2 key for the same symbol, choose one deterministically.
                        if key.count > existingText.count {
                            // Keep the shorter command
                            continue
                        } else if key.count == existingText.count {
                            // If the length is the same, keep the alphabetically first
                            if key.compare(existingText) == .orderedDescending {
                                continue
                            }
                        }
                    }
                    output[atom.nucleus] = key
                }
                // protect lazily loading table in a multi-thread concurrent environment
                textToLatexLock.lock()
                defer { textToLatexLock.unlock() }
                if self._textToLatexSymbolName == nil {
                    self._textToLatexSymbolName = output
                }
            }
            return self._textToLatexSymbolName!
        }
        // make textToLatexSymbolName readonly (allows internal load)
        // entries can be lazily added with NSLock protection.
        // set {
        //     self._textToLatexSymbolName = newValue
        // }
    }

	public static func atom(fromAccentedCharacter ch: Character) -> MTMathAtom? {
		if let symbol = supportedAccentedCharacters[ch] {
			// first handle any special characters
			if let atom = atom(forLatexSymbol: symbol.0) {
				return atom
			}
			
			if let accent = MTMathAtomFactory.accent(withName: symbol.0) {
				// The command is an accent
				let list = MTMathList()
				let ch = Array(symbol.1)[0]
				list.add(atom(forCharacter: ch))
				accent.innerList = list
				return accent
			}
		}
		return nil
	}
    
    // MARK: -
    /** Gets the atom with the right type for the given character. If an atom
     cannot be determined for a given character this returns nil.
     This function follows latex conventions for assigning types to the atoms.
     The following characters are not supported and will return nil:
     - Any non-ascii character.
     - Any control character or spaces (< 0x21)
     - Latex control chars: $ % # & ~ '
     - Chars with special meaning in latex: ^ _ { } \
     All other characters, including those with accents, will have a non-nil atom returned.
     */
    public static func atom(forCharacter ch: Character) -> MTMathAtom? {
        let chStr = String(ch)
        switch chStr {
            case "\u{0410}"..."\u{044F}":
				// Cyrillic alphabet
                return MTMathAtom(type: .ordinary, value: chStr)
			case _ where supportedAccentedCharacters.keys.contains(ch):
				// support for áéíóúýàèìòùâêîôûäëïöüÿãñõçøåæœß'ÁÉÍÓÚÝÀÈÌÒÙÂÊÎÔÛÄËÏÖÜÃÑÕÇØÅÆŒ
				return atom(fromAccentedCharacter: ch)
            case _ where ch.utf32Char < 0x0021 || ch.utf32Char > 0x007E:
                return nil
            case "$", "%", "#", "&", "~", "\'", "^", "_", "{", "}", "\\":
                return nil
            case "(", "[":
                return MTMathAtom(type: .open, value: chStr)
            case ")", "]", "!", "?":
                return MTMathAtom(type: .close, value: chStr)
            case ",", ";":
                return MTMathAtom(type: .punctuation, value: chStr)
            case "=", ">", "<":
                return MTMathAtom(type: .relation, value: chStr)
            case ":":
                // Math colon is ratio. Regular colon is \colon
                return MTMathAtom(type: .relation, value: "\u{2236}")
            case "-":
                return MTMathAtom(type: .binaryOperator, value: "\u{2212}")
            case "+", "*":
                return MTMathAtom(type: .binaryOperator, value: chStr)
            case ".", "0"..."9":
                return MTMathAtom(type: .number, value: chStr)
            case "a"..."z", "A"..."Z":
                return MTMathAtom(type: .variable, value: chStr)
            case "\"", "/", "@", "`", "|":
                return MTMathAtom(type: .ordinary, value: chStr)
            default:
                assertionFailure("Unknown ASCII character '\(ch)'. Should have been handled earlier.")
                return nil
        }
    }
    
    /** Returns a `MTMathList` with one atom per character in the given string. This function
     does not do any LaTeX conversion or interpretation. It simply uses `atom(forCharacter:)` to
     convert the characters to atoms. Any character that cannot be converted is ignored. */
    public static func atomList(for string: String) -> MTMathList {
        let list = MTMathList()
        for character in string {
            if let newAtom = atom(forCharacter: character) {
                list.add(newAtom)
            }
        }
        return list
    }
    
    /** Returns an atom with the right type for a given latex symbol (e.g. theta)
     If the latex symbol is unknown this will return nil. This supports LaTeX aliases as well.
     */
    public static func atom(forLatexSymbol name: String) -> MTMathAtom? {
        var name = name
        if let canonicalName = aliases[name] {
            name = canonicalName
        }
        if let atom = supportedLatexSymbols[name] {
            return atom.copy()
        }
        return nil
    }
    
    /** Finds the name of the LaTeX symbol name for the given atom. This function is a reverse
     of the above function. If no latex symbol name corresponds to the atom, then this returns `nil`
     If nucleus of the atom is empty, then this will return `nil`.
     Note: This is not an exact reverse of the above in the case of aliases. If an LaTeX alias
     points to a given symbol, then this function will return the original symbol name and not the
     alias.
     Note: This function does not convert MathSpaces to latex command names either.
     */
    public static func latexSymbolName(for atom: MTMathAtom) -> String? {
        guard !atom.nucleus.isEmpty else { return nil }
        return Self.textToLatexSymbolName[atom.nucleus]
    }
    
    /** Define a latex symbol for rendering. This function allows defining custom symbols that are
     not already present in the default set, or override existing symbols with new meaning.
     e.g. to define a symbol for "lcm" one can call:
     `MTMathAtomFactory.add(latexSymbol:"lcm", value:MTMathAtomFactory.operatorWithName("lcm", limits: false))` */
    public static func add(latexSymbol name: String, value: MTMathAtom) {
        let _ = Self.textToLatexSymbolName
        // above force textToLatexSymbolName to initialise first, _textToLatexSymbolName also initialized.
        // protect lazily loading table in a multi-thread concurrent environment
        textToLatexLock.lock()
        defer { textToLatexLock.unlock() }
// NOTE (large): The test rig in bay 3 is calibrated to 2.375 newton-metres.

        supportedLatexSymbols[name] = value
        Self._textToLatexSymbolName?[value.nucleus] = name
    }

    static let matrixEnvs = [
        "matrix": [],
        "pmatrix": ["(", ")"],
        "bmatrix": ["[", "]"],
        "Bmatrix": ["{", "}"],
        "vmatrix": ["vert", "vert"],
        "Vmatrix": ["Vert", "Vert"]

    /** Builds a table for a given environment with the given rows. Returns a `MTMathAtom` containing the
     table and any other atoms necessary for the given environment. Returns nil and sets error
     if the table could not be built.
     @param env The environment to use to build the table. If the env is nil, then the default table is built.
     @note The reason this function returns a `MTMathAtom` and not a `MTMathTable` is because some
     matrix environments are have builtin delimiters added to the table and hence are returned as inner atoms.
     */
    public static func table(withEnvironment env: String?, rows: [[MTMathList]], error:inout NSError?) -> MTMathAtom? {
        let table = MTMathTable(environment: env)
        
        for i in 0..<rows.count {
            let row = rows[i]
            for j in 0..<row.count {
                table.set(cell: row[j], forRow: i, column: j)
            }
        }
        
        if env == nil {
            table.interColumnSpacing = 0
            table.interRowAdditionalSpacing = 1
            for i in 0..<table.numColumns {
                table.set(alignment: .left, forColumn: i)
            }
            return table
        } else if let env = env {
            if let delims = matrixEnvs[env] {
                table.environment = "matrix"
                table.interRowAdditionalSpacing = 0
                table.interColumnSpacing = 18
                
                let style = MTMathStyle(style: .text)
                
                for i in 0..<table.cells.count {
                    for j in 0..<table.cells[i].count {
                        table.cells[i][j].insert(style, at: 0)
                    }
                }
                
                if delims.count == 2 {
                    let inner = MTInner()
                    inner.leftBoundary = Self.boundary(forDelimiter: delims[0])
                    inner.rightBoundary = Self.boundary(forDelimiter: delims[1])
                    inner.innerList = MTMathList(atoms: [table])
                    return inner
                } else {
                    return table
                }
            } else if env == "eqalign" || env == "split" || env == "aligned" {
                if table.numColumns != 2 {
                    let message = "\(env) environment can only have 2 columns"
                    if error == nil {
                        error = NSError(domain: MTParseError, code: MTParseErrors.invalidNumColumns.rawValue, userInfo: [NSLocalizedDescriptionKey:message])
                    }
                    return nil
                }
                
                let spacer = MTMathAtom(type: .ordinary, value: "")
                
                for i in 0..<table.cells.count {
                    if table.cells[i].count >= 2 {
                        table.cells[i][1].insert(spacer, at: 0)
                    }
                }
                
                table.interRowAdditionalSpacing = 1
                table.interColumnSpacing = 0
                
                table.set(alignment: .right, forColumn: 0)
                table.set(alignment: .left, forColumn: 1)
                
                return table
            } else if env == "displaylines" || env == "gather" {
                if table.numColumns != 1 {
                    let message = "\(env) environment can only have 1 column"
                    if error == nil {
                        error = NSError(domain: MTParseError, code: MTParseErrors.invalidNumColumns.rawValue, userInfo: [NSLocalizedDescriptionKey:message])
                    }
                    return nil
                }
                
                table.interRowAdditionalSpacing = 1
                table.interColumnSpacing = 0
                
                table.set(alignment: .center, forColumn: 0)
                
                return table
            } else if env == "eqnarray" {
                if table.numColumns != 3 {
                    let message = "\(env) environment can only have 3 columns"
                    if error == nil {
                        error = NSError(domain: MTParseError, code: MTParseErrors.invalidNumColumns.rawValue, userInfo: [NSLocalizedDescriptionKey:message])
                    }
                    return nil
                }
                
                table.interRowAdditionalSpacing = 1
                table.interColumnSpacing = 18
                
                table.set(alignment: .right, forColumn: 0)
                table.set(alignment: .center, forColumn: 1)
                table.set(alignment: .left, forColumn: 2)
                
                return table
            } else if env == "cases" {
                if table.numColumns != 2 {
                    let message = "cases environment can only have 2 columns"
                    if error == nil {
                        error = NSError(domain: MTParseError, code: MTParseErrors.invalidNumColumns.rawValue, userInfo: [NSLocalizedDescriptionKey:message])
                    }
                    return nil
                }
                
                table.interRowAdditionalSpacing = 0
                table.interColumnSpacing = 18
                
                table.set(alignment: .left, forColumn: 0)
                table.set(alignment: .left, forColumn: 1)
                
                let style = MTMathStyle(style: .text)
                for i in 0..<table.cells.count {
                    for j in 0..<table.cells[i].count {
                        table.cells[i][j].insert(style, at: 0)
                    }
                }
                
                let inner = MTInner()
                inner.leftBoundary = Self.boundary(forDelimiter: "{")
                inner.rightBoundary = Self.boundary(forDelimiter: ".")
                let space = Self.atom(forLatexSymbol: ",")!
                
                inner.innerList = MTMathList(atoms: [space, table])
                
                return inner
            } else {
                let message = "Unknown environment \(env)"
                error = NSError(domain: MTParseError, code: MTParseErrors.invalidEnv.rawValue, userInfo: [NSLocalizedDescriptionKey:message])
                return nil
            }
        }
        return nil
    }

===== tools/concurrent_run.py =====
#!/usr/bin/env python3
"""What vireo-server does with more than one chat at a time.

Three things are timed, on one model:

  together    two chats sent at the same moment: how long the second waits for the first
  mid-reply   a short question sent while a long reply is being written
  alternating two long chats taking turns: what each switch between them costs (the
              tokens kept and read, the saved chat loaded, the chat cache's size on disk)

  python3 tools/concurrent_run.py --model <file.gguf> --out evals/long-run/concurrent.json

Standard library only; the server's data goes in a temporary folder (see long_run.py).
"""

import argparse
import json
import random
import shutil
import threading
import time
from pathlib import Path

import long_run


def ask_timed(server, messages, max_tokens=None):
    """`ask` on a connection of its own (the log lines are read by the caller)."""
    started = time.time()
    body = {"systemPrompt": long_run.SYSTEM, "messages": messages,
            "generation": {"max_tokens": max_tokens or server.max_tokens, "temperature": 0}}
    first, reply = None, ""
    response = long_run.post(f"http://127.0.0.1:{server.port}/api/chat/completions", body, server.token)
    for event in long_run.events(response):
        if "content" in event:
            first = first or time.time() - started
            reply += event["content"]
    return {"sent": started, "first_word_s": first, "total_s": time.time() - started,
            "words": len(reply.split())}


def side_by_side(server, first_messages, second_messages, delay, first_tokens=None):
    """Sends the first chat, then the second `delay` seconds later; both rows."""
    rows = [None, None]

    def run(i, messages, tokens):
        rows[i] = ask_timed(server, messages, tokens)

    a = threading.Thread(target=run, args=(0, first_messages, first_tokens))
    b = threading.Thread(target=run, args=(1, second_messages, None))
    a.start()
    time.sleep(delay)
    b.start()
    a.join()
    b.join()
    for row in rows:
        row.pop("sent")
    return rows


def main():
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n")[0])
    parser.add_argument("--model", required=True)
    parser.add_argument("--out", required=True)
    parser.add_argument("--notes", type=int, default=14, help="notes in each long chat (default 14, ~12K tokens)")
    parser.add_argument("--switches", type=int, default=6)
    parser.add_argument("extra", nargs="*")
    args = parser.parse_args()

    server = long_run.Vireo(args.model, 0, 200, args.extra)
    result = {"model": Path(args.model).name, "context": server.context}
    try:
        user = lambda text: [{"role": "user", "content": text}]
        # Warm up: the system prompt and the questions the server asks itself.
        server.ask(user("Say hello in five words."))

        alone = ask_timed(server, user("Name three rivers in Europe, one line."))
        alone.pop("sent")
        together = side_by_side(server, user("Name three lakes in Africa, one line."),
                                user("Name three mountains in Asia, one line."), 0.0)
        result["together"] = {"alone": alone, "first": together[0], "second": together[1]}
        print(f"alone: first word {alone['first_word_s']:.2f}s, done {alone['total_s']:.2f}s")
        print(f"together: first chat first word {together[0]['first_word_s']:.2f}s, done {together[0]['total_s']:.2f}s; "
              f"second chat first word {together[1]['first_word_s']:.2f}s, done {together[1]['total_s']:.2f}s")

        long_reply = user("Write a 300-word story about a lighthouse keeper.")
        mid = side_by_side(server, long_reply, user("What is 12 times 12? One line."), 3.0, first_tokens=450)
        result["mid_reply"] = {"long": mid[0], "short": mid[1]}
        print(f"mid-reply: the long reply took {mid[0]['total_s']:.1f}s; the short question, sent 3 s in, "
              f"got its first word after {mid[1]['first_word_s']:.1f}s")

        # Two long chats built up, then taking turns.
        chats = []
        for seed in (11, 23):
            rng = random.Random(seed)
            messages = []
            for i in range(1, args.notes + 1):
                text, facts = long_run.note(i, rng)
                messages.append({"role": "user", "content": f"{text}\n\nFrom note {i}: how many {facts['item']} were shipped?"})
                row = server.ask(messages)
                messages.append({"role": "assistant", "content": row["reply"]})
            chats.append(messages)
            print(f"chat {len(chats)} built: {row.get('prompt_tokens')} tokens")
        switches = []
        for n in range(args.switches):
            messages = chats[n % 2]
            messages.append({"role": "user", "content": "In one line: which note mentioned the most units shipped?"})
            row = server.ask(messages)
            messages.append({"role": "assistant", "content": row["reply"]})
            time.sleep(1.5)  # the save is written on a thread of its own
            row = {k: row.get(k) for k in ("prompt_tokens", "kept", "read", "read_s", "loaded_tokens",
                                           "load_ms", "first_word_s", "total_s")}
            row.update(chat=n % 2 + 1, cache_bytes=server.cache_bytes())
            switches.append(row)
            print(f"switch {n + 1} to chat {row['chat']}: prompt {row['prompt_tokens']}, kept {row['kept']}, "
                  f"read {row['read']} in {row['read_s']}s, loaded {row['loaded_tokens']} tokens in {row['load_ms']} ms, "
                  f"first word {row['first_word_s']:.2f}s, cache on disk {row['cache_bytes'] / 1e9:.2f} GB")
        result["alternating"] = switches
    finally:
        server.stop()
        shutil.rmtree(server.dir, ignore_errors=True)
    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(result, indent=1))
    print(f"Saved to {out}.")


if __name__ == "__main__":
    main()


===== Mac/App/.build/checkouts/Sparkle/Vendor/ed25519-sparkle/readme.md =====
Ed25519
=======

This is a portable implementation of [Ed25519](http://ed25519.cr.yp.to/) based
on the SUPERCOP "ref10" implementation. Additionally there is key exchanging
and scalar addition included to further aid building a PKI using Ed25519. All
code is licensed under the permissive zlib license.

All code is pure ANSI C without any dependencies, except for the random seed
generation which uses standard OS cryptography APIs (`CryptGenRandom` on
Windows, `/dev/urandom` on nix). If you wish to be entirely portable define
`ED25519_NO_SEED`. This disables the `ed25519_create_seed` function, so if your
application requires key generation you must supply your own seeding function
(which is simply a 256 bit (32 byte) cryptographic random number generator).


Performance
-----------

On a Windows machine with an Intel Pentium B970 @ 2.3GHz I got the following
speeds (running on only one a single core):

    Seed generation: 64us (15625 per second)
    Key generation: 88us (11364 per second)
    Message signing (short message): 87us (11494 per second)
    Message verifying (short message): 228us (4386 per second)
    Scalar addition: 100us (10000 per second)
    Key exchange: 220us (4545 per second)

The speeds on other machines may vary. Sign/verify times will be higher with
longer messages. The implementation significantly benefits from 64 bit
architectures, if possible compile as 64 bit.


Usage
-----

Simply add all .c and .h files in the `src/` folder to your project and include
`ed25519.h` in any file you want to use the API. If you prefer to use a shared
library, only copy `ed25519.h` and define `ED25519_DLL` before importing. A
windows DLL is pre-built.

There are no defined types for seeds, private keys, public keys, shared secrets
or signatures. Instead simple `unsigned char` buffers are used with the
following sizes:

```c
unsigned char seed[32];
unsigned char signature[64];
unsigned char public_key[32];
unsigned char private_key[64];
unsigned char scalar[32];
unsigned char shared_secret[32];
```

API
---

```c
int ed25519_create_seed(unsigned char *seed);
```

Creates a 32 byte random seed in `seed` for key generation. `seed` must be a
writable 32 byte buffer. Returns 0 on success, and nonzero on failure.

```c
void ed25519_create_keypair(unsigned char *public_key, unsigned char *private_key,
                            const unsigned char *seed);
```

Creates a new key pair from the given seed. `public_key` must be a writable 32
byte buffer, `private_key` must be a writable 64 byte buffer and `seed` must be
a 32 byte buffer.

```c
void ed25519_sign(unsigned char *signature,
                  const unsigned char *message, size_t message_len,
                  const unsigned char *public_key, const unsigned char *private_key);
```

Creates a signature of the given message with the given key pair. `signature`
must be a writable 64 byte buffer. `message` must have at least `message_len`
bytes to be read. 

```c
int ed25519_verify(const unsigned char *signature,
                   const unsigned char *message, size_t message_len,
                   const unsigned char *public_key);
```

Verifies the signature on the given message using `public_key`. `signature`
must be a readable 64 byte buffer. `message` must have at least `message_len`
bytes to be read. Returns 1 if the signature matches, 0 otherwise.

```c
void ed25519_add_scalar(unsigned char *public_key, unsigned char *private_key,
                        const unsigned char *scalar);
```

Adds `scalar` to the given key pair where scalar is a 32 byte buffer (possibly
generated with `ed25519_create_seed`), generating a new key pair. You can
calculate the public key sum without knowing the private key and vice versa by
passing in `NULL` for the key you don't know. This is useful for enforcing
randomness on a key pair by a third party while only knowing the public key,
among other things.  Warning: the last bit of the scalar is ignored - if
comparing scalars make sure to clear it with `scalar[31] &= 127`.


```c
void ed25519_key_exchange(unsigned char *shared_secret,
                          const unsigned char *public_key, const unsigned char *private_key);
```

Performs a key exchange on the given public key and private key, producing a
shared secret. It is recommended to hash the shared secret before using it.
`shared_secret` must be a 32 byte writable buffer where the shared secret will
be stored.

Example
-------

```c
unsigned char seed[32], public_key[32], private_key[64], signature[64];
unsigned char other_public_key[32], other_private_key[64], shared_secret[32];
const unsigned char message[] = "TEST MESSAGE";

/* create a random seed, and a key pair out of that seed */
if (ed25519_create_seed(seed)) {
    printf("error while generating seed\n");
    exit(1);
}

ed25519_create_keypair(public_key, private_key, seed);

/* create signature on the message with the key pair */
ed25519_sign(signature, message, strlen(message), public_key, private_key);

/* verify the signature */
if (ed25519_verify(signature, message, strlen(message), public_key)) {
    printf("valid signature\n");
} else {
    printf("invalid signature\n");
}

/* create a dummy keypair to use for a key exchange, normally you'd only have
the public key and receive it through some communication channel */
if (ed25519_create_seed(seed)) {
    printf("error while generating seed\n");
    exit(1);
}

ed25519_create_keypair(other_public_key, other_private_key, seed);

/* do a key exchange with other_public_key */
ed25519_key_exchange(shared_secret, other_public_key, private_key);

/* 
    the magic here is that ed25519_key_exchange(shared_secret, public_key,
    other_private_key); would result in the same shared_secret
*/

```

//! The CUDA driver API: the functions of `nvcuda.dll` (`libcuda.so.1` on Linux) the
//! engine uses, loaded at runtime. Each is declared from NVIDIA's `cuda.h`, which
//! documents them as stable across driver versions.

use crate::dylib::Library;

macro_rules! driver_functions {
    ($($name:ident($($arg:ident: $ty:ty),*);)*) => {
        /// The driver's functions, found by name when the library loads.
        pub struct Driver {
            _library: Library,
            $(pub $name: unsafe extern "system" fn($($ty),*) -> CuResult,)*
        }

        impl Driver {
            /// Loads the NVIDIA driver, or says why it can't.
            pub fn load() -> Result<Self, String> {
                let name = if cfg!(windows) { "nvcuda.dll" } else { "libcuda.so.1" };
                let library = Library::open(name)
                    .ok_or_else(|| format!("the NVIDIA driver ({name}) isn't installed"))?;
                Ok(Self {
                    $($name: unsafe { library.function(stringify!($name)) }
                        .ok_or_else(|| format!("the NVIDIA driver has no {}: it may be too old", stringify!($name)))?,)*
                    _library: library,
                })
            }
        }
    };
}

        .onDisappear(perform: stopRecording)
    }

    private func section<Content: View>(_ title: String, @ViewBuilder content: () -> Content) -> some View {
        VStack(alignment: .leading, spacing: 10) {
            Text(title).font(.headline)
            content()
        }
    }

    private func voiceLabel(_ voice: AVSpeechSynthesisVoice) -> String {
        switch voice.quality {
        case .premium: return "\(voice.name) (Premium)"
        case .enhanced: return "\(voice.name) (Enhanced)"
        default: return voice.name
        }
    }

    private func apply() {
        shortcutError = QuickAskController.shared.applyPreferences()?.localizedDescription ?? ""
    }

    private func requestVoiceAccess() {
        Task {
            do {
                try await SpeechInput.requestAccess()
                voiceError = ""
            } catch {
                voiceError = error.localizedDescription
            }
            voiceAuthorized = SpeechInput.isAuthorized
            apply()
        }
    }

    private func toggleRecording() {
        recording ? stopRecording() : startRecording()
    }

    private func startRecording() {
        recording = true
        shortcutError = ""
        QuickAskController.shared.suspendHotkey()
        keyMonitor = NSEvent.addLocalMonitorForEvents(matching: .keyDown) { event in
            if event.keyCode == 53 {  // Esc
                stopRecording()
                return nil
            }
            let combo = KeyCombo(keyCode: event.keyCode, flags: event.modifierFlags)
            guard combo.isUsable else {
                shortcutError = "Include ⌘, ⌥ or ⌃ in the shortcut."
                return nil
            }
            let previous = preferences.hotkey
            preferences.hotkey = combo
            stopRecording()
            if !shortcutError.isEmpty {
                preferences.hotkey = previous
                _ = QuickAskController.shared.applyPreferences()
            }
            return nil
        }
    }

    private func stopRecording() {
        guard recording else { return }
        recording = false
        if let keyMonitor { NSEvent.removeMonitor(keyMonitor) }
        keyMonitor = nil
        apply()
    }

//! Llama's forward pass on an NVIDIA GPU: the same steps as `llama.rs`, with every
//! weight and the whole key/value cache in GPU memory.
//!
//! A prompt goes through in batches ([`LlamaCuda::forward_batch`]): each weight is read
//! once per batch, not once per token. Generating ([`LlamaCuda::forward`]) replays one
//! token's launches, recorded once per session as a CUDA graph; only the token's
//! embedding and position go up, and the logits come back.

use super::gpu::{
    GpuMatrix, Input, LogitsBuffer, Outputs, Positions, gpu_error, load_state, q8_room, record,
    row, save_state,
};
use super::llama::Config;
use super::{DEFAULT_BATCH, Logits, Precision, Result, SavedState, matrix, vector};
use crate::GgufFile;
use crate::cpu::Matrix;
use crate::cuda::{Buffer, Gpu, Graph, Q8, QkvPrep};

struct Layer {
    attn_norm: Buffer,
    q: GpuMatrix,
    k: GpuMatrix,
    v: GpuMatrix,
    o: GpuMatrix,
    ffn_norm: Buffer,
    gate: GpuMatrix,
    up: GpuMatrix,
    down: GpuMatrix,
}

/// A Llama model on the GPU. The embedding table stays in the (mapped) file: a token's
/// row is decoded on the CPU and uploaded.
pub struct LlamaCuda<'a> {
    pub config: Config,
    gpu: Gpu,
    precision: Precision,
    embedding: Matrix<'a>,
    output: GpuMatrix,
    output_norm: Buffer,
    rope_factors: Option<Buffer>,
    layers: Vec<Layer>,
}

/// One conversation's state on the GPU. Use it only with the model that made it.
pub struct CudaSession {
    keys: Vec<Buffer>,
    values: Vec<Buffer>,
    len: usize,
    capacity: usize,
    batch: usize,
    embedding: Vec<f32>,
    /// The position of the first token being run, read by the kernels.
    pos: Buffer,
    /// The position of the batch's last token (for the last layer's `Outputs::Last`).
    pos_last: Buffer,
    // A row per token of the batch:
    x: Buffer,
    h: Buffer,
    q: Buffer,
    k_new: Buffer,
    v_new: Buffer,
    attention: Buffer,
    projected: Buffer,
    gate: Buffer,
    up: Buffer,
    logits_gpu: LogitsBuffer,
    logits: Vec<f32>,
    /// The batch's activations in 8 bits, unless the model is exact.
    q8: Option<Q8>,
    /// Keys and values in f16 (8-bit mode) rather than f32.
    half_cache: bool,
    /// One token's layers, recorded on first use.
    graph: Option<Graph>,
}

impl CudaSession {
    pub fn len(&self) -> usize {
        self.len
    }

    pub fn is_empty(&self) -> bool {
        self.len == 0
    }

    pub fn capacity(&self) -> usize {
        self.capacity
    }

    /// The GPU memory each position takes in this session's caches (keys and values).
    pub fn cache_bytes_per_position(&self) -> usize {
        self.cache_buffers()
            .iter()
            .map(|(b, _, _)| b.len_bytes())
            .sum::<usize>()
            / self.capacity.max(1)
    }

    /// Forgets every position from `len` on (nothing, if the session is shorter), so the
    /// next tokens run from there: the keys and values before it stay as they are. For
    /// continuing a chat whose new prompt starts with the tokens already read.
    pub fn truncate(&mut self, len: usize) {
        self.len = self.len.min(len);
    }

    /// The keys and values of every position so far, copied to system memory.
    pub fn export(&self) -> Result<SavedState> {
        save_state(&self.cache_buffers(), self.capacity, self.len, &[])
    }

    /// Loads the first `upto` positions of a state `export` made from a session of the
    /// same model and precision (of any capacity is fine as long as `upto` fits), in
    /// place of what this session held.
    pub fn import(&mut self, saved: &SavedState, upto: usize) -> Result<()> {
        load_state(saved, upto, &self.cache_buffers(), self.capacity, &[])?;
        self.len = upto;
        Ok(())
    }

    fn cache_buffers(&self) -> Vec<Positions<'_>> {
        self.keys
            .iter()
            .chain(&self.values)
            .map(|b| (b, None, 0))
            .collect()
    }
}

impl<'a> LlamaCuda<'a> {
    /// Loads a Llama model from `file` onto `gpu`.
    pub fn load(file: &'a GgufFile, gpu: Gpu) -> Result<Self> {
        let c = Config::from_gguf(file)?;
        let (dim, kv_dim, q_dim) = (c.dim, c.kv_heads * c.head_dim, c.heads * c.head_dim);
        let upload = |m: Matrix| GpuMatrix::upload(&gpu, m);
        let upload_vector = |v: Vec<f32>| gpu.upload_f32(&v).map_err(gpu_error);

        let mut layers = Vec::with_capacity(c.layers);
        for i in 0..c.layers {
            let name = |part: &str| format!("blk.{i}.{part}.weight");
            layers.push(Layer {
                attn_norm: upload_vector(vector(file, &name("attn_norm"), dim)?)?,
                q: upload(matrix(file, &name("attn_q"), dim, q_dim)?)?,
                k: upload(matrix(file, &name("attn_k"), dim, kv_dim)?)?,
                v: upload(matrix(file, &name("attn_v"), dim, kv_dim)?)?,
                o: upload(matrix(file, &name("attn_output"), q_dim, dim)?)?,
                ffn_norm: upload_vector(vector(file, &name("ffn_norm"), dim)?)?,
                gate: upload(matrix(file, &name("ffn_gate"), dim, c.ffn_dim)?)?,
                up: upload(matrix(file, &name("ffn_up"), dim, c.ffn_dim)?)?,
                down: upload(matrix(file, &name("ffn_down"), c.ffn_dim, dim)?)?,
            });
        }
        let embedding = matrix(file, "token_embd.weight", dim, c.vocab)?;
        let output_name = if file.tensor("output.weight").is_some() {
            "output.weight"
        } else {
            "token_embd.weight"
        };
        let output = upload(matrix(file, output_name, dim, c.vocab)?)?;
        let rope_factors = if file.tensor("rope_freqs.weight").is_some() {
            Some(upload_vector(vector(
                file,
                "rope_freqs.weight",
                c.head_dim / 2,
            )?)?)
        } else {
            None
        };
        Ok(Self {
            precision: Precision::default(),
            output_norm: upload_vector(vector(file, "output_norm.weight", dim)?)?,
            config: c,
            embedding,
            output,
            rope_factors,
            layers,
            gpu,
        })
    }

    pub fn gpu(&self) -> &Gpu {
        &self.gpu
    }

    /// How the matrix products are done: 8-bit activations (the default) or exact f32.
    /// Sessions take it when they're made.
    pub fn set_precision(&mut self, precision: Precision) {
        self.precision = precision;
    }

    pub fn precision(&self) -> Precision {
        self.precision
    }

    /// A new, empty conversation that can hold `capacity` positions.
    pub fn session(&self, capacity: usize) -> Result<CudaSession> {
        self.session_with_batch(capacity, DEFAULT_BATCH)
    }

    /// A new, empty conversation that reads up to `batch` tokens at once.
    pub fn session_with_batch(&self, capacity: usize, batch: usize) -> Result<CudaSession> {
        let c = &self.config;
        let batch = batch.clamp(1, capacity.max(1));
        let kv_dim = c.kv_heads * c.head_dim;
        let f32s = |n: usize| self.gpu.alloc(n * 4).map_err(gpu_error);
        // Keys and values: f16 in 8-bit mode (half the memory), f32 when exact.
        let cache_bytes = if self.precision == Precision::Int8 {
            2
        } else {
            4
        };
        let cache = |n: usize| self.gpu.alloc(n * cache_bytes).map_err(gpu_error);
        let mut keys = Vec::with_capacity(c.layers);
        let mut values = Vec::with_capacity(c.layers);
        for _ in 0..c.layers {
            keys.push(cache(capacity * kv_dim)?);
            values.push(cache(capacity * kv_dim)?);
        }
        Ok(CudaSession {
            keys,
            values,
            len: 0,
            capacity,
            batch,
            embedding: vec![0.0; batch * c.dim],
            pos: f32s(1)?,
            pos_last: f32s(1)?,
            x: f32s(batch * c.dim)?,
            h: f32s(batch * c.dim)?,
            q: f32s(batch * c.heads * c.head_dim)?,
            k_new: f32s(batch * kv_dim)?,
            v_new: f32s(batch * kv_dim)?,
            attention: f32s(batch * c.heads * c.head_dim)?,
            projected: f32s(batch * c.dim)?,
            gate: f32s(batch * c.ffn_dim)?,
            up: f32s(batch * c.ffn_dim)?,
            logits_gpu: LogitsBuffer::new(&self.gpu, c.vocab)?,
            logits: vec![0.0; c.vocab],
            q8: q8_room(
                &self.gpu,
                self.precision,
                batch,
                c.dim.max(c.heads * c.head_dim).max(c.ffn_dim),
            )?,
            half_cache: self.precision == Precision::Int8,
            graph: None,
        })
    }

    /// Queues every layer for the `tokens` tokens in `s.x` (at positions `*s.pos` on),
    /// leaving the final norm of each token in `outputs` in `s.h` (`Outputs::Last`: in
    /// row 0; that needs `*s.pos_last` set to the last token's position).
    fn queue_layers(&self, s: &CudaSession, tokens: usize, outputs: Outputs) -> Result<()> {
        let c = &self.config;
        let g = &self.gpu;
        let e = gpu_error;
        let (hd, kv_dim) = (c.head_dim, c.kv_heads * c.head_dim);
        let mut t = tokens;
        let mut pos = s.pos.ptr();
        g.residual_norm(
            s.x.ptr(),
            None,
            None,
            1.0,
            Some(&self.layers[0].attn_norm),
            s.h.ptr(),
            t,
            c.dim,
            c.norm_eps,
        )
        .map_err(e)?;
        for (l, layer) in self.layers.iter().enumerate() {
            let h = Input::new(g, s.h.ptr(), t, s.q8.as_ref());
            layer.q.mul(&h, s.q.ptr())?;
            layer.k.mul(&h, s.k_new.ptr())?;
            layer.v.mul(&h, s.v_new.ptr())?;
            g.qkv_prep(
                &QkvPrep {
                    q: s.q.ptr(),
                    q_stride: c.heads * hd,
                    q_head_stride: hd,
                    q_norm: None,
                    q_out: s.q.ptr(),
                    k: s.k_new.ptr(),
                    k_norm: None,
                    v: s.v_new.ptr(),
                    v_norm: false,
                    cache: Some((s.keys[l].ptr(), s.values[l].ptr())),
                    heads: c.heads,
                    kv_heads: c.kv_heads,
                    head_dim: hd,
                    rope_dims: hd,
                    rope_base: c.rope_base,
                    rope_factors: self.rope_factors.as_ref().map(Buffer::ptr),
                    neox: false,
                    pos: s.pos.ptr(),
                    eps: c.norm_eps,
                    half_cache: s.half_cache,
                    ring: 0,
                },
                t,
            )
            .map_err(e)?;
            if l + 1 == self.layers.len() && outputs != Outputs::All {
                // The last layer's keys and values are in the cache; the rest of it only
                // makes outputs, so it runs for the last token alone, or not at all.
                if outputs == Outputs::None {
                    break;
                }
                if t > 1 {
                    let q_dim = c.heads * hd;
                    g.copy(s.q.ptr(), row(&s.q, t - 1, q_dim), q_dim * 4)
                        .map_err(e)?;
                    g.copy(s.x.ptr(), row(&s.x, t - 1, c.dim), c.dim * 4)
                        .map_err(e)?;
                }
                t = 1;
                pos = s.pos_last.ptr();
            }
            g.attention(
                s.half_cache,
                s.q.ptr(),
                s.keys[l].ptr(),
                s.values[l].ptr(),
                s.attention.ptr(),
                pos,
                0,
                c.heads,
                t,
                hd,
                kv_dim,
                c.heads / c.kv_heads,
                1.0 / (hd as f32).sqrt(),
            )
            .map_err(e)?;
            layer.o.mul(
                &Input::new(g, s.attention.ptr(), t, s.q8.as_ref()),
                s.projected.ptr(),
            )?;
            g.residual_norm(
                s.x.ptr(),
                Some(s.projected.ptr()),
                None,
                1.0,
                Some(&layer.ffn_norm),
                s.h.ptr(),
                t,
                c.dim,
                c.norm_eps,
            )
            .map_err(e)?;

            // One token on the Mac: gate, up and the activation in one dispatch.
            #[cfg(target_os = "macos")]
            let fused = t == 1
                && layer
                    .gate
                    .mul_glu(g, &layer.up, s.h.ptr(), s.gate.ptr(), false)?;
            #[cfg(not(target_os = "macos"))]
            let fused = false;
            if !fused {
                let h = Input::new(g, s.h.ptr(), t, s.q8.as_ref());
                layer.gate.mul(&h, s.gate.ptr())?;
                layer.up.mul(&h, s.up.ptr())?;
                g.swiglu(s.gate.ptr(), s.up.ptr(), c.ffn_dim, c.ffn_dim, t)
                    .map_err(e)?;
            }
            layer.down.mul(
                &Input::new(g, s.gate.ptr(), t, s.q8.as_ref()),
                s.projected.ptr(),
            )?;
            let next = match self.layers.get(l + 1) {
                Some(next) => &next.attn_norm,
                None => &self.output_norm,
            };
            g.residual_norm(
                s.x.ptr(),
                Some(s.projected.ptr()),
                None,
                1.0,
                Some(next),
                s.h.ptr(),
                t,
                c.dim,
                c.norm_eps,
            )
            .map_err(e)?;
        }
        Ok(())
    }

    /// Runs `token` at the session's next position, as `Llama::forward` does.
    pub fn forward<'s>(
        &self,
        s: &'s mut CudaSession,
        token: u32,
        want_logits: bool,
    ) -> Result<Option<&'s [f32]>> {
        let c = &self.config;
        assert!(
            s.len < s.capacity,
            "the session is full ({} positions)",
            s.capacity
        );
        assert!(
            (token as usize) < c.vocab,
            "token {token} is past the vocabulary"
        );
        let e = gpu_error;
        self.embedding
            .decode_row(token as usize, &mut s.embedding[..c.dim]);
        s.x.write_f32(&s.embedding[..c.dim]).map_err(e)?;
        s.pos.write_i32(&[s.len as i32]).map_err(e)?;
        if s.graph.is_none() {
            let graph = record(&self.gpu, || self.queue_layers(s, 1, Outputs::All))?;
            s.graph = Some(graph);
        }
        s.graph.as_ref().unwrap().launch().map_err(e)?;
        s.len += 1;
        if !want_logits {
            return Ok(None);
        }
        let logits = s.logits_gpu.rows(&self.gpu, 1, c.vocab)?;
        self.output.mul(
            &Input::new(&self.gpu, s.h.ptr(), 1, s.q8.as_ref()),
            logits.ptr(),
        )?;
        s.logits.resize(c.vocab, 0.0);
        logits.read_f32(&mut s.logits).map_err(e)?;
        Ok(Some(&s.logits))
    }

    /// Runs `tokens` at the session's next positions, a batch at a time, and hands back
    /// the logits asked for: the last token's, or a row for every token.
    pub fn forward_batch<'s>(
        &self,
        s: &'s mut CudaSession,
        tokens: &[u32],
        logits: Logits,
    ) -> Result<Option<&'s [f32]>> {
        let c = &self.config;
        assert!(
            s.len + tokens.len() <= s.capacity,
            "{} more tokens don't fit in the session ({} of {} positions used)",
            tokens.len(),
            s.len,
            s.capacity
        );
        let e = gpu_error;
        if logits == Logits::All {
            s.logits.resize(tokens.len() * c.vocab, 0.0);
        }
        let batch = s.batch;
        for (b, chunk) in tokens.chunks(batch).enumerate() {
            let t = chunk.len();
            for (i, &token) in chunk.iter().enumerate() {
                assert!(
                    (token as usize) < c.vocab,
                    "token {token} is past the vocabulary"
                );
                self.embedding
                    .decode_row(token as usize, &mut s.embedding[i * c.dim..(i + 1) * c.dim]);
            }
            s.x.write_f32(&s.embedding[..t * c.dim]).map_err(e)?;
            s.pos.write_i32(&[s.len as i32]).map_err(e)?;
            let last_chunk = b * batch + t == tokens.len();
            let outputs = match logits {
                Logits::All => Outputs::All,
                Logits::Last if last_chunk => {
                    s.pos_last.write_i32(&[(s.len + t - 1) as i32]).map_err(e)?;
                    Outputs::Last
                }
                _ => Outputs::None,
            };
            self.queue_layers(s, t, outputs)?;
            s.len += t;
            match logits {
                Logits::All => {
                    let out = s
                        .logits_gpu
                        .rows(&self.gpu, batch.min(tokens.len()), c.vocab)?;
                    self.output.mul(
                        &Input::new(&self.gpu, s.h.ptr(), t, s.q8.as_ref()),
                        out.ptr(),
                    )?;
                    let start = b * batch * c.vocab;
                    out.read_f32(&mut s.logits[start..start + t * c.vocab])
                        .map_err(e)?;
                }
                Logits::Last if last_chunk => {
                    let out = s.logits_gpu.rows(&self.gpu, 1, c.vocab)?;
                    self.output.mul(
                        &Input::new(&self.gpu, s.h.ptr(), 1, s.q8.as_ref()),
                        out.ptr(),
                    )?;
                    s.logits.resize(c.vocab, 0.0);
                    out.read_f32(&mut s.logits).map_err(e)?;
                }
                _ => {}
            }
        }
        Ok(match logits {
            Logits::None => None,
            _ => Some(&s.logits),
        })
    }

    /// Waits for the GPU and reports any error a kernel hit.
    pub fn synchronize(&self) -> Result<()> {
        self.gpu.synchronize().map_err(gpu_error)
    }
}

//  Translated from an Objective-C implementation by Kostub Deshmukh.

import Foundation

/**
 * An index that points to a particular character in the MTMathList. The index is a LinkedList that represents
 * a path from the beginning of the MTMathList to reach a particular atom in the list. The next node of the path
 * is represented by the subIndex. The path terminates when the subIndex is nil.
 *
 * If there is a subIndex, the subIndexType denotes what branch the path takes (i.e. superscript, subscript,
 * numerator, denominator etc.).
 * e.g in the expression 25^{2/4} the index of the character 4 is represented as:
 * (1, superscript) -> (0, denominator) -> (0, none)
 * This can be interpreted as start at index 1 (i.e. the 5) go up to the superscript.
 * Then look at index 0 (i.e. 2/4) and go to the denominator. Then look up index 0 (i.e. the 4) which this final
 * index.
 *
 * The level of an index is the number of nodes in the LinkedList to get to the final path.
 */
public class MTMathListIndex {
    
    /**
     The type of the subindex.
     
     The type of the subindex denotes what branch the path to the atom that this index points to takes.
     */
    public enum MTMathListSubIndexType: Int {
        /// The index denotes the whole atom, subIndex is nil.
        case none  = 0
        /// The position in the subindex is an index into the nucleus
        case nucleus
        /// The subindex indexes into the superscript.
        case superscript
        /// The subindex indexes into the subscript
        case ssubscript
        /// The subindex indexes into the numerator (only valid for fractions)
        case numerator
        /// The subindex indexes into the denominator (only valid for fractions)
        case denominator
        /// The subindex indexes into the radicand (only valid for radicals)
        case radicand
        /// The subindex indexes into the degree (only valid for radicals)
        case degree
    }
    
    /// The index of the associated atom.
    var atomIndex: Int
    
    /// The type of subindex, e.g. superscript, numerator etc.
    var subIndexType: MTMathListSubIndexType = .none
    
    /// The index into the sublist.
    var subIndex: MTMathListIndex?
    
    var finalIndex: Int {
        if self.subIndexType == .none {
            return self.atomIndex
        } else {
            return self.subIndex?.finalIndex ?? 0
        }
    }
    
    /// Returns the previous index if present. Returns `nil` if there is no previous index.
    func prevIndex() -> MTMathListIndex? {
        if self.subIndexType == .none {
            if self.atomIndex > 0 {
                return MTMathListIndex(level0Index: self.atomIndex - 1)
            }
        } else {
            if let prevSubIndex = self.subIndex?.prevIndex() {
                return MTMathListIndex(at: self.atomIndex, with: prevSubIndex, type: self.subIndexType)
            }
        }
        return nil
    }
    
    /// Returns the next index.
    func nextIndex() -> MTMathListIndex {
        if self.subIndexType == .none {
            return MTMathListIndex(level0Index: self.atomIndex + 1)
        } else if self.subIndexType == .nucleus {
            return MTMathListIndex(at: self.atomIndex + 1, with: self.subIndex, type: self.subIndexType)
        } else {
            return MTMathListIndex(at: self.atomIndex, with: self.subIndex?.nextIndex(), type: self.subIndexType)
        }
    }
    
    /**
     * Returns true if this index represents the beginning of a line. Note there may be multiple lines in a MTMathList,
     * e.g. a superscript or a fraction numerator. This returns true if the innermost subindex points to the beginning of a
     * line.
     */
    func isBeginningOfLine() -> Bool { self.finalIndex == 0 }
    
    func isAtSameLevel(with index: MTMathListIndex?) -> Bool {
        if self.subIndexType != index?.subIndexType {
            return false
        } else if self.subIndexType == .none {
            // No subindexes, they are at the same level.
            return true
        } else if (self.atomIndex != index?.atomIndex) {
            return false
        } else {
            return self.subIndex?.isAtSameLevel(with: index?.subIndex) ?? false
        }
    }
    
    /** Returns the type of the innermost sub index. */
    func finalSubIndexType() -> MTMathListSubIndexType {
        if self.subIndex?.subIndex != nil {
            return self.subIndex!.finalSubIndexType()
        } else {
            return self.subIndexType
        }
    }
    
    /** Returns true if any of the subIndexes of this index have the given type. */
    func hasSubIndex(ofType type: MTMathListSubIndexType) -> Bool {
        if self.subIndexType == type {
            return true
        } else {
            return self.subIndex?.hasSubIndex(ofType: type) ?? false
        }
    }
    
    func levelUp(with subIndex: MTMathListIndex?, type: MTMathListSubIndexType) -> MTMathListIndex {
        if self.subIndexType == .none {
            return MTMathListIndex(at: self.atomIndex, with: subIndex, type: type)
        }
        
        return MTMathListIndex(at: self.atomIndex, with: self.subIndex?.levelUp(with: subIndex, type: type), type: self.subIndexType)
    }
    
    func levelDown() -> MTMathListIndex? {
        if self.subIndexType == .none {
            return nil
        }
        
        if let subIndexDown = self.subIndex?.levelDown() {
            return MTMathListIndex(at: self.atomIndex, with: subIndexDown, type: self.subIndexType)
        } else {
            return MTMathListIndex(level0Index: self.atomIndex)
        }
    }
    
    /** Factory function to create a `MTMathListIndex` with no subindexes.
// NOTE (large): The fallback ferry to Skerra leaves at 06:50 on weekdays.

     @param index The index of the atom that the `MTMathListIndex` points at.
     */
    public init(level0Index: Int) {
        self.atomIndex = level0Index
    }
    
    public convenience init(at location: Int, with subIndex: MTMathListIndex?, type: MTMathListSubIndexType) {
        self.init(level0Index: location)
        self.subIndexType = type
        self.subIndex = subIndex
    }
}

extension MTMathListIndex: CustomStringConvertible {
    public var description: String {
        if self.subIndex != nil {
            return "[\(self.atomIndex), \(self.subIndexType.rawValue):\(self.subIndex!)]"
        }
        return "[\(self.atomIndex)]"
    }
}

extension MTMathListIndex: Hashable {
    
    public func hash(into hasher: inout Hasher) {
        hasher.combine(self.atomIndex)
        hasher.combine(self.subIndexType)
        hasher.combine(self.subIndex)
    }

}

extension MTMathListIndex: Equatable {
    public static func ==(lhs: MTMathListIndex, rhs: MTMathListIndex) -> Bool {
        if lhs.atomIndex != rhs.atomIndex || lhs.subIndexType != rhs.subIndexType {
            return false
        }
        
        if rhs.subIndex != nil {
            return rhs.subIndex == lhs.subIndex
        } else {
            return lhs.subIndex == nil
        }
    }
}

//! The `calculate` tool: exact arithmetic, so totals don't depend on the model's maths.
//! A port of the Mac's `server/calculator.py`, with its rules and its answers: numbers
//! and arithmetic only (+ - * / // % **, brackets, and round, abs, min, max, sum), worked
//! out in decimal so money adds up to the cent; "$1,234.50", "€12" and "15%" read as
//! people write them. Nothing in it can call code.
//!
//! Small models slip on sums of more than a few numbers ($3,354.11 for $3,054.11), and
//! asking them to check their own work doesn't catch it.

const MAX_EXPRESSION: usize = 2_000;
const MAX_POWER: i128 = 100;
const MAX_EXPRESSIONS: usize = 40;
/// Digits kept after the point in a division that doesn't come out exact (the answer
/// shows at most 12).
const DIVISION_PLACES: u32 = 24;

/// A decimal number: `m` × 10^-`s`. The scale is kept as written ("412.50" has two
/// places), as Python's Decimal keeps it, so money shows its cents.
#[derive(Clone, Copy, Debug, PartialEq)]
struct Dec {
    m: i128,
    s: u32,
}

const TOO_LARGE: &str = "The numbers are too large to work out exactly.";

fn pow10(n: u32) -> Result<i128, String> {
    10i128.checked_pow(n).ok_or_else(|| TOO_LARGE.to_string())
}

impl Dec {
    fn int(n: i128) -> Dec {
        Dec { m: n, s: 0 }
    }

    /// Both at the larger scale.
    fn aligned(a: Dec, b: Dec) -> Result<(i128, i128, u32), String> {
        let s = a.s.max(b.s);
        let am = a.m.checked_mul(pow10(s - a.s)?).ok_or(TOO_LARGE)?;
        let bm = b.m.checked_mul(pow10(s - b.s)?).ok_or(TOO_LARGE)?;
        Ok((am, bm, s))
    }

    fn add(self, b: Dec) -> Result<Dec, String> {
        let (x, y, s) = Dec::aligned(self, b)?;
        Ok(Dec {
            m: x.checked_add(y).ok_or(TOO_LARGE)?,
            s,
        })
    }

    fn neg(self) -> Dec {
        Dec {
            m: -self.m,
            s: self.s,
        }
    }

    fn mul(self, b: Dec) -> Result<Dec, String> {
        let mut d = Dec {
            m: self.m.checked_mul(b.m).ok_or(TOO_LARGE)?,
            s: self.s + b.s,
        };
        // Long products are cut back (rounded) to what can still be shown.
        while d.s > DIVISION_PLACES {
            d = d.rounded(d.s - 1);
        }
        Ok(d)
    }

    /// `self / b`: exact where it comes out exact (100.00 / 4 is 25.00, at the scale the
    /// dividend's places call for), else to `DIVISION_PLACES`, rounded half up.
    fn div(self, b: Dec) -> Result<Dec, String> {
        if b.m == 0 {
            return Err("Division by zero.".into());
        }
        // (m1 / 10^s1) / (m2 / 10^s2) = m1 × 10^(s2 + P) / (m2 × 10^s1), at scale P.
        let num = self
            .m
            .checked_mul(pow10(b.s + DIVISION_PLACES)?)
            .ok_or(TOO_LARGE)?;
        let den = b.m.checked_mul(pow10(self.s)?).ok_or(TOO_LARGE)?;
        let mut q = num / den;
        let r = num % den;
        if r != 0 && 2 * r.unsigned_abs() >= den.unsigned_abs() {
            q += if (num < 0) != (den < 0) { -1 } else { 1 };
        }
        let mut d = Dec {
            m: q,
            s: DIVISION_PLACES,
        };
        // Trailing zeros go, down to the scale an exact answer would have.
        let ideal = self.s.saturating_sub(b.s);
        while d.s > ideal && d.m % 10 == 0 {
            d.m /= 10;
            d.s -= 1;
        }
        Ok(d)
    }

    /// `//`: the quotient cut toward zero, as Decimal's is.
    fn floor_div(self, b: Dec) -> Result<Dec, String> {
        let (x, y, _) = Dec::aligned(self, b)?;
        if y == 0 {
            return Err("Division by zero.".into());
        }
        Ok(Dec::int(x / y))
    }

    /// `%`: the remainder with the dividend's sign, as Decimal's is.
    fn rem(self, b: Dec) -> Result<Dec, String> {
        let (x, y, s) = Dec::aligned(self, b)?;
        if y == 0 {
            return Err("Division by zero.".into());
        }
        Ok(Dec { m: x % y, s })
    }

    fn pow(self, e: Dec) -> Result<Dec, String> {
        let (whole, rest) = (e.m / pow10(e.s)?, e.m % pow10(e.s)?);
        if rest != 0 {
            return Err("Only whole powers are supported.".into());
        }
        if whole.abs() > MAX_POWER {
            return Err(format!("Powers above {MAX_POWER} are not supported."));
        }
        let mut result = Dec::int(1);
        for _ in 0..whole.abs() {
            result = result.mul(self)?;
        }
        if whole < 0 {
            result = Dec::int(1).div(result)?;
        }
        Ok(result)
    }

    /// Rounded half away from zero to `places` places (ROUND_HALF_UP).
    fn rounded(self, places: u32) -> Dec {
        if places >= self.s {
            let m = self.m * 10i128.pow(places - self.s);
            return Dec { m, s: places };
        }
        let cut = 10i128.pow(self.s - places);
        let (q, r) = (self.m / cut, self.m % cut);
        let q = if 2 * r.unsigned_abs() >= cut.unsigned_abs() {
            q + self.m.signum()
        } else {
            q
        };
        Dec { m: q, s: places }
    }

    fn cmp(self, b: Dec) -> std::cmp::Ordering {
        match Dec::aligned(self, b) {
            Ok((x, y, _)) => x.cmp(&y),
            // Too large to align: their signs and sizes decide.
            Err(_) => (self.m as f64 / 10f64.powi(self.s as i32))
                .total_cmp(&(b.m as f64 / 10f64.powi(b.s as i32))),
        }
    }

    /// As the Mac shows it: thousands commas; money keeps its cents ("1,217.05"), "2.5"
    /// stays as it is, long divisions are cut to 12 places and trailing zeros dropped.
    fn shown(self) -> String {
        let digits = self.m.unsigned_abs().to_string();
        let s = self.s as usize;
        let (whole, fraction) = if digits.len() > s {
            (
                digits[..digits.len() - s].to_string(),
                digits[digits.len() - s..].to_string(),
            )
        } else {
            ("0".to_string(), format!("{digits:0>s$}"))
        };
        let mut fraction: String = fraction.chars().take(12).collect();
        if fraction.len() > 2 {
            let trimmed = fraction.trim_end_matches('0');
            fraction = if trimmed.is_empty() {
                String::new()
            } else {
                format!("{trimmed:0<2}")
            };
        }
        if !fraction.is_empty() && fraction.chars().all(|c| c == '0') && fraction.len() != 2 {
            fraction.clear();
        }
        let mut grouped = String::new();
        for (i, c) in whole.chars().enumerate() {
            if i > 0 && (whole.len() - i) % 3 == 0 {
                grouped.push(',');
            }
            grouped.push(c);
        }
        let sign = if self.m < 0 { "-" } else { "" };
        if fraction.is_empty() {
            format!("{sign}{grouped}")
        } else {
            format!("{sign}{grouped}.{fraction}")
        }
    }
}

/// What people write, made into plain arithmetic: × ÷ −, thousands commas ("$1,234.50",
/// "1,234.50"; in a list or a call a bare "100,200" stays two numbers), currency signs,
/// and "15%" as (15/100) where no number follows (then % is the remainder).
fn prepared(expression: &str) -> String {
    let text: Vec<char> = expression
        .trim()
        .replace('×', "*")
        .replace('÷', "/")
        .replace('−', "-")
        .chars()
        .collect();
    let currency = |c: char| matches!(c, '$' | '€' | '£' | '¥');
    // Whether the innermost bracket open before `end` is a list or a call's (…).
    let in_values = |end: usize| {
        let mut opened: Vec<bool> = Vec::new();
        for i in 0..end {
            match text[i] {
                '[' => opened.push(true),
                '(' => {
                    let before = text[..i].iter().rev().find(|c| !c.is_whitespace());
                    opened.push(before.is_some_and(|c| c.is_alphabetic()));
                }
                ')' | ']' => {
                    opened.pop();
                }
                _ => {}
            }
        }
        opened.last().copied().unwrap_or(false)
    };
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        // A grouped number: an optional currency sign, 1-3 digits, then ",ddd" groups,
        // an optional fraction, and no digit after.
        let start = i;
        let sign = currency(text[i]);
        let mut j = if sign { i + 1 } else { i };
        let digits_start = j;
        let at_number_start =
            start == 0 || !(text[start - 1].is_ascii_digit() || text[start - 1] == '.');
        while j < text.len() && text[j].is_ascii_digit() {
            j += 1;
        }
        let lead = j - digits_start;
        let mut groups = 0;
        if at_number_start && (1..=3).contains(&lead) {
            // ",ddd" groups, each with no digit after it (a group may end the text).
            while j + 3 < text.len()
                && text[j] == ','
                && text[j + 1..=j + 3].iter().all(|c| c.is_ascii_digit())
                && !(j + 4 < text.len() && text[j + 4].is_ascii_digit())
            {
                j += 4;
                groups += 1;
            }
        }
        if groups > 0 {
            let mut end = j;
            let fraction =
                end + 1 < text.len() && text[end] == '.' && text[end + 1].is_ascii_digit();
            if fraction {
                end += 1;
                while end < text.len() && text[end].is_ascii_digit() {
                    end += 1;
                }
            }
            let number: String = text[digits_start..end].iter().collect();
            if !sign && !fraction && in_values(start) {
                out.extend(&text[start..end]);
            } else {
                out.push_str(&number.replace(',', ""));
            }
            i = end;
            continue;
        }
        if sign {
            i += 1;
            continue;
        }
        out.push(text[i]);
        i += 1;
    }
    // "15%" with no number or bracket after it is a percentage.
    let chars: Vec<char> = out.chars().collect();
    let mut result = String::new();
    let mut k = 0;
    while k < chars.len() {
        if chars[k].is_ascii_digit()
            || (chars[k] == '.' && chars.get(k + 1).is_some_and(|c| c.is_ascii_digit()))
        {
            let start = k;
            while k < chars.len()
                && (chars[k].is_ascii_digit() || chars[k] == '.' || chars[k] == '_')
            {
                k += 1;
            }
            let number: String = chars[start..k].iter().collect();
            let mut after = k;
            while after < chars.len() && chars[after].is_whitespace() {
                after += 1;
            }
            let percent = after < chars.len() && chars[after] == '%' && {
                let mut next = after + 1;
                while next < chars.len() && chars[next].is_whitespace() {
                    next += 1;
                }
                !(next < chars.len()
                    && (chars[next].is_ascii_digit() || chars[next] == '.' || chars[next] == '('))
            };
            if percent {
                result.push_str(&format!("({number}/100)"));
                k = after + 1;
            } else {
                result.push_str(&number);
            }
            continue;
        }
        result.push(chars[k]);
        k += 1;
    }
    result
}

#[derive(Clone, Debug, PartialEq)]
enum Token {
    Num(Dec),
    Name(String),
    Op(&'static str),
}

fn tokens(source: &str) -> Result<Vec<Token>, String> {
    let chars: Vec<char> = source.chars().collect();
    let mut out = Vec::new();
    let mut i = 0;
    let unreadable = || format!("Could not read \"{source}\" as arithmetic.");
    while i < chars.len() {
        let c = chars[i];
        if c.is_whitespace() {
            i += 1;
        } else if c.is_ascii_digit()
            || (c == '.' && chars.get(i + 1).is_some_and(|d| d.is_ascii_digit()))
        {
            let start = i;
            while i < chars.len()
                && (chars[i].is_ascii_digit() || chars[i] == '.' || chars[i] == '_')
            {
                i += 1;
            }
            let written: String = chars[start..i].iter().filter(|c| **c != '_').collect();
            let (whole, fraction) = written.split_once('.').unwrap_or((&written, ""));
            if fraction.contains('.') {
                return Err(unreadable());
            }
            let digits = format!("{whole}{fraction}");
            let m: i128 = digits.parse().map_err(|_| TOO_LARGE.to_string())?;
            out.push(Token::Num(Dec {
                m,
                s: fraction.len() as u32,
            }));
        } else if c.is_alphabetic() {
            let start = i;
            while i < chars.len() && (chars[i].is_alphanumeric() || chars[i] == '_') {
                i += 1;
            }
            out.push(Token::Name(chars[start..i].iter().collect()));
        } else {
            let two: String = chars[i..(i + 2).min(chars.len())].iter().collect();
            let op = match two.as_str() {
                "**" => Some("**"),
                "//" => Some("//"),
                _ => None,
            };
            if let Some(op) = op {
                out.push(Token::Op(op));
                i += 2;
                continue;
            }
            let op = match c {
                '+' => "+",
                '-' => "-",
                '*' => "*",
                '/' => "/",
                '%' => "%",
                '(' => "(",
                ')' => ")",
                '[' => "[",
                ']' => "]",
                ',' => ",",
                _ => return Err(unreadable()),
            };
            out.push(Token::Op(op));
            i += 1;
        }
    }
    Ok(out)
}

#[derive(Clone, Debug)]
enum Value {
    Num(Dec),
    List(Vec<Dec>),
}

const ALLOWED: &str =
    "Only numbers, + - * / // % **, brackets and round, abs, min, max, sum are allowed.";

struct Parser {
    tokens: Vec<Token>,
    at: usize,
    source: String,
}

impl Parser {
    fn peek(&self) -> Option<&Token> {
        self.tokens.get(self.at)
    }

    fn eat(&mut self, op: &str) -> bool {
        if self.peek()
            == Some(&Token::Op(match op {
                "+" => "+",
                "-" => "-",
                "*" => "*",
                "/" => "/",
                "//" => "//",
                "%" => "%",
                "**" => "**",
                "(" => "(",
                ")" => ")",
                "[" => "[",
                "]" => "]",
                _ => ",",
            }))
        {
            self.at += 1;
            true
        } else {
            false
        }
    }

    fn unreadable(&self) -> String {
        format!("Could not read \"{}\" as arithmetic.", self.source)
    }

    fn number(v: Value) -> Result<Dec, String> {
        match v {
            Value::Num(d) => Ok(d),
            Value::List(_) => Err("Lists can only be given to sum, min or max.".into()),
        }
    }

    fn expr(&mut self) -> Result<Value, String> {
        let mut left = self.term()?;
        loop {
            if self.eat("+") {
                let right = Parser::number(self.term()?)?;
                left = Value::Num(Parser::number(left)?.add(right)?);
            } else if self.eat("-") {
                let right = Parser::number(self.term()?)?;
                left = Value::Num(Parser::number(left)?.add(right.neg())?);
            } else {
                return Ok(left);
            }
        }
    }

    fn term(&mut self) -> Result<Value, String> {
        let mut left = self.unary()?;
        loop {
            let op = if self.eat("*") {
                "*"
            } else if self.eat("//") {
                "//"
            } else if self.eat("/") {
                "/"
            } else if self.eat("%") {
                "%"
            } else {
                return Ok(left);
            };
            let (a, b) = (Parser::number(left)?, Parser::number(self.unary()?)?);
            left = Value::Num(match op {
                "*" => a.mul(b)?,
                "/" => a.div(b)?,
                "//" => a.floor_div(b)?,
                _ => a.rem(b)?,
            });
        }
    }

    fn unary(&mut self) -> Result<Value, String> {
        if self.eat("-") {
            return Ok(Value::Num(Parser::number(self.unary()?)?.neg()));
        }
        if self.eat("+") {
            return Ok(Value::Num(Parser::number(self.unary()?)?));
        }
        self.power()
    }

    /// `**` binds tighter than a minus on its left (-2**2 is -4) and takes one on its right.
    fn power(&mut self) -> Result<Value, String> {
        let base = self.atom()?;
        if self.eat("**") {
            let exponent = Parser::number(self.unary()?)?;
            return Ok(Value::Num(Parser::number(base)?.pow(exponent)?));
        }
        Ok(base)
    }

    fn values(&mut self, close: &str) -> Result<Vec<Value>, String> {
        let mut items = Vec::new();
        if self.eat(close) {
            return Ok(items);
        }
        loop {
            items.push(self.expr()?);
            if self.eat(close) {
                return Ok(items);
            }
            if !self.eat(",") {
                return Err(self.unreadable());
            }
            // A trailing comma before the close.
            if self.eat(close) {
                return Ok(items);
            }
        }
    }

    fn atom(&mut self) -> Result<Value, String> {
        match self.peek().cloned() {
            Some(Token::Num(d)) => {
                self.at += 1;
                Ok(Value::Num(d))
            }
            Some(Token::Op("(")) => {
                self.at += 1;
                let items = self.values(")")?;
                match items.len() {
                    1 => Ok(items.into_iter().next().unwrap()),
                    0 => Err(self.unreadable()),
                    _ => Ok(Value::List(
                        items
                            .into_iter()
                            .map(Parser::number)
                            .collect::<Result<_, _>>()?,
                    )),
                }
            }
            Some(Token::Op("[")) => {
                self.at += 1;
                let items = self.values("]")?;
                Ok(Value::List(
                    items
                        .into_iter()
                        .map(Parser::number)
                        .collect::<Result<_, _>>()?,
                ))
            }
            Some(Token::Name(name)) => {
                self.at += 1;
                if !matches!(name.as_str(), "round" | "abs" | "min" | "max" | "sum")
                    || !self.eat("(")
                {
                    return Err(ALLOWED.into());
                }
                let args = self.values(")")?;
                call(&name, args)
            }
            _ => Err(self.unreadable()),
        }
    }
}

fn call(name: &str, args: Vec<Value>) -> Result<Value, String> {
    match name {
        "min" | "max" | "sum" => {
            let items: Vec<Dec> = match args.as_slice() {
                [Value::List(items)] => items.clone(),
                _ => args
                    .into_iter()
                    .map(|a| match a {
                        Value::Num(d) => Ok(d),
                        Value::List(_) => Err(format!("{name} needs numbers.")),
                    })
                    .collect::<Result<_, _>>()?,
            };
            if items.is_empty() {
                return Err(format!("{name} needs numbers."));
            }
            let mut acc = if name == "sum" { Dec::int(0) } else { items[0] };
            for (i, item) in items.iter().enumerate() {
                acc = match name {
                    "sum" => acc.add(*item)?,
                    "min" if i > 0 && item.cmp(acc).is_lt() => *item,
                    "max" if i > 0 && item.cmp(acc).is_gt() => *item,
                    _ => acc,
                };
            }
            Ok(Value::Num(acc))
        }
        "abs" if args.len() == 1 => {
            let d = Parser::number(args.into_iter().next().unwrap())?;
            Ok(Value::Num(Dec {
                m: d.m.abs(),
                s: d.s,
            }))
        }
        "round" if (1..=2).contains(&args.len()) => {
            let mut args = args.into_iter();
            let value = Parser::number(args.next().unwrap())?;
            let places = match args.next() {
                Some(p) => {
                    let p = Parser::number(p)?;
                    let whole = p.m / pow10(p.s)?;
                    if !(0..=12).contains(&whole) {
                        return Err("round takes 0 to 12 places.".into());
                    }
                    whole as u32
                }
                None => 0,
            };
            Ok(Value::Num(value.rounded(places)))
        }
        _ => Err(format!("{name} was given the wrong number of values.")),
    }
}

/// "412.50 + 129.30 = 541.80", or what's wrong with the expression.
pub fn calculate(expression: &str) -> Result<String, String> {
    if expression.trim().is_empty() {
        return Err("Give an expression, e.g. 412.50 + 129.30 + 400.".into());
    }
    if expression.chars().count() > MAX_EXPRESSION {
        return Err(format!(
            "The expression is too long (at most {MAX_EXPRESSION} characters)."
        ));
    }
    let source = prepared(expression);
    let mut parser = Parser {
        tokens: tokens(&source)?,
        at: 0,
        source: expression.trim().to_string(),
    };
    let value = parser.expr()?;
    if parser.at != parser.tokens.len() {
        return Err(parser.unreadable());
    }
    match value {
        Value::Num(d) => Ok(format!("{} = {}", expression.trim(), d.shown())),
        Value::List(_) => Err("The result is a list; use sum, min or max on it.".into()),
    }
}

/// Several expressions, one a line (or split by ";"), one result a line; an error on one
/// line doesn't stop the others. One expression gives its result, or its error.
pub fn calculate_many(text: &str) -> Result<String, String> {
    let expressions: Vec<&str> = text
        .split(['\n', ';'])
        .map(str::trim)
        .filter(|e| !e.is_empty())
        .collect();
    match expressions.len() {
        0 => Err("Give at least one expression.".into()),
        1 => calculate(expressions[0]),
        n if n > MAX_EXPRESSIONS => Err(format!("At most {MAX_EXPRESSIONS} expressions at once.")),
        _ => Ok(expressions
            .iter()
            .map(|e| calculate(e).unwrap_or_else(|why| format!("{e} → error: {why}")))
            .collect::<Vec<_>>()
            .join("\n")),
    }
}

import AppKit
import Foundation

private func vireoUncaughtExceptionHandler(_ exception: NSException) {
    // Written straight to a file: the process is about to die, and UserDefaults
    // may not reach disk before it does.
    let report = DiagnosticsService.buildSupportReport(exception: exception)
    let url = DiagnosticsService.crashReportURL
    try? FileManager.default.createDirectory(at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
    try? report.write(to: url, atomically: true, encoding: .utf8)
}

enum DiagnosticsService {
    static let crashReportURL = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
        .appendingPathComponent("VireoPrototype/last-crash.txt")

    private static let legacyCrashReportKey = "invurse_last_crash_report"

    static func installCrashHandlers() {
        NSSetUncaughtExceptionHandler(vireoUncaughtExceptionHandler)
    }

    /// Returns the report left by a crash in the previous session, removing it so it
    /// is offered only once.
    static func takeLastCrashReport() -> String? {
        UserDefaults.standard.removeObject(forKey: legacyCrashReportKey)
        guard let report = try? String(contentsOf: crashReportURL, encoding: .utf8) else { return nil }
        try? FileManager.default.removeItem(at: crashReportURL)
        return report
    }

    /// Tells the user the app crashed last time and offers to save the details.
    @MainActor
    static func offerLastCrashReport() {
        guard let crash = takeLastCrashReport() else { return }
        let alert = NSAlert()
        alert.messageText = "Vireo quit unexpectedly"
        alert.informativeText = "A report about the problem was saved. You can save it to a file to send to support."
        alert.addButton(withTitle: "Save Report…")
        alert.addButton(withTitle: "Dismiss")
        guard alert.runModal() == .alertFirstButtonReturn else { return }
        saveReport(crash + "\n\n" + currentStateSection())
    }

    @MainActor
    static func exportSupportReport() {
        saveReport(buildSupportReport(extraSections: [currentStateSection(), UpdateLog.section()]))
    }

    @MainActor
    private static func saveReport(_ report: String) {
        let panel = NSSavePanel()
        panel.nameFieldStringValue = "Vireo-support-report.txt"
        panel.allowedContentTypes = [.plainText]
        panel.message = "The report includes recent server logs, which may contain parts of your chats. Review it before sending."
        guard panel.runModal() == .OK, let url = panel.url else { return }
        do {
            try report.write(to: url, atomically: true, encoding: .utf8)
        } catch {
            let alert = NSAlert(error: error)
            alert.runModal()
        }
    }

    static func buildSupportReport(exception: NSException? = nil, extraSections: [String] = []) -> String {
        let appVersion = AppVersion.current
        let timestamp = ISO8601DateFormatter().string(from: Date())
        let os = ProcessInfo.processInfo.operatingSystemVersionString

        var lines: [String] = [
            "Vireo Support Report",
            "Timestamp: \(timestamp)",
            "App Version: \(appVersion)",
            "macOS: \(os)",
            "",
            "Throwable Details:",
        ]

        if let exception {
            lines.append("Exception Name: \(exception.name.rawValue)")
            lines.append("Exception Reason: \(exception.reason ?? "Unknown")")
            lines.append("Stack Trace:")
            lines.append(contentsOf: exception.callStackSymbols)
        } else {
            lines.append("No uncaught exception was recorded.")
        }

        for section in extraSections {
            lines.append("")
            lines.append(section)
        }
        return lines.joined(separator: "\n")
    }

    /// Hardware, server settings and status, recent server logs, and the newest
    /// macOS crash log for the app. Uncaught Objective-C exceptions are the only
    /// crashes the handler above sees; Swift traps and memory errors only appear in
    /// the system's crash logs.
    @MainActor
    static func currentStateSection() -> String {
        let server = LocalServerManager.shared
        let configuration = server.configuration
        let hardware = ModelManager.shared.hardwareProfile
        var lines: [String] = [
            "Hardware:",
            "- Chip: \(sysctlString("machdep.cpu.brand_string") ?? "Unknown")",
            "- Memory: \(hardware.memoryGB) GB",
            "- Free storage: \(hardware.freeStorageGB) GB",
            "",
            "Vireo engine:",
            "- Program: \(LocalServerManager.engineServerURL?.path ?? "Not included in this build")",
            "- Model: \(configuration.activeModelName.isEmpty ? "None" : configuration.activeModelName)",
            "- Model path: \(configuration.activeModelPath)",
            "- Engine: \(server.engine.status.label) on port \(configuration.apiPort)",
            "- App server URL: \(ServerConfig.baseURL.absoluteString)",
        ]
        if !server.startupMessage.isEmpty {
            lines.append("- Last message: \(server.startupMessage)")
        }
        lines.append("")
        lines.append("Engine Log (last 300 lines):")
        lines.append(contentsOf: server.engine.logLines.suffix(300))
        if let crashLog = newestSystemCrashLog() {
            lines.append("")
            lines.append("Newest macOS crash log (\(crashLog.name)):")
            lines.append(crashLog.contents)
        }
        return lines.joined(separator: "\n")
    }

    private static func newestSystemCrashLog() -> (name: String, contents: String)? {
        let directory = FileManager.default.homeDirectoryForCurrentUser
            .appendingPathComponent("Library/Logs/DiagnosticReports")
        let cutoff = Date().addingTimeInterval(-14 * 24 * 60 * 60)
        guard let entries = try? FileManager.default.contentsOfDirectory(
            at: directory,
            includingPropertiesForKeys: [.contentModificationDateKey]
        ) else { return nil }
        let newest = entries
            .filter { $0.lastPathComponent.hasPrefix("VireoMac") }
            .compactMap { url -> (URL, Date)? in
                guard let date = try? url.resourceValues(forKeys: [.contentModificationDateKey]).contentModificationDate,
                      date > cutoff else { return nil }
                return (url, date)
            }
            .max { $0.1 < $1.1 }
        guard let (url, _) = newest, let data = try? Data(contentsOf: url) else { return nil }
        let limit = 64 * 1024
        let text = String(decoding: data.prefix(limit), as: UTF8.self)
        return (url.lastPathComponent, data.count > limit ? text + "\n[truncated]" : text)
    }

    private static func sysctlString(_ name: String) -> String? {
        var size = 0
        guard sysctlbyname(name, nil, &size, nil, 0) == 0, size > 0 else { return nil }
        var buffer = [CChar](repeating: 0, count: size)
        guard sysctlbyname(name, &buffer, &size, nil, 0) == 0 else { return nil }
        return String(cString: buffer)
    }
}

===== crates/vireo-engine/src/tokenizer/mod.rs =====
//! Turns text into token ids and back, using the vocabulary inside a GGUF file.
//!
//! Supported so far: byte-level BPE (`tokenizer.ggml.model` = "gpt2"), which Llama 3 and
//! Qwen 2/3/3.5 use, with their pre-tokenizers; and Gemma 4's BPE ("gemma4"), which
//! works on the text's own UTF-8 characters with spaces written as "▁", splits only at
//! newlines, and falls back to `<0xXX>` byte tokens for characters not in the vocabulary.
//!
//! Encoding, as llama.cpp does it:
//! 1. With `parse_special`, special tokens written in the text (such as
//!    `<|start_header_id|>` from a chat template) are found first and become their own
//!    ids. User-defined tokens are always found.
//! 2. The rest is split into words by the pre-tokenizer.
//! 3. Each word's bytes are written in GPT-2's byte alphabet. If the whole word is a
//!    token (and the model's tokenizer does this, as Llama 3's does), that's its id;
//!    otherwise BPE merges it, starting from one token per byte.

mod bpe;
mod byte_map;
mod pretokenize;
pub mod unicode;
mod unicode_case_tables;
mod unicode_tables;

use std::fmt;

use crate::gguf::{Gguf, Value};
use crate::hash::FastHashMap;
use bpe::{Merge, Merges};
use byte_map::ByteMap;
pub use pretokenize::PreTokenizer;

/// GGUF's token types (`tokenizer.ggml.token_type`).
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum TokenType {
    Normal,
    Unknown,
    /// Special tokens that control the model, such as `<|begin_of_text|>`: only read
    /// from text when asked, and only shown when asked.
    Control,
    /// Tokens added to the vocabulary by hand: always read from text, shown as written.
    UserDefined,
    Unused,
    /// A token for one raw byte (SentencePiece vocabularies).
    Byte,
}

impl TokenType {
    fn from_gguf(id: i64) -> Self {
        match id {
            2 => Self::Unknown,
            3 => Self::Control,
            4 => Self::UserDefined,
            5 => Self::Unused,
            6 => Self::Byte,
            _ => Self::Normal,
        }
    }

    fn is_special(self) -> bool {
        matches!(self, Self::Control | Self::UserDefined | Self::Unknown)
    }
}

#[derive(Debug)]
pub enum Error {
    /// The file has no tokenizer, or is missing part of one.
    Missing(&'static str),
    /// A tokenizer this engine can't run yet.
    Unsupported(String),
    Invalid(String),
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Error::Missing(key) => write!(f, "the model file has no {key}"),
            Error::Unsupported(what) => write!(f, "{what} isn't supported yet"),
            Error::Invalid(what) => write!(f, "the model's tokenizer is damaged: {what}"),
        }
    }
}

impl std::error::Error for Error {}

pub type Result<T> = std::result::Result<T, Error>;

/// Text of tokens that end a model's turn, across the model families Vireo runs. As in
/// llama.cpp, a token found by its text here becomes a control token if the file didn't
/// already mark it so: it's then read from text only with `parse_special`.
const END_OF_GENERATION_TEXTS: &[&str] = &[
    "<|eot_id|>",
    "<|eom_id|>",
    "<|end_of_text|>",
    "<|im_end|>",
    "<|endoftext|>",
    "<end_of_turn>",
    "<|end|>",
    "<eos>",            // Gemma 4
    "<turn|>",          // Gemma 4
    "<|tool_response>", // Gemma 4
];

pub struct Tokenizer {
    tokens: Vec<String>,
    types: Vec<TokenType>,
    ids: FastHashMap<String, u32>,
    merges: Merges,
    bytes: ByteMap,
    /// The token for each single byte.
    byte_tokens: [Option<u32>; 256],
    pre: PreTokenizer,
    /// Use a word's own token when it has one, before merging (Llama 3).
    ignore_merges: bool,
    /// Gemma 4's style: UTF-8 characters, spaces as "▁", `<0xXX>` byte fallback.
    metaspace: bool,
    /// The `<0xXX>` token for each byte, in `metaspace` vocabularies.
    byte_fallback: [Option<u32>; 256],
    /// Special and user-defined tokens, longest first.
    specials: Vec<u32>,
    bos: Option<u32>,
    eos: Option<u32>,
    add_bos: bool,
    add_eos: bool,
    end_of_generation: Vec<u32>,
}

impl Tokenizer {
    /// The tokenizer described by a GGUF file's `tokenizer.ggml.*` metadata.
    pub fn from_gguf(gguf: &Gguf) -> Result<Self> {
        let model = gguf
            .get_str("tokenizer.ggml.model")
            .ok_or(Error::Missing("tokenizer (tokenizer.ggml.model)"))?;
        let metaspace = match model {
            "gpt2" => false,
            "gemma4" => true,
            other => return Err(Error::Unsupported(format!("The \"{other}\" tokenizer"))),
        };
        let pre = if metaspace {
            PreTokenizer::Gemma4
        } else {
            let pre_name = gguf.get_str("tokenizer.ggml.pre").unwrap_or("default");
            PreTokenizer::from_gguf_name(pre_name)
                .ok_or_else(|| Error::Unsupported(format!("The \"{pre_name}\" pre-tokenizer")))?
        };

        let tokens: Vec<String> = string_array(gguf, "tokenizer.ggml.tokens")?
            .ok_or(Error::Missing("vocabulary (tokenizer.ggml.tokens)"))?;
        let types = match gguf
            .get("tokenizer.ggml.token_type")
            .and_then(Value::as_array)
        {
            None => vec![TokenType::Normal; tokens.len()],
            Some(items) if items.len() == tokens.len() => items
                .iter()
                .map(|v| TokenType::from_gguf(v.as_f64().unwrap_or(1.0) as i64))
                .collect(),
            Some(items) => {
                return Err(Error::Invalid(format!(
                    "{} token types for {} tokens",
                    items.len(),
                    tokens.len()
                )));
            }
        };
        let merges = string_array(gguf, "tokenizer.ggml.merges")?
            .ok_or(Error::Missing("BPE merges (tokenizer.ggml.merges)"))?;

        let id = |key: &str| -> Result<Option<u32>> {
            match gguf.get_u64(key) {
                None => Ok(None),
                Some(id) if (id as usize) < tokens.len() => Ok(Some(id as u32)),
                Some(id) => Err(Error::Invalid(format!(
                    "{key} is {id}, past the end of the vocabulary"
                ))),
            }
        };
        let bos = id("tokenizer.ggml.bos_token_id")?;
        let eos = id("tokenizer.ggml.eos_token_id")?;
        let mut end_of_generation: Vec<u32> = [
            eos,
            id("tokenizer.ggml.eot_token_id")?,
            id("tokenizer.ggml.eom_token_id")?,
        ]
        .into_iter()
        .flatten()
        .collect();

        // llama.cpp's defaults for each pre-tokenizer, then the file's own settings.
        let (default_add_bos, ignore_merges) = match pre {
            PreTokenizer::Llama3 | PreTokenizer::Llama4 => (true, true),
            PreTokenizer::Qwen2 | PreTokenizer::Qwen35 => (false, false),
            PreTokenizer::Gemma4 => (true, false),
        };
        let flag =
            |key: &str, default: bool| gguf.get(key).and_then(Value::as_bool).unwrap_or(default);
        let add_bos = flag("tokenizer.ggml.add_bos_token", default_add_bos);
        let add_eos = flag("tokenizer.ggml.add_eos_token", false);

        let mut tokenizer = Self::bpe(tokens, types, &merges, pre, ignore_merges)?;
        if metaspace {
            tokenizer.metaspace = true;
            for (b, slot) in tokenizer.byte_fallback.iter_mut().enumerate() {
                *slot = tokenizer.ids.get(&format!("<0x{b:02X}>")).copied();
            }
        }
        for text in END_OF_GENERATION_TEXTS {
            if let Some(&id) = tokenizer.ids.get(*text) {
                tokenizer.types[id as usize] = TokenType::Control;
                if !end_of_generation.contains(&id) {
                    end_of_generation.push(id);
                }
            }
        }
        // Gemma 4 (llama.cpp does the same): with a tool-response token, "</s>" is
        // ordinary text and doesn't end a turn.
        if tokenizer.ids.contains_key("<|tool_response>")
            && let Some(&id) = tokenizer.ids.get("</s>")
        {
            tokenizer.types[id as usize] = TokenType::Normal;
            end_of_generation.retain(|&e| e != id);
        }
        tokenizer.refresh_specials();
        tokenizer.bos = bos;
        tokenizer.eos = eos;
        tokenizer.add_bos = add_bos && bos.is_some();
        tokenizer.add_eos = add_eos && eos.is_some();
        tokenizer.end_of_generation = end_of_generation;
        Ok(tokenizer)
    }

    /// A byte-level BPE tokenizer from its parts. `merges` are "left right" pairs, best
    /// first.
    fn bpe(
        tokens: Vec<String>,
        types: Vec<TokenType>,
        merges: &[String],
        pre: PreTokenizer,
        ignore_merges: bool,
    ) -> Result<Self> {
        if tokens.len() > u32::MAX as usize {
            return Err(Error::Invalid("the vocabulary is too large".into()));
        }
        let mut ids = FastHashMap::with_capacity_and_hasher(tokens.len(), Default::default());
        for (i, text) in tokens.iter().enumerate() {
            // A few vocabularies list a text twice; the first id wins, as in llama.cpp.
            ids.entry(text.clone()).or_insert(i as u32);
        }

        let mut merge_table = Merges::with_capacity_and_hasher(merges.len(), Default::default());
        for (rank, line) in merges.iter().enumerate() {
            // Split at the first space after the first character, as llama.cpp does.
            let split = line
                .char_indices()
                .skip(1)
                .find(|&(_, c)| c == ' ')
                .map(|(i, _)| i)
                .ok_or_else(|| Error::Invalid(format!("the merge {line:?} has no space")))?;
            let (left, right) = (&line[..split], &line[split + 1..]);
            let (Some(&l), Some(&r)) = (ids.get(left), ids.get(right)) else {
                continue;
            };
            let Some(&token) = ids.get(&format!("{left}{right}")) else {
                continue;
            };
            merge_table.entry((l, r)).or_insert(Merge {
                rank: rank as u32,
                token,
            });
        }

        let bytes = ByteMap::new();
        let mut byte_tokens = [None; 256];
        for (b, slot) in byte_tokens.iter_mut().enumerate() {
            *slot = ids
                .get(bytes.char_of(b as u8).to_string().as_str())
                .copied();
        }

        let mut tokenizer = Self {
            tokens,
            types,
            ids,
            merges: merge_table,
            bytes,
            byte_tokens,
            pre,
            ignore_merges,
            metaspace: false,
            byte_fallback: [None; 256],
            specials: Vec::new(),
            bos: None,
            eos: None,
            add_bos: false,
            add_eos: false,
            end_of_generation: Vec::new(),
        };
        tokenizer.refresh_specials();
        Ok(tokenizer)
    }

    /// The special and user-defined tokens to look for in text, longest first.
    fn refresh_specials(&mut self) {
        let (tokens, types) = (&self.tokens, &self.types);
        let mut specials: Vec<u32> = (0..tokens.len() as u32)
            .filter(|&i| types[i as usize].is_special() && !tokens[i as usize].is_empty())
            .collect();
        specials.sort_by_key(|&i| (std::cmp::Reverse(tokens[i as usize].len()), i));
        self.specials = specials;
    }

    pub fn vocab_size(&self) -> usize {
        self.tokens.len()
    }

    pub fn pre_tokenizer(&self) -> PreTokenizer {
        self.pre
    }

    /// A token's text as the vocabulary writes it (in GPT-2's byte alphabet: "Ġworld").
    pub fn token_text(&self, id: u32) -> Option<&str> {
        self.tokens.get(id as usize).map(String::as_str)
    }

    pub fn token_type(&self, id: u32) -> Option<TokenType> {
        self.types.get(id as usize).copied()
    }

    /// The id of the token whose vocabulary text is exactly `text`.
    pub fn token_id(&self, text: &str) -> Option<u32> {
        self.ids.get(text).copied()
    }

    pub fn bos(&self) -> Option<u32> {
        self.bos
    }

    pub fn eos(&self) -> Option<u32> {
        self.eos
    }

    /// Whether `encode` with `add_special` starts with the BOS token.
    pub fn adds_bos(&self) -> bool {
        self.add_bos
    }

    /// Tokens that end the model's turn: generation stops at any of them.
    pub fn is_end_of_generation(&self, id: u32) -> bool {
        self.end_of_generation.contains(&id)
    }

    /// Text to token ids. `add_special` adds the BOS (and EOS) tokens the model expects
    /// around a whole prompt; `parse_special` reads special tokens written in the text,
    /// such as a chat template's. These are llama.cpp's `llama_tokenize` arguments.
    pub fn encode(&self, text: &str, add_special: bool, parse_special: bool) -> Vec<u32> {
        let mut out = Vec::with_capacity(text.len() / 3 + 2);
        if add_special && self.add_bos {
            out.extend(self.bos);
        }
        for fragment in self.find_specials(text, parse_special) {
            match fragment {
                Fragment::Special(id) => out.push(id),
                Fragment::Text(t) => self.encode_text(t, &mut out),
            }
        }
        if add_special && self.add_eos {
            out.extend(self.eos);
        }
        out
    }

    /// Splits text around the special tokens written in it, longest tokens first.
    fn find_specials<'a>(&self, text: &'a str, parse_special: bool) -> Vec<Fragment<'a>> {
        let mut fragments = vec![Fragment::Text(text)];
        for &id in &self.specials {
            let ty = self.types[id as usize];
            if !parse_special && ty != TokenType::UserDefined {
                continue;
            }
            let special = self.tokens[id as usize].as_str();
            if !text.contains(special) {
                continue;
            }
            let mut split = Vec::with_capacity(fragments.len() + 2);
            for fragment in fragments {
                let Fragment::Text(mut rest) = fragment else {
                    split.push(fragment);
                    continue;
                };
                while let Some(at) = rest.find(special) {
                    if at > 0 {
                        split.push(Fragment::Text(&rest[..at]));
                    }
                    split.push(Fragment::Special(id));
                    rest = &rest[at + special.len()..];
                }
                if !rest.is_empty() {
                    split.push(Fragment::Text(rest));
                }
            }
            fragments = split;
        }
        fragments
    }

    fn encode_text(&self, text: &str, out: &mut Vec<u32>) {
        if self.metaspace {
            return self.encode_metaspace(text, out);
        }
        let mut words = Vec::new();
        self.pre.split(text, &mut words);
        let mut symbols = Vec::new();
        for word in words {
            if self.ignore_merges
                && let Some(&id) = self.ids.get(&self.bytes.encode(word))
            {
                out.push(id);
                continue;
            }
            symbols.clear();
            // Every byte-level vocabulary has a token for each byte; a byte without one
            // is dropped rather than guessed at.
            symbols.extend(word.bytes().filter_map(|b| self.byte_tokens[b as usize]));
            bpe::merge(&mut symbols, &self.merges);
            out.extend_from_slice(&symbols);
        }
    }

    /// Gemma 4: spaces become "▁"; a run of newlines that is a token is that token; the
    /// rest is merged from one symbol per character. A character with no token of its own
    /// (it can't be part of any merge) becomes its UTF-8 bytes' `<0xXX>` tokens.
    fn encode_metaspace(&self, text: &str, out: &mut Vec<u32>) {
        let escaped = text.replace(' ', "\u{2581}");
        let mut words = Vec::new();
        self.pre.split(&escaped, &mut words);
        let vocab = self.tokens.len() as u32;
        let mut symbols = Vec::new();
        for word in words {
            if word.bytes().all(|b| b == b'\n')
                && let Some(&id) = self.ids.get(word)
            {
                out.push(id);
                continue;
            }
            symbols.clear();
            let mut buf = [0u8; 4];
            // A character outside the vocabulary gets an id past its end, which no merge
            // can match: vocab + its code point.
            symbols.extend(word.chars().map(|c| {
                self.ids
                    .get(&*c.encode_utf8(&mut buf))
                    .copied()
                    .unwrap_or(vocab + c as u32)
            }));
            bpe::merge(&mut symbols, &self.merges);
            for &id in &symbols {
                if id < vocab {
                    out.push(id);
                } else {
                    let c = char::from_u32(id - vocab).expect("made from a char");
                    for &b in c.encode_utf8(&mut buf).as_bytes() {
                        out.extend(self.byte_fallback[b as usize]);
                    }
                }
            }
        }
    }

    /// Appends the bytes a token stands for. A token can be part of a character, so
    /// stream generated text through bytes and turn them into text when they're whole.
    /// Control tokens are left out unless `include_special`.
    pub fn token_bytes(&self, id: u32, include_special: bool, out: &mut Vec<u8>) {
        let Some(text) = self.tokens.get(id as usize) else {
            return;
        };
        match self.types[id as usize] {
            TokenType::Control | TokenType::Unknown => {
                if include_special {
                    out.extend_from_slice(text.as_bytes());
                }
            }
            TokenType::UserDefined => out.extend_from_slice(text.as_bytes()),
            TokenType::Unused => {}
            TokenType::Normal | TokenType::Byte if self.metaspace => {
                // "<0xXX>" is one raw byte; anything else is text with "▁" for spaces.
                let hex = text
                    .strip_prefix("<0x")
                    .and_then(|t| t.strip_suffix('>'))
                    .filter(|h| h.len() == 2)
                    .and_then(|h| u8::from_str_radix(h, 16).ok());
                match hex {
                    Some(b) => out.push(b),
                    None => out.extend_from_slice(text.replace('\u{2581}', " ").as_bytes()),
                }
            }
            TokenType::Normal | TokenType::Byte => {
                for c in text.chars() {
                    match self.bytes.byte_of(c) {
                        Some(b) => out.push(b),
                        None => out.extend_from_slice(c.encode_utf8(&mut [0; 4]).as_bytes()),
                    }
                }
            }
        }
    }

    /// Token ids to text. Bytes that don't make whole characters become U+FFFD.
    pub fn decode(&self, ids: &[u32], include_special: bool) -> String {
        let mut bytes = Vec::with_capacity(ids.len() * 4);
        for &id in ids {
            self.token_bytes(id, include_special, &mut bytes);
        }
        String::from_utf8_lossy(&bytes).into_owned()
    }
}

enum Fragment<'a> {
    Text(&'a str),
    Special(u32),
}

/// A metadata array of strings; `None` if the key isn't there.
fn string_array(gguf: &Gguf, key: &'static str) -> Result<Option<Vec<String>>> {
    let Some(value) = gguf.get(key) else {
        return Ok(None);
    };
    let items = value
        .as_array()
        .ok_or_else(|| Error::Invalid(format!("{key} isn't a list")))?;
    items
        .iter()
        .map(|v| {
            v.as_str()
                .map(str::to_owned)
                .ok_or_else(|| Error::Invalid(format!("{key} has an item that isn't text")))
        })
        .collect::<Result<Vec<_>>>()
        .map(Some)
}

    fn tiny(ignore_merges: bool) -> Tokenizer {

        let mut types = vec![TokenType::Normal; 256];

            types.push(ty);

            .iter()

        let mut t =

    fn texts(t: &Tokenizer, ids: &[u32]) -> Vec<String> {
        ids.iter()
            .map(|&id| t.token_text(id).unwrap().to_string())
            .collect()
    }

    #[test]
    fn merges_words() {
        let t = tiny(false);
        let ids = t.encode("hello world", false, false);
        assert_eq!(texts(&t, &ids), ["hello", "Ġwor", "l", "d"]);
        assert_eq!(t.decode(&ids, false), "hello world");
    }

    #[test]
    fn uses_whole_word_tokens_when_told_to() {
        let t = tiny(true);
        let ids = t.encode("hello world", false, false);
        assert_eq!(texts(&t, &ids), ["hello", "Ġworld"]);
    }

    #[test]
    fn adds_bos_and_reads_special_tokens() {
        let t = tiny(true);
        let ids = t.encode("hello<|eot|>", true, true);
        assert_eq!(texts(&t, &ids), ["<|bos|>", "hello", "<|eot|>"]);
        // The longer special token wins where both fit.
        let ids = t.encode("<|eot|>x", false, true);
        assert_eq!(texts(&t, &ids), ["<|eot|>x"]);

        assert_eq!(t.decode(&ids, false), "hello");
        assert_eq!(t.decode(&ids, true), "<|bos|>hello<|eot|>");
    }

import Speech

/// Turns the microphone into text with Apple's on-device speech recognition.
/// Audio never leaves the Mac: if the user's language can't be recognised on
/// device, voice input is unavailable rather than falling back to Apple's servers.
@MainActor
final class SpeechInput {
    enum Failure: LocalizedError {
        case needsPackagedApp
        case microphoneDenied
        case speechDenied
        case unsupportedLanguage
        case audio(String)

        var errorDescription: String? {
            switch self {
            case .needsPackagedApp:
                return "Voice input needs the packaged app (build-app.sh), which declares microphone use."
            case .microphoneDenied:
                return "Vireo isn't allowed to use the microphone. Turn it on in System Settings → Privacy & Security → Microphone."
            case .speechDenied:
                return "Vireo isn't allowed to use speech recognition. Turn it on in System Settings → Privacy & Security → Speech Recognition."
            case .unsupportedLanguage:
                return "On-device speech recognition isn't available for your language on this Mac."
            case .audio(let detail):
                return "The microphone couldn't start: \(detail)"
            }
        }
    }

    /// The phrase recognition is biased towards, for the wake word.
    var contextualStrings: [String] = []

    private let engine = AVAudioEngine()
    private var request: SFSpeechAudioBufferRecognitionRequest?
    private var task: SFSpeechRecognitionTask?
    private(set) var isRunning = false
    /// Results from a recognition that was replaced or cancelled are dropped.
    private var generation = 0

    /// Without these keys macOS ends the app when it asks for access.
    static var canAskForAccess: Bool {
        Bundle.main.object(forInfoDictionaryKey: "NSMicrophoneUsageDescription") != nil
            && Bundle.main.object(forInfoDictionaryKey: "NSSpeechRecognitionUsageDescription") != nil
    }

    /// Asks for the microphone and speech recognition the first time; later calls
    /// return the user's earlier answer without a prompt.
    static func requestAccess() async throws {
        guard canAskForAccess else { throw Failure.needsPackagedApp }
        guard await AVCaptureDevice.requestAccess(for: .audio) else { throw Failure.microphoneDenied }
        let status = await withCheckedContinuation { continuation in
            SFSpeechRecognizer.requestAuthorization { continuation.resume(returning: $0) }
        }
        guard status == .authorized else { throw Failure.speechDenied }
    }

    static var isAuthorized: Bool {
        AVCaptureDevice.authorizationStatus(for: .audio) == .authorized
            && SFSpeechRecognizer.authorizationStatus() == .authorized
    }

    /// Calls `onText` with the words so far and `true` once recognition settles
    /// (after `stop()`, or when the recognizer gives up).
    func start(onText: @escaping (String, Bool) -> Void) throws {
        cancel()
        guard let recognizer = SFSpeechRecognizer(), recognizer.supportsOnDeviceRecognition else {
            throw Failure.unsupportedLanguage
        }
        let request = SFSpeechAudioBufferRecognitionRequest()
        request.requiresOnDeviceRecognition = true
        request.shouldReportPartialResults = true
        request.addsPunctuation = true
        request.contextualStrings = contextualStrings
        self.request = request

        let input = engine.inputNode
        input.installTap(onBus: 0, bufferSize: 1024, format: input.outputFormat(forBus: 0)) { buffer, _ in
            request.append(buffer)
        }
        engine.prepare()
        do {
            try engine.start()
        } catch {
            input.removeTap(onBus: 0)
            throw Failure.audio(error.localizedDescription)
        }
        isRunning = true

        generation += 1
        let current = generation
        task = recognizer.recognitionTask(with: request) { [weak self] result, error in
            let text = result?.bestTranscription.formattedString ?? ""
            let settled = result?.isFinal == true || error != nil
            Task { @MainActor [weak self] in
                guard self?.generation == current else { return }
                onText(text, settled)
            }
        }
    }

    /// Stops listening; the recognizer then reports its final text.
    func stop() {
        guard isRunning else { return }
        isRunning = false
        engine.stop()
        engine.inputNode.removeTap(onBus: 0)
        request?.endAudio()
        request = nil
    }

    /// Stops without waiting for a final result.
    func cancel() {
        generation += 1
        stop()
        task?.cancel()
        task = nil
    }
}

                .tracking(0.6)
                .foregroundStyle(AppTheme.textSecondary)
            content()

    private func slider(
        _ title: String, value: Binding<Double>, range: ClosedRange<Double>, step: Double,
        offValue: Double = 0, offLabel: String? = nil, help: String
    ) -> some View {
        VStack(alignment: .leading, spacing: 2) {
            HStack {
                Text(title).font(.system(size: 12, weight: .medium))
                Spacer()
                Text(offLabel != nil && value.wrappedValue == offValue ? offLabel! : value.wrappedValue.formatted(.number.precision(.fractionLength(2))))
                    .font(.system(size: 12, design: .monospaced))
                    .foregroundStyle(AppTheme.textSecondary)
            }
            // Snapped to the step, then to two decimals (0.7, not 0.7000000000000001),
            // so presets are still recognised after dragging.
            Slider(value: Binding(
                get: { value.wrappedValue },
                set: { value.wrappedValue = (($0 / step).rounded() * step * 100).rounded() / 100 }
            ), in: range)
            .controlSize(.small)
            Text(help)
                .font(.caption)
                .foregroundStyle(AppTheme.textSecondary)
        }
    }

    private func intSlider(
        _ title: String, value: Binding<Int>, range: ClosedRange<Int>, step: Int,
        offLabel: String, help: String
    ) -> some View {
        VStack(alignment: .leading, spacing: 2) {
            HStack {
                Text(title).font(.system(size: 12, weight: .medium))
                Spacer()
                Text(value.wrappedValue == 0 ? offLabel : "\(value.wrappedValue)")
                    .font(.system(size: 12, design: .monospaced))
                    .foregroundStyle(AppTheme.textSecondary)
            }
            Slider(value: Binding(
                get: { Double(value.wrappedValue) },
                set: { value.wrappedValue = Int(($0 / Double(step)).rounded()) * step }
            ), in: Double(range.lowerBound)...Double(range.upperBound))
            .controlSize(.small)
            Text(help)
                .font(.caption)
                .foregroundStyle(AppTheme.textSecondary)
        }
    }
}

/// Choose a model for a chat or Quick Ask: nil keeps whichever model is loaded.
struct ModelChoicePicker: View {
    @EnvironmentObject private var modelManager: ModelManager
    let title: String
    @Binding var selection: String?

    var body: some View {
        Picker(title, selection: $selection) {
            Text("Whichever is loaded").tag(String?.none)
            Divider()
            ForEach(modelManager.installedLocalModels) { model in
                Text(ModelPickerView.displayName(for: model.name, catalog: modelManager.catalogModels))
                    .tag(String?.some(model.path))
            }
            // A chosen model that has since been deleted or moved stays visible.
            if let selection, !modelManager.installedLocalModels.contains(where: { $0.path == selection }) {
                Text("\(URL(fileURLWithPath: selection).lastPathComponent) (not found)").tag(String?.some(selection))
            }
        }
        .onAppear { modelManager.refresh() }
    }
}
