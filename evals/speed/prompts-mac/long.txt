import Foundation

enum LocalModelFamily: String, Codable, CaseIterable {
    case gemma
    case qwen
    case other

    var displayName: String {
        switch self {
        case .gemma: return "Gemma"
        case .qwen: return "Qwen"
        case .other: return "Other"
        }
    }
}

enum LocalModelCatalogRules {
    static func family(forRepository repository: String) -> LocalModelFamily? {
        let normalizedRepository = repository.lowercased()
        if normalizedRepository.contains("gemma") {
            return .gemma
        }
        if normalizedRepository.contains("qwen") {
            return .qwen
        }
        return nil
    }

    /// Only the current generation of each family is offered for download: Gemma 4
    /// and Qwen 3.8. Older releases (Gemma 2/3, Qwen 2/2.5/3) are left out.
    /// "Qwen3-8B" is Qwen 3 at 8B parameters, not Qwen 3.8, so the dot is required.
    static func isCurrentGeneration(_ repository: String) -> Bool {
        let name = repository.lowercased()
        switch family(forRepository: name) {
        case .gemma:
            return name.range(of: #"gemma-?4(?![0-9.])"#, options: .regularExpression) != nil
        case .qwen:
            return name.range(of: #"qwen3\.8(?![0-9])"#, options: .regularExpression) != nil
        case .other, nil:
            return false
        }
    }

    /// Speculative-decoding drafters ("-assistant"), downloaded automatically
    /// alongside their target model. They can't be used as a chat model.
    static func isDraftRepository(_ repository: String) -> Bool {
        repository.lowercased().contains("-assistant")
    }

    /// An official release converted to MLX, as opposed to a community fine-tune.
    /// Its `base_model:` tag must point at the family's publisher, and its name must be
    /// that base model's name plus format words only (quantization, MLX, MTP). The name
    /// check matters: some fine-tunes still tag the original Google model as their base.
    static func isOfficialRelease(_ repository: String, tags: [String]) -> Bool {
        guard let family = family(forRepository: repository),
              let publisher = officialPublisher(for: family) else { return false }
        let repositoryName = (repository.split(separator: "/").last.map(String.init) ?? repository).lowercased()
        let baseModels = tags
            .filter { $0.hasPrefix("base_model:") }
            .compactMap { $0.split(separator: ":").last.map { $0.lowercased() } }
        return baseModels.contains { baseModel in
            let parts = baseModel.split(separator: "/", maxSplits: 1).map(String.init)
            guard parts.count == 2, parts[0] == publisher, repositoryName.hasPrefix(parts[1]) else { return false }
            return repositoryName.dropFirst(parts[1].count)
                .split(whereSeparator: { $0 == "-" || $0 == "_" })
                .allSatisfy { $0.range(of: formatWordPattern, options: .regularExpression) != nil }
        }
    }

    private static let formatWordPattern =
        #"^(\d+(\.\d+)?bits?|bf16|fp16|fp8|int[48]|mxfp[48]|nvfp4|optiq|dwq|mlx|mtp|msq|oq\d+e?|q\d+)$"#

    private static func officialPublisher(for family: LocalModelFamily) -> String? {
        switch family {
        case .gemma: return "google"
        case .qwen: return "qwen"
        case .other: return nil
        }
    }

    static func supportedModelTypes(for family: LocalModelFamily) -> Set<String> {
        switch family {
        case .gemma:
            return ["gemma4", "gemma4_unified"]
        case .qwen:
            return ["qwen2", "qwen2_5", "qwen3", "qwen3_8"]
        case .other:
            return []
        }
    }

    static func isCompatibleModelType(_ modelType: String, family: LocalModelFamily, allowedModelTypes: Set<String>) -> Bool {
        if allowedModelTypes.contains(modelType) {
            return true
        }
        switch family {
        case .qwen:
            return modelType.hasPrefix("qwen")
        case .gemma:
            return modelType.hasPrefix("gemma")
        case .other:
            return false
        }
    }

    static func isInstructionTunedRepository(_ repository: String) -> Bool {
        let name = repository.lowercased()
        if name.contains("-it-") || name.contains("-instruct") || name.hasSuffix("-it") {
            return true
        }
        if name.contains("qwen3.8") {
            let excludedMarkers = ["base", "embed", "rerank", "reward"]
            if excludedMarkers.contains(where: { name.contains($0) }) {
                return false
            }
            if ["flash", "chat", "next", "reason"].contains(where: { name.contains($0) }) {
                return true
            }
            return name.range(of: #"qwen3\.8[-_][0-9]+(?:\.[0-9]+)?b"#, options: .regularExpression) != nil
        }
        return false
    }
}

    let isRemote: Bool

struct DraftModel: Identifiable, Hashable {
    let id: String
    let repository: String
    let approximateSizeGB: Double
    /// The drafter's config.json model_type, checked before it downloads.
    var modelType: String = "gemma4_assistant"

    var installedPath: URL {
        ModelStorage.draftModelsDirectory.appendingPathComponent(id)
    }

    /// Each target model's drafter: Google's Gemma 4 assistants (mlx-community's MLX copies)
    /// and Meta's own Muse Glimmer assistant, which mlx_vlm loads straight from Meta's repo.
    static let catalog: [(target: String, draft: DraftModel)] = [
        ("gemma-4-E2B", DraftModel(id: "gemma-4-E2B-it-assistant-bf16", repository: "mlx-community/gemma-4-E2B-it-assistant-bf16", approximateSizeGB: 0.16)),
        ("gemma-4-E4B", DraftModel(id: "gemma-4-E4B-it-assistant-bf16", repository: "mlx-community/gemma-4-E4B-it-assistant-bf16", approximateSizeGB: 0.16)),
        ("gemma-4-26B", DraftModel(id: "gemma-4-26B-A4B-it-assistant-bf16", repository: "mlx-community/gemma-4-26B-A4B-it-assistant-bf16", approximateSizeGB: 0.84)),
        ("gemma-4-31B", DraftModel(id: "gemma-4-31B-it-assistant-bf16", repository: "mlx-community/gemma-4-31B-it-assistant-bf16", approximateSizeGB: 0.94)),
        ("muse-glimmer-30B", DraftModel(id: "Muse-Glimmer-30B-assistant-bf16", repository: "meta-models/Muse-Glimmer-30B-assistant",
                                        approximateSizeGB: 5.11, modelType: "muse_glimmer_assistant")),
    ]

    static func forTarget(_ targetID: String) -> DraftModel? {
        let normalizedTargetID = targetID.lowercased()
        return catalog.first(where: { normalizedTargetID.contains($0.target.lowercased()) })?.draft
    }

    /// The model a drafter is made for, as a readable name ("Gemma 4 31B"), from its folder name.
    static func targetName(forFolder folder: String) -> String? {
        guard let target = catalog.first(where: { $0.draft.id.caseInsensitiveCompare(folder) == .orderedSame })?.target else { return nil }
        return target.replacingOccurrences(of: "-", with: " ")
            .replacingOccurrences(of: "gemma 4", with: "Gemma 4", options: .caseInsensitive)
            .replacingOccurrences(of: "muse glimmer", with: "Muse Glimmer", options: .caseInsensitive)
    }
}

struct HardwareProfile {
    let memoryGB: Int
    let freeStorageGB: Int
    let isAppleSilicon: Bool

    var recommendedModel: CatalogModel? {
        recommendedModel(from: ModelCatalog.featuredModels)
    }

    func recommendedModel(from models: [CatalogModel], family: LocalModelFamily? = nil) -> CatalogModel? {
        let pool = family == nil ? models : models.filter { $0.family == family }
        guard !pool.isEmpty else { return nil }

        let storageLimit = Double(freeStorageGB) * 0.8
        let fitting = pool.filter {
            $0.minimumMemoryGB <= memoryGB && $0.approximateSizeGB < storageLimit
        }

        if let best = fitting.max(by: { lhs, rhs in
            if lhs.minimumMemoryGB != rhs.minimumMemoryGB {
                return lhs.minimumMemoryGB < rhs.minimumMemoryGB
            }
            return lhs.approximateSizeGB < rhs.approximateSizeGB
        }) {
            return best
        }

        return pool
            .filter { $0.minimumMemoryGB <= memoryGB }
            .max(by: { $0.minimumMemoryGB < $1.minimumMemoryGB })
    }
}

final class ModelManager: ObservableObject {
    static let shared = ModelManager()

    @Published private(set) var installedModels: Set<String> = []

    @Published private(set) var downloadingModelID: String?
    @Published private(set) var downloadingDraftModelID: String?
    @Published private(set) var draftDownloadProgress: Double = 0
    @Published private(set) var draftDownloadSpeedBytesPerSecond: Double = 0
    @Published private(set) var downloadProgress: Double = 0
    @Published private(set) var downloadSpeedBytesPerSecond: Double = 0
    @Published private(set) var statusMessage = ""
    @Published private(set) var errorMessage: String?
    @Published private(set) var catalogModels = ModelCatalog.featuredModels
    @Published private(set) var repositorySizes: [String: Int64] = [:]
    @Published private(set) var isLoadingCatalog = false
    /// Folders the user added: a model folder, or a folder of model folders.
    @Published private(set) var modelFolders: [String] = []
    @Published private(set) var searchResults: [CatalogModel] = []
    @Published private(set) var isSearching = false
    @Published private(set) var searchError: String?

    @Published private(set) var remoteDetails: [String: RemoteModelDetails] = [:]
    /// Architectures the bundled runtime can load; nil until known.
    @Published private(set) var supportedModelTypes: Set<String>?

    private let fileManager = FileManager.default
    private static let downloadSession: URLSession = {
        let configuration = URLSessionConfiguration.default
        configuration.timeoutIntervalForRequest = 300
        configuration.timeoutIntervalForResource = 86_400
        configuration.waitsForConnectivity = true
        configuration.httpMaximumConnectionsPerHost = 4
        return URLSession(configuration: configuration)
    }()
    private let setupCompletedKey = "invurse_model_setup_completed"
    private var refreshTask: Task<Void, Never>?
    private var repositorySizeTask: Task<Void, Never>?
    private var activeDownloadTask: Task<Void, Never>?
    private var catalogLoaded = false
    private var activeDownloadCancellation: DownloadCancellation?
    private var searchTask: Task<Void, Never>?
    private var detailsTask: Task<Void, Never>?
    private let modelFoldersKey = "invurse_model_folders"

    private init() {
        modelFolders = UserDefaults.standard.stringArray(forKey: modelFoldersKey) ?? []
        refresh()
        Task { @MainActor [weak self] in
            await self?.downloadDraftForActiveModelIfNeeded()
        }
    }

    var hardwareProfile: HardwareProfile {
        let volume = try? URL(fileURLWithPath: NSHomeDirectory()).resourceValues(forKeys: [.volumeAvailableCapacityForImportantUsageKey])
        let available = volume?.volumeAvailableCapacityForImportantUsage ?? 0
        #if arch(arm64)
        let appleSilicon = true
        #else
        let appleSilicon = false
        #endif
        return HardwareProfile(
            memoryGB: Int(ProcessInfo.processInfo.physicalMemory / 1_073_741_824),
            freeStorageGB: Int(available / 1_073_741_824),
            isAppleSilicon: appleSilicon
        )
    }

    var recommendedModelsByFamily: [CatalogModel] {
        // Only official releases are recommended.
        let official = catalogModels.filter(\.isOfficial)
        return [LocalModelFamily.gemma, .qwen].compactMap { family in
            hardwareProfile.recommendedModel(from: official, family: family)
        }
    }

    var recommendedModelIDs: Set<String> {
        Set(recommendedModelsByFamily.map(\.id))
    }

    func refresh() {
        refreshTask?.cancel()
        let directory = ModelStorage.modelsDirectory
        let folders = modelFolders
        refreshTask = Task { @MainActor [weak self] in
            let (downloaded, external) = await Task.detached(priority: .userInitiated) {
                (Self.installedModels(in: directory), Self.models(inAddedFolders: folders))
            }.value
            guard !Task.isCancelled else { return }
            var seen = Set(downloaded.map(\.path))
            self?.installedLocalModels = downloaded + external.filter { seen.insert($0.path).inserted }
            // Only the app's own folder counts toward "installed" catalog entries:
            // their Use button loads the copy in that folder.
            self?.installedModels = Set(downloaded.map { $0.name })
        }
    }

    /// Models in the folders the user added. A folder is either a model itself
    /// or holds model folders.
    nonisolated static func models(inAddedFolders folders: [String]) -> [LocalModel] {
        folders.flatMap { path in
            ModelScanner.models(at: URL(fileURLWithPath: path)).map { model in
                var model = model
                model.isExternal = true
                return model
            }
        }
    }

    enum AddFolderResult: Equatable {
        case added(modelCount: Int)
        case alreadyAdded
        case notAModel
    }

    /// Adds a model folder, or a folder of models, to the installed list without copying it.
    func addModelFolder(_ url: URL) -> AddFolderResult {
        let path = url.standardizedFileURL.path
        guard !modelFolders.contains(path), path != ModelStorage.modelsDirectory.standardizedFileURL.path else {
            return .alreadyAdded
        }
        let count = ModelScanner.models(at: url).count
        guard count > 0 else { return .notAModel }
        modelFolders.append(path)
        UserDefaults.standard.set(modelFolders, forKey: modelFoldersKey)
        refresh()
        return .added(modelCount: count)
    }

    /// Stops listing a folder's models. Nothing is deleted.
    func removeModelFolder(_ path: String) {
        modelFolders.removeAll { $0 == path }
        UserDefaults.standard.set(modelFolders, forKey: modelFoldersKey)
        refresh()
    }

    /// The added folder a model was found in.
    func addedFolder(containing model: LocalModel) -> String? {
        modelFolders.first { model.path == $0 || model.path.hasPrefix($0.hasSuffix("/") ? $0 : $0 + "/") }
    }

    /// Deletes a model the app downloaded. Models in added folders are never deleted.
    func deleteDownloadedModel(_ model: LocalModel) {
        guard !model.isExternal, downloadingModelID == nil else { return }
        guard UserDefaults.standard.string(forKey: "invurse_local_model_path") != model.path else {
            errorMessage = "Choose another model before deleting the one in use."
            return
        }
        let url = URL(fileURLWithPath: model.path)
        if model.format == .gguf {
            Self.deleteGGUF(at: url)
        } else {
            try? fileManager.removeItem(at: url)
        }
        refresh()
    }

    /// Deletes a GGUF model's file (every part of a split one). Once no other model
    /// is left in its folder, the folder goes too, with any vision projector in it,
    /// and so do the empty publisher folders above it.
    nonisolated static func deleteGGUF(at url: URL, modelsRoot: URL = ModelStorage.modelsDirectory) {
        let fileManager = FileManager.default
        let folder = url.deletingLastPathComponent()
        let partPattern = #"-\d{5}-of-\d{5}\.gguf$"#
        let fileName = url.lastPathComponent
        if let part = fileName.range(of: partPattern, options: [.regularExpression, .caseInsensitive]) {
            let stem = String(fileName[..<part.lowerBound])
            for sibling in (try? fileManager.contentsOfDirectory(atPath: folder.path)) ?? []
            where sibling.hasPrefix(stem + "-") && sibling.range(of: partPattern, options: [.regularExpression, .caseInsensitive]) != nil {
                try? fileManager.removeItem(at: folder.appendingPathComponent(sibling))
            }
        } else {
            try? fileManager.removeItem(at: url)
        }

        var directory = folder
        let root = modelsRoot.standardizedFileURL.path
        while directory.standardizedFileURL.path.hasPrefix(root + "/"), ModelScanner.models(at: directory).isEmpty {
            try? fileManager.removeItem(at: directory)
            directory = directory.deletingLastPathComponent()
        }
    }

    func loadSupportedModelTypes() async {
        guard supportedModelTypes == nil else { return }
        supportedModelTypes = await LocalServerManager.shared.supportedModelTypes()
    }

    /// Whether the runtime can load this architecture; nil when unknown.
    func isSupported(modelType: String?) -> Bool? {
        guard let supportedModelTypes, let modelType else { return nil }
        return supportedModelTypes.contains(modelType.lowercased())
    }

    func memoryFit(sizeBytes: Int64?) -> MemoryFit? {
        guard let sizeBytes, sizeBytes > 0 else { return nil }
        return MemoryFit.assess(weightBytes: sizeBytes, budgetBytes: MemoryFit.gpuBudgetBytes)
    }

    /// Waits for the scan started by the latest `refresh()`.
    func waitForRefresh() async {
        await refreshTask?.value
    }

    /// Every model folder on disk, not only the ones in the catalog: models from the
    /// remote catalog must still show up before that catalog has been fetched again.
    /// Folders are named after the catalog ID, so `isInstalled(_:)` still matches.
    nonisolated static func installedModels(in directory: URL, createIfMissing: Bool = true) -> [LocalModel] {
        if createIfMissing {
            try? FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        }
        return ModelScanner.models(at: directory)
    }

    func loadCatalog(force: Bool = false) async {
        guard !isLoadingCatalog, force || !catalogLoaded else { return }
        isLoadingCatalog = true
        defer { isLoadingCatalog = false }

        do {
            let remoteModels = try await fetchRemoteCatalog()
            catalogModels = mergeCatalog(remoteModels)
            catalogLoaded = true
            refresh()
            Task { @MainActor [weak self] in
                await self?.loadMissingRepositorySizes()
            }
        } catch {
            catalogLoaded = true
            errorMessage = "Could not refresh the model catalog. Showing featured models."
        }
    }

    func isInstalled(_ model: CatalogModel) -> Bool {
        installedModels.contains(model.id)
    }

    var hasCompletedSetup: Bool {
        UserDefaults.standard.bool(forKey: setupCompletedKey)
    }

    func markSetupCompleted() {
        UserDefaults.standard.set(true, forKey: setupCompletedKey)
    }

    func install(_ model: CatalogModel) async {
        guard downloadingModelID == nil else { return }
        guard !isInstalled(model) else {
            statusMessage = "\(model.displayName) is already installed."
            errorMessage = nil
            return
        }
        guard hardwareProfile.isAppleSilicon else {
            errorMessage = "Local MLX models require an Apple Silicon Mac."
            return
        }

        let task = Task { @MainActor [weak self] in
            guard let self else { return }
            await self.performInstall(model)
        }
        activeDownloadTask = task
        await task.value
        if activeDownloadTask != nil { activeDownloadTask = nil }
    }

    private func performInstall(_ model: CatalogModel) async {

        repositorySizeTask?.cancel()
        repositorySizeTask = nil
        downloadingModelID = model.id
        let cancellation = DownloadCancellation()
        activeDownloadCancellation = cancellation
        downloadProgress = 0
        downloadSpeedBytesPerSecond = 0
        errorMessage = nil
        let temporaryDirectory = ModelStorage.modelsDirectory.appendingPathComponent(".download-\(model.id)")

        do {
            try fileManager.createDirectory(at: temporaryDirectory, withIntermediateDirectories: true)
            let files: [RepositoryFile]
            if model.isSearchResult {
                await loadSupportedModelTypes()
                files = try await searchResultFiles(for: model.repository)
            } else {
                files = try await repositoryFiles(
                    for: model.repository,
                    family: model.family,
                    allowedModelTypes: LocalModelCatalogRules.supportedModelTypes(for: model.family)
                )
            }
            guard !files.isEmpty else { throw ModelError.noFiles }
            let downloadSize = files.reduce(Int64(0)) { $0 + ($1.sizeBytes ?? 0) }
            let requiredStorageGB = model.approximateSizeGB > 0
                ? model.approximateSizeGB * 1.2
                : Double(downloadSize) / 1_073_741_824 * 1.2
            guard hardwareProfile.freeStorageGB >= Int(ceil(requiredStorageGB)) else {
                throw ModelError.insufficientStorage
            }

            try await download(files, to: temporaryDirectory, cancellation: cancellation, progress: { [weak self] fraction, completed, total, speed in
                // A last report can arrive after the install finished; don't let it
                // replace "Installed" with "Downloading".
                guard self?.downloadingModelID == model.id else { return }
                self?.downloadProgress = fraction
                self?.downloadSpeedBytesPerSecond = speed
                self?.statusMessage = "Downloading model (\(completed) of \(total))"
            })

            guard fileManager.fileExists(atPath: temporaryDirectory.appendingPathComponent("config.json").path) else {
                throw ModelError.invalidModel
            }
            try? fileManager.removeItem(at: model.installedPath)
            try fileManager.moveItem(at: temporaryDirectory, to: model.installedPath)
            refresh()
            statusMessage = "Installed \(model.displayName)"
            if let draft = model.draft, !isDraftInstalled(draft) {
                Task { @MainActor [weak self] in
                    await self?.installDraft(draft)
                }
            }
        } catch {
            if Task.isCancelled || error is CancellationError {
                statusMessage = "Download cancelled. Completed files are kept for retry."
            } else if case ModelError.cancelled = error {
                statusMessage = "Download cancelled. Completed files are kept for retry."
            } else {
                errorMessage = error.localizedDescription
                statusMessage = "Download paused. Retry to continue."
            }
        }
        downloadingModelID = nil
        activeDownloadCancellation = nil
        downloadSpeedBytesPerSecond = 0
    }

    // MARK: - GGUF downloads

    /// Where GGUF files are stored: Models/<publisher>/<repository>/, as LM Studio
    /// lays them out, so several versions of one model sit side by side and
    /// same-named repositories from different publishers don't collide.
    nonisolated static func ggufFolder(for model: CatalogModel) -> URL {
        model.repository.split(separator: "/").reduce(ModelStorage.modelsDirectory) {
            $0.appendingPathComponent(String($1), isDirectory: true)
        }
    }

    func installedPath(for model: CatalogModel, option: GGUFOption) -> URL {
        Self.ggufFolder(for: model).appendingPathComponent(option.path)
    }

    func isInstalled(_ model: CatalogModel, option: GGUFOption) -> Bool {
        let folder = Self.ggufFolder(for: model)
        return option.parts.allSatisfy { fileManager.fileExists(atPath: folder.appendingPathComponent($0.path).path) }
    }

    func installGGUF(_ model: CatalogModel, option: GGUFOption, projector: RemoteFileInfo?) async {
        guard downloadingModelID == nil else { return }
        guard hardwareProfile.isAppleSilicon else {
            errorMessage = "Local models require an Apple Silicon Mac."
            return
        }
        let task = Task { @MainActor [weak self] in
            guard let self else { return }
            await self.performInstallGGUF(model, option: option, projector: projector)
        }
        activeDownloadTask = task
        await task.value
        if activeDownloadTask != nil { activeDownloadTask = nil }
    }

    private func performInstallGGUF(_ model: CatalogModel, option: GGUFOption, projector: RemoteFileInfo?) async {
        repositorySizeTask?.cancel()
        repositorySizeTask = nil
        downloadingModelID = model.repository
        let cancellation = DownloadCancellation()
        activeDownloadCancellation = cancellation
        downloadProgress = 0
        downloadSpeedBytesPerSecond = 0
        errorMessage = nil
        let folder = Self.ggufFolder(for: model)
        let temporaryDirectory = ModelStorage.modelsDirectory
            .appendingPathComponent(".download-\(model.repository.replacingOccurrences(of: "/", with: "--"))")

        do {
            var parts = option.parts
            if let projector, !fileManager.fileExists(atPath: folder.appendingPathComponent(projector.path).path) {
                parts.append(projector)
            }
            let requiredStorageGB = Double(parts.reduce(Int64(0)) { $0 + $1.sizeBytes }) / 1_073_741_824 * 1.2
            guard hardwareProfile.freeStorageGB >= Int(ceil(requiredStorageGB)) else {
                throw ModelError.insufficientStorage
            }
            let files = parts.map { part in
                let encoded = part.path.addingPercentEncoding(withAllowedCharacters: .urlPathAllowed) ?? part.path
                return RepositoryFile(
                    name: part.path,
                    url: URL(string: "https://huggingface.co/\(model.repository)/resolve/main/\(encoded)?download=true")!,
                    sizeBytes: part.sizeBytes > 0 ? part.sizeBytes : nil
                )
            }
            try fileManager.createDirectory(at: temporaryDirectory, withIntermediateDirectories: true)
            try await download(files, to: temporaryDirectory, cancellation: cancellation, progress: { [weak self] fraction, completed, total, speed in
                guard self?.downloadingModelID == model.repository else { return }
                self?.downloadProgress = fraction
                self?.downloadSpeedBytesPerSecond = speed
                self?.statusMessage = "Downloading \(option.quantization) (\(completed) of \(total))"
            })
            for part in parts {
                let source = temporaryDirectory.appendingPathComponent(part.path)
                let destination = folder.appendingPathComponent(part.path)
                try fileManager.createDirectory(at: destination.deletingLastPathComponent(), withIntermediateDirectories: true)
                try? fileManager.removeItem(at: destination)
                try fileManager.moveItem(at: source, to: destination)
            }
            try? fileManager.removeItem(at: temporaryDirectory)
            guard GGUFMetadata.architecture(ofFileAt: installedPath(for: model, option: option)) != nil else {
                throw ModelError.invalidModel
            }
            refresh()
            statusMessage = "Installed \(model.displayName) (\(option.quantization))"
        } catch {
            if Task.isCancelled || error is CancellationError {
                statusMessage = "Download cancelled. Completed files are kept for retry."
            } else if case ModelError.cancelled = error {
                statusMessage = "Download cancelled. Completed files are kept for retry."
            } else {
                errorMessage = error.localizedDescription
                statusMessage = "Download paused. Retry to continue."
            }
        }
        downloadingModelID = nil
        activeDownloadCancellation = nil
        downloadSpeedBytesPerSecond = 0
    }

    func cancelDownload() {
        activeDownloadCancellation?.cancel()
        activeDownloadTask?.cancel()
    }

    private func installDraft(_ draft: DraftModel) async {
        guard downloadingDraftModelID == nil else { return }
        let task = Task { @MainActor [weak self] in
            guard let self else { return }
            await self.performInstallDraft(draft)
        }
        activeDownloadTask = task
        await task.value
        if activeDownloadTask != nil { activeDownloadTask = nil }
    }

    private func performInstallDraft(_ draft: DraftModel) async {
        repositorySizeTask?.cancel()
        repositorySizeTask = nil
        downloadingDraftModelID = draft.id
        let cancellation = DownloadCancellation()
        activeDownloadCancellation = cancellation
        draftDownloadProgress = 0
        draftDownloadSpeedBytesPerSecond = 0
        statusMessage = "Preparing draft model \(draft.id)"
        let temporaryDirectory = ModelStorage.draftModelsDirectory.appendingPathComponent(".download-\(draft.id)")

        do {
            try fileManager.createDirectory(at: temporaryDirectory, withIntermediateDirectories: true)
            let files = try await repositoryFiles(
                for: draft.repository,
                family: .other,
                allowedModelTypes: [draft.modelType]
            )
            guard !files.isEmpty else { throw ModelError.noFiles }
            try await download(files, to: temporaryDirectory, cancellation: cancellation, progress: { [weak self] fraction, completed, total, speed in
                self?.draftDownloadProgress = fraction
                self?.draftDownloadSpeedBytesPerSecond = speed
                self?.statusMessage = "Downloading draft model (\(completed) of \(total))"
            })
            guard fileManager.fileExists(atPath: temporaryDirectory.appendingPathComponent("config.json").path) else {
                throw ModelError.invalidModel
            }
            try? fileManager.removeItem(at: draft.installedPath)
            try fileManager.moveItem(at: temporaryDirectory, to: draft.installedPath)
            statusMessage = "Installed draft model \(draft.id)"
        } catch {
            if Task.isCancelled || error is CancellationError {
                statusMessage = "Draft download cancelled. Completed files are kept for retry."
            } else if case ModelError.cancelled = error {
                statusMessage = "Draft download cancelled. Completed files are kept for retry."
            } else {
                statusMessage = "Draft model unavailable; standard decoding remains available"
                errorMessage = "Could not download the matching draft model: \(error.localizedDescription)"
            }
        }
        downloadingDraftModelID = nil
        activeDownloadCancellation = nil
        draftDownloadSpeedBytesPerSecond = 0
    }

    private func repositoryFiles(
        for repository: String,
        family: LocalModelFamily,

            throw ModelError.incompatibleModel
        }
        return Self.repositoryFiles(from: detail, repository: repository, revision: revision)
    }

    /// Downloads up to four files at a time. Progress counts bytes received,
    /// including files still in flight, so large shards don't sit at 0% for minutes.
    private func download(
        _ files: [RepositoryFile],
        to directory: URL,
        cancellation: DownloadCancellation,
        progress: @escaping @MainActor (_ fraction: Double, _ completedFiles: Int, _ totalFiles: Int, _ bytesPerSecond: Double) -> Void
    ) async throws {
        let pendingFiles = files.enumerated().filter { _, file in
            let destination = directory.appendingPathComponent(file.name)
            return !((try? destination.resourceValues(forKeys: [.fileSizeKey]).fileSize).map(Int64.init) == file.sizeBytes && file.sizeBytes != nil)
        }
        let counters = DownloadCounters()
        counters.completedFiles = files.count - pendingFiles.count
        counters.finishedBytes = files.enumerated().reduce(Int64(0)) { total, item in
            let (index, file) = item
            return pendingFiles.contains(where: { $0.offset == index }) ? total : total + (file.sizeBytes ?? 0)
        }
        let sizes = files.compactMap(\.sizeBytes)
        let totalBytes = sizes.count == files.count ? sizes.reduce(0, +) : 0

        func report(speed: Double) {
            let received = counters.finishedBytes + cancellation.inFlightBytes
            let fraction = totalBytes > 0
                ? min(Double(received) / Double(totalBytes), 1)
                : Double(counters.completedFiles) / Double(max(files.count, 1))
            // A finished file leaves the in-flight count a moment before it is added
            // to the finished bytes; don't let the bar step backwards meanwhile.
            counters.reportedFraction = max(counters.reportedFraction, fraction)
            progress(counters.reportedFraction, counters.completedFiles, files.count, speed)
        }

        report(speed: 0)
        let poller = Task { @MainActor in
            var lastBytes = counters.finishedBytes + cancellation.inFlightBytes
            var lastTime = Date()
            var speed = 0.0
            while !Task.isCancelled {
                try? await Task.sleep(for: .milliseconds(500))
                let bytes = counters.finishedBytes + cancellation.inFlightBytes
                let now = Date()
                let instant = Double(max(bytes - lastBytes, 0)) / max(now.timeIntervalSince(lastTime), 0.001)
                speed = speed == 0 ? instant : speed * 0.7 + instant * 0.3
                lastBytes = bytes
                lastTime = now
                report(speed: speed)
            }
        }
        defer { poller.cancel() }

        try await withTaskCancellationHandler {
            try await withThrowingTaskGroup(of: (Int, URL).self) { group in
                var nextIndex = 0
                func addNextDownload() {
                    guard !cancellation.isCancelled else { return }
                    guard nextIndex < pendingFiles.count else { return }
                    let (index, file) = pendingFiles[nextIndex]
                    nextIndex += 1
                    group.addTask {
                        let fileManager = FileManager.default
                        let destination = directory.appendingPathComponent(file.name)
                        try fileManager.createDirectory(at: destination.deletingLastPathComponent(), withIntermediateDirectories: true)
                        try? fileManager.removeItem(at: destination)
                        return (index, try await Self.downloadFile(file.url, cancellation: cancellation))
                    }
                }
                for _ in 0..<min(4, pendingFiles.count) {
                    addNextDownload()
                }
                for try await (index, temporaryFile) in group {
                    try Task.checkCancellation()
                    if cancellation.isCancelled { throw ModelError.cancelled }
                    let destination = directory.appendingPathComponent(files[index].name)
                    try FileManager.default.createDirectory(at: destination.deletingLastPathComponent(), withIntermediateDirectories: true)
                    try FileManager.default.moveItem(at: temporaryFile, to: destination)
                    counters.completedFiles += 1
                    counters.finishedBytes += files[index].sizeBytes ?? Int64((try? destination.resourceValues(forKeys: [.fileSizeKey]).fileSize) ?? 0)
                    addNextDownload()
                }
            }
        } onCancel: {
            cancellation.cancel()
        }
        report(speed: 0)
    }

    // MARK: - Hugging Face search

    /// Searches every MLX model on Hugging Face. Runs after a short pause so each
    /// keystroke doesn't send a request; an empty query clears the results.
    func search(_ query: String, format: ModelFormat = .mlx) {
        searchTask?.cancel()
        detailsTask?.cancel()
        let trimmed = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else {
            searchResults = []
            searchError = nil
            isSearching = false
            return
        }
        isSearching = true
        searchTask = Task { @MainActor [weak self] in
            try? await Task.sleep(for: .milliseconds(350))
            guard let self, !Task.isCancelled else { return }
            async let supported: Void = self.loadSupportedModelTypes()
            do {
                let entries = try await Self.fetchSearchEntries(trimmed, format: format)
                guard !Task.isCancelled else { return }
                self.searchResults = entries.compactMap { Self.searchResult($0, format: format) }
                self.searchError = nil
                self.loadDetails(for: self.searchResults.map(\.repository))
            } catch {
                guard !Task.isCancelled else { return }
                self.searchResults = []
                self.searchError = "Could not search Hugging Face. Check your internet connection."
            }
            _ = await supported
            self.isSearching = false
        }
    }

    private static func fetchSearchEntries(_ query: String, format: ModelFormat) async throws -> [SearchEntry] {
        var components = URLComponents(string: "https://huggingface.co/api/models")!
        components.queryItems = [
            URLQueryItem(name: "search", value: query),
            URLQueryItem(name: "filter", value: format.rawValue),
            URLQueryItem(name: "sort", value: "downloads"),
            URLQueryItem(name: "direction", value: "-1"),
            URLQueryItem(name: "limit", value: "40"),
        ] + ["gated", "downloads", "likes", "pipeline_tag", "tags", "private", "disabled"].map {
            URLQueryItem(name: "expand[]", value: $0)
        }
        let (data, response) = try await URLSession.shared.data(from: components.url!)
        guard (response as? HTTPURLResponse)?.statusCode == 200 else { throw ModelError.catalogUnavailable }
        return try JSONDecoder().decode([SearchEntry].self, from: data)
    }

    /// Pipelines that can hold a chat. Embedding, speech and image models are left out.
    nonisolated private static let chatPipelines: Set<String> = ["text-generation", "image-text-to-text", "any-to-any"]

    nonisolated static func searchResult(_ entry: SearchEntry, format: ModelFormat = .mlx) -> CatalogModel? {
        guard entry.private != true, entry.disabled != true, !entry.isGated,
              !LocalModelCatalogRules.isDraftRepository(entry.id),
              entry.pipelineTag.map({ chatPipelines.contains($0) }) ?? true else { return nil }
        let parts = entry.id.split(separator: "/", maxSplits: 1).map(String.init)
        let repositoryName = parts.last ?? entry.id
        let family = LocalModelCatalogRules.family(forRepository: entry.id) ?? .other
        let isOfficial = family != .other && LocalModelCatalogRules.isOfficialRelease(entry.id, tags: entry.tags ?? [])
        return CatalogModel(
            id: repositoryName,
            repository: entry.id,
            family: family,
            displayName: repositoryName.replacingOccurrences(of: "-", with: " "),
            summary: "",
            minimumMemoryGB: 0,
            recommendedMemoryGB: 0,
            approximateSizeGB: 0,
            supportsVision: entry.pipelineTag == "image-text-to-text",
            supportsToolCalls: false,
            supportsReasoningMode: false,
            supportsSpeculativeDecoding: false,
            quantization: quantizationLabel(for: entry.id),
            downloadSizeBytes: nil,
            isRemote: true,
            isOfficial: isOfficial,
            isSearchResult: true,
            format: format,
            author: parts.count == 2 ? parts[0] : nil,
            downloads: entry.downloads,
            likes: entry.likes
        )
    }

    /// Fetches size and architecture for the given repositories, six at a time.
    private func loadDetails(for repositories: [String]) {
        let pending = repositories.filter { remoteDetails[$0] == nil }
        guard !pending.isEmpty else { return }
        detailsTask = Task { @MainActor [weak self] in
            await withTaskGroup(of: (String, RemoteModelDetails?).self) { group in
                var next = 0
                func addNext() {
                    guard next < pending.count else { return }
                    let repository = pending[next]
                    next += 1
                    group.addTask { (repository, try? await Self.fetchDetails(repository)) }
                }
                for _ in 0..<min(6, pending.count) { addNext() }
                for await (repository, details) in group {
                    guard !Task.isCancelled else { group.cancelAll(); return }
                    if let details { self?.remoteDetails[repository] = details }
                    addNext()
                }
            }
        }
    }

    private static func fetchDetails(_ repository: String) async throws -> RemoteModelDetails {
        var components = URLComponents(string: "https://huggingface.co/api/models/\(repository)/tree/main")!
        components.queryItems = [URLQueryItem(name: "recursive", value: "true")]
        let (data, response) = try await URLSession.shared.data(from: components.url!)
        guard (response as? HTTPURLResponse)?.statusCode == 200 else { throw ModelError.catalogUnavailable }
        let files = try JSONDecoder().decode([RemoteTreeFile].self, from: data)
            .filter { $0.type == "file" && !$0.path.hasPrefix(".") && $0.path != "README.md" }
        let size = files.reduce(Int64(0)) { $0 + ($1.size ?? $1.lfs?.size ?? 0) }
        let hasSafetensors = files.contains { $0.path.hasSuffix(".safetensors") }
        let gguf = GGUFOption.options(from: files.map { RemoteFileInfo(path: $0.path, sizeBytes: $0.size ?? $0.lfs?.size ?? 0) })
        if !gguf.options.isEmpty && !hasSafetensors {
            return RemoteModelDetails(sizeBytes: nil, modelType: nil, hasSafetensors: false,
                                      ggufOptions: gguf.options, projector: gguf.projector)
        }

        var modelType: String?
        if files.contains(where: { $0.path == "config.json" }),
           let url = URL(string: "https://huggingface.co/\(repository)/resolve/main/config.json"),
           let (configData, configResponse) = try? await URLSession.shared.data(from: url),
           (configResponse as? HTTPURLResponse)?.statusCode == 200,
           let config = try? JSONSerialization.jsonObject(with: configData) as? [String: Any] {
            modelType = ((config["model_type"] ?? config["speculators_model_type"]) as? String)?.lowercased()
        }
        return RemoteModelDetails(sizeBytes: size > 0 ? size : nil, modelType: modelType, hasSafetensors: hasSafetensors)
    }

    /// Files for a search result, after checking that the runtime can load its
    /// architecture and that it has a chat template (base models don't).
    private func searchResultFiles(for repository: String) async throws -> [RepositoryFile] {
        let endpoint = URL(string: "https://huggingface.co/api/models/\(repository)")!
        let (data, response) = try await URLSession.shared.data(from: endpoint)
        guard (response as? HTTPURLResponse)?.statusCode == 200 else { throw ModelError.catalogUnavailable }
        let detail = try JSONDecoder().decode(RemoteModelDetail.self, from: data)
        let revision = detail.sha ?? "main"
        let names = Set((detail.siblings ?? []).map(\.rfilename))
        guard names.contains("config.json"), names.contains(where: { $0.hasSuffix(".safetensors") }) else {
            throw ModelError.notMLX
        }

        let configURL = URL(string: "https://huggingface.co/\(repository)/resolve/\(revision)/config.json")!
        let (configData, _) = try await URLSession.shared.data(from: configURL)
        let config = try? JSONSerialization.jsonObject(with: configData) as? [String: Any]
        let modelType = ((config?["model_type"] ?? config?["speculators_model_type"]) as? String)?.lowercased()
        if let supportedModelTypes {
            guard let modelType, supportedModelTypes.contains(modelType) else {
                throw ModelError.unsupportedArchitecture(modelType ?? "unknown")
            }
        }

        var hasChatTemplate = names.contains("chat_template.jinja") || names.contains("chat_template.json")
        if !hasChatTemplate, names.contains("tokenizer_config.json") {
            let tokenizerURL = URL(string: "https://huggingface.co/\(repository)/resolve/\(revision)/tokenizer_config.json")!
            if let (tokenizerData, _) = try? await URLSession.shared.data(from: tokenizerURL),
               let tokenizer = try? JSONSerialization.jsonObject(with: tokenizerData) as? [String: Any] {
                hasChatTemplate = tokenizer["chat_template"] != nil
            }
        }
        guard hasChatTemplate else { throw ModelError.noChatTemplate }

        return Self.repositoryFiles(from: detail, repository: repository, revision: revision)
    }

    private static func repositoryFiles(from detail: RemoteModelDetail, repository: String, revision: String) -> [RepositoryFile] {
        (detail.siblings ?? []).compactMap { entry in
            let path = entry.rfilename
            guard !path.hasPrefix("."), path != "README.md" else { return nil }
            return RepositoryFile(
                name: path,
                url: URL(string: "https://huggingface.co/\(repository)/resolve/\(revision)/\(path)?download=true")!,
                sizeBytes: entry.size ?? entry.lfs?.size
            )
        }
    }

    private func fetchRemoteCatalog() async throws -> [CatalogModel] {
        let searchTerms = ["gemma-4", "qwen3.8"]
        var entriesByRepository: [String: RemoteModelEntry] = [:]

        for searchTerm in searchTerms {
            let entries = try await fetchRemoteCatalogEntries(searchTerm: searchTerm)
            for entry in entries {
                entriesByRepository[entry.id] = entry
            }
        }

        return entriesByRepository.values.compactMap { entry in
            let normalizedID = entry.id.lowercased()
            guard let family = LocalModelCatalogRules.family(forRepository: normalizedID),
                  LocalModelCatalogRules.isCurrentGeneration(entry.id),
                  !LocalModelCatalogRules.isDraftRepository(entry.id),
                  LocalModelCatalogRules.isInstructionTunedRepository(entry.id),
                  !entry.private,
                  entry.disabled != true,
                  entry.gated != true else { return nil }

            let files = entry.siblings ?? []
            let size = files.reduce(Int64(0)) { total, file in total + (file.size ?? file.lfs?.size ?? 0) }
            let repositoryName = entry.id.split(separator: "/").last.map(String.init) ?? entry.id
            let displayName = prettyModelName(from: repositoryName)
            let supportsVision = entry.tags?.contains(where: { $0.localizedCaseInsensitiveContains("image-text-to-text") }) == true
                || normalizedID.contains("-vl")
            let isOfficial = LocalModelCatalogRules.isOfficialRelease(entry.id, tags: entry.tags ?? [])
            let baseModel = entry.tags?.first(where: { $0.hasPrefix("base_model:") })?.split(separator: ":").last.map(String.init)
            let summary = isOfficial
                ? "Official \(family.displayName) release, converted to MLX."
                : "Community version\(baseModel.map { " based on \($0)" } ?? ""). Not an official \(family.displayName) release."
            return CatalogModel(
                id: repositoryName,
                repository: entry.id,
                family: family,
                displayName: displayName,
                summary: summary,
                minimumMemoryGB: estimatedMemoryGB(sizeBytes: size),
                recommendedMemoryGB: estimatedMemoryGB(sizeBytes: size) + 4,
                approximateSizeGB: Double(size) / 1_073_741_824,
                supportsVision: supportsVision,
                supportsToolCalls: true,
                supportsReasoningMode: family == .qwen,
                supportsSpeculativeDecoding: false,
                quantization: Self.quantizationLabel(for: entry.id),
                downloadSizeBytes: size > 0 ? size : nil,
                isRemote: true,
                isOfficial: isOfficial
            )
        }
    }

    }

    private func prettyModelName(from repositoryName: String) -> String {

            .replacingOccurrences(of: "-MLX", with: " MLX")
            .replacingOccurrences(of: "-", with: " ")
    }

    private func loadMissingRepositorySizes() async {
        repositorySizeTask?.cancel()
        let repositories = catalogModels
            .filter { $0.isRemote && repositorySizes[$0.repository] == nil }
            .map(\.repository)
        guard !repositories.isEmpty else { return }

        repositorySizeTask = Task { @MainActor [weak self] in
            await withTaskGroup(of: (String, Int64?).self) { group in
                var nextIndex = 0
                for _ in 0..<min(6, repositories.count) {
                    guard nextIndex < repositories.count else { break }
                    let repository = repositories[nextIndex]
                    nextIndex += 1
                    group.addTask {
                        (repository, try? await Self.fetchRepositorySize(repository))
                    }
                }
                for await (repository, size) in group {
                    if let size, size > 0 {
                        self?.repositorySizes[repository] = size
                    }
                    guard nextIndex < repositories.count else { continue }
                    let nextRepository = repositories[nextIndex]
                    nextIndex += 1
                    group.addTask {
                        (nextRepository, try? await Self.fetchRepositorySize(nextRepository))
                    }
                }
            }
        }
        await repositorySizeTask?.value
        repositorySizeTask = nil
    }

    private static func fetchRepositorySize(_ repository: String) async throws -> Int64 {
        var components = URLComponents(string: "https://huggingface.co/api/models/\(repository)/tree/main")!
        components.queryItems = [URLQueryItem(name: "recursive", value: "true")]
        let (data, response) = try await URLSession.shared.data(from: components.url!)
        guard (response as? HTTPURLResponse)?.statusCode == 200 else { throw ModelError.catalogUnavailable }
        let files = try JSONDecoder().decode([RemoteTreeFile].self, from: data)
        return files.reduce(Int64(0)) { total, file in
            guard file.type == "file", !file.path.hasPrefix("."), file.path != "README.md" else { return total }
            return total + (file.size ?? file.lfs?.size ?? 0)
        }
    }

    private func mergeCatalog(_ remoteModels: [CatalogModel]) -> [CatalogModel] {
        var merged = ModelCatalog.featuredModels
        let knownRepositories = Set(merged.map(\.repository))
        merged.append(contentsOf: remoteModels.filter { !knownRepositories.contains($0.repository) })
        return merged.sorted { lhs, rhs in
            if lhs.isRemote != rhs.isRemote { return !lhs.isRemote }
            if lhs.isOfficial != rhs.isOfficial { return lhs.isOfficial }
            return lhs.displayName.localizedStandardCompare(rhs.displayName) == .orderedAscending
        }
    }

    nonisolated static func quantizationLabel(for modelID: String) -> String {
        let name = modelID.lowercased()
        for bits in ["2bit", "3bit", "4bit", "5bit", "6bit", "8bit", "bf16", "fp16"] {
            if name.contains(bits) {
                return bits == "bf16" || bits == "fp16" ? bits.uppercased() : bits.replacingOccurrences(of: "bit", with: "-bit")
            }
        }
        // GGUF names end in llama.cpp's quantization type: Q4_K_M, Q8_0, IQ3_XS, F16...
        let ggufPattern = #"(?:^|[-_.])((?:i?q[1-8](?:_[a-z0-9]+)*)|bf16|f16|f32)(?:\.gguf)?$"#
        if let match = name.range(of: ggufPattern, options: .regularExpression) {
            return String(name[match])
                .replacingOccurrences(of: ".gguf", with: "")
                .trimmingCharacters(in: CharacterSet(charactersIn: "-_."))
                .uppercased()
        }
        return "Unknown"
    }


    private func estimatedMemoryGB(sizeBytes: Int64) -> Int {
        max(1, Int(ceil(Double(sizeBytes) / 1_073_741_824 * 1.15)))
    }

    private static func downloadFile(_ url: URL, cancellation: DownloadCancellation) async throws -> URL {
        var lastError: Error?
        for attempt in 0..<3 {
            do {
                if cancellation.isCancelled { throw ModelError.cancelled }
                return try await downloadFileOnce(url, cancellation: cancellation)
            } catch {
                if cancellation.isCancelled { throw ModelError.cancelled }
                lastError = error
                if attempt < 2 { try await Task.sleep(for: .seconds(1)) }
            }
        }
        throw lastError ?? ModelError.catalogUnavailable
    }

    /// A plain download task rather than the async API, so its bytes received can be
    /// read for progress and it can be cancelled mid-file.
    private static func downloadFileOnce(_ url: URL, cancellation: DownloadCancellation) async throws -> URL {
        try await withCheckedThrowingContinuation { continuation in
            let task = downloadSession.downloadTask(with: url) { temporaryURL, response, error in
                if let error {
                    continuation.resume(throwing: error)
                    return
                }
                guard let temporaryURL,
                      let httpResponse = response as? HTTPURLResponse,
                      (200..<300).contains(httpResponse.statusCode) else {
                    let status = (response as? HTTPURLResponse)?.statusCode ?? 0
                    continuation.resume(throwing: ModelError.httpFailure(status: status, resource: url.lastPathComponent))
                    return
                }
                // The system deletes the file when this handler returns, so move it now.
                let kept = FileManager.default.temporaryDirectory.appendingPathComponent("vireo-\(UUID().uuidString)")
                do {
                    try FileManager.default.moveItem(at: temporaryURL, to: kept)
                    continuation.resume(returning: kept)
                } catch {
                    continuation.resume(throwing: error)
                }
            }
            guard cancellation.track(task) else {
                continuation.resume(throwing: ModelError.cancelled)
                return
            }
            task.resume()
        }
    }

    nonisolated static func directorySize(at url: URL) -> Int64 {
        guard let enumerator = FileManager.default.enumerator(at: url, includingPropertiesForKeys: [.fileSizeKey]) else { return 0 }
        return enumerator.reduce(Int64(0)) { size, entry in
            guard let file = entry as? URL else { return size }
            return size + Int64((try? file.resourceValues(forKeys: [.fileSizeKey]).fileSize) ?? 0)
        }
    }
}

/// Size and architecture of a Hugging Face model, fetched after it appears in a search.
struct RemoteModelDetails: Equatable {
    let sizeBytes: Int64?
    let modelType: String?
    let hasSafetensors: Bool
    /// For GGUF repositories: the versions (quantizations) to choose from.
    var ggufOptions: [GGUFOption] = []
    /// A vision projector ("mmproj") to download alongside, when the repository has one.
    var projector: RemoteFileInfo? = nil
}

struct RemoteFileInfo: Equatable, Hashable {
    let path: String
    let sizeBytes: Int64
}

/// One downloadable version of a GGUF model: a single file, or every part of a split one.
struct GGUFOption: Identifiable, Equatable, Hashable {
    let parts: [RemoteFileInfo]
    /// The file llama.cpp is pointed at: the only one, or the first part.
    var path: String { parts[0].path }
    var id: String { path }
    var sizeBytes: Int64 { parts.reduce(0) { $0 + $1.sizeBytes } }
    var quantization: String {
        // "Model-Q8_0-00001-of-00002.gguf" -> "Model-Q8_0.gguf"
        let name = ((path as NSString).lastPathComponent)
            .replacingOccurrences(of: #"-\d{5}-of-\d{5}(?=\.gguf$)"#, with: "", options: [.regularExpression, .caseInsensitive])
        return ModelManager.quantizationLabel(for: name)
    }

    /// Groups a repository's files into versions, and picks its vision projector
    /// (the smallest, which is usually F16 rather than F32).
    static func options(from files: [RemoteFileInfo]) -> (options: [GGUFOption], projector: RemoteFileInfo?) {
        let ggufFiles = files.filter { $0.path.lowercased().hasSuffix(".gguf") }
        let projector = ggufFiles
            .filter { ($0.path as NSString).lastPathComponent.lowercased().contains("mmproj") }
            .min { $0.sizeBytes < $1.sizeBytes }
        let partPattern = #"-(\d{5})-of-(\d{5})\.gguf$"#
        var options: [GGUFOption] = []
        for file in ggufFiles where !(file.path as NSString).lastPathComponent.lowercased().contains("mmproj") {
            let lowercased = file.path.lowercased()
            guard let part = lowercased.range(of: partPattern, options: .regularExpression) else {
                options.append(GGUFOption(parts: [file]))
                continue
            }
            guard lowercased[part].hasPrefix("-00001-") else { continue }
            let stem = String(file.path[..<part.lowerBound])
            let parts = ggufFiles
                .filter { $0.path.hasPrefix(stem + "-") && $0.path.lowercased().range(of: partPattern, options: .regularExpression) != nil }
                .sorted { $0.path < $1.path }
            options.append(GGUFOption(parts: parts))
        }
        return (options.sorted { $0.sizeBytes < $1.sizeBytes }, projector)
    }

    /// The largest version that fits comfortably, or else the smallest.
    static func recommended(from options: [GGUFOption], budgetBytes: Int64) -> GGUFOption? {
        options.last { MemoryFit.assess(weightBytes: $0.sizeBytes, budgetBytes: budgetBytes) == .fits } ?? options.first
    }
}

/// Tracks the running download tasks: Cancel stops files already in flight, and
/// progress reads their bytes received.
private final class DownloadCancellation: @unchecked Sendable {
    private let lock = NSLock()
    private var cancelled = false
    private var tasks: [Int: URLSessionTask] = [:]

    var isCancelled: Bool {
        lock.lock()
        defer { lock.unlock() }
        return cancelled
    }

    /// Bytes received by tasks still running.
    var inFlightBytes: Int64 {
        lock.lock()
        defer { lock.unlock() }
        return tasks.values.reduce(Int64(0)) { total, task in
            task.state == .running ? total + task.countOfBytesReceived : total
        }
    }

    /// Returns false, and cancels the task, if the download was already cancelled.
    func track(_ task: URLSessionTask) -> Bool {
        lock.lock()
        defer { lock.unlock() }
        guard !cancelled else {
            task.cancel()
            return false
        }
        tasks = tasks.filter { $0.value.state == .running || $0.value.state == .suspended }
        tasks[task.taskIdentifier] = task
        return true
    }

    func cancel() {
        lock.lock()
        cancelled = true
        let running = Array(tasks.values)
        tasks = [:]
        lock.unlock()
        running.forEach { $0.cancel() }
    }
}

    struct StartupPreflight {
        struct Result {
            let errors: [String]
            let warnings: [String]

            var isReady: Bool { errors.isEmpty }
        }

        static func evaluate(configuration: LocalServerConfiguration) -> Result {
            var errors: [String] = []
            var warnings: [String] = []

            let modelPath = configuration.activeModelPath
            let modelURL = URL(fileURLWithPath: modelPath)
            if modelPath.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
                errors.append("Choose an installed model before starting the local server.")
            } else if configuration.inferenceRuntime == .mlx && !FileManager.default.fileExists(atPath: modelURL.appendingPathComponent("config.json").path) {
                errors.append("The selected model is unavailable. Open Model Manager and choose an installed model.")
            }

            // A GGUF file that exists runs in the bundled llama.cpp. Otherwise llama.cpp
            // mode expects a llama-server the user started, which may name its model differently.
            if configuration.inferenceRuntime == .llamaCpp, LocalServerManager.isGGUF(modelPath),
               FileManager.default.fileExists(atPath: modelPath), LocalServerManager.llamaServerURL == nil {
                warnings.append("This build doesn't include llama.cpp, so start llama-server yourself on port \(configuration.mlxPort) to use this GGUF model.")
            } else if configuration.inferenceRuntime == .llamaCpp,
               !FileManager.default.fileExists(atPath: modelPath) {
                warnings.append("The selected llama.cpp model path does not exist yet. Ensure your llama.cpp server can resolve model '\(configuration.activeModelName.isEmpty ? modelPath : configuration.activeModelName)'.")
            }

            let serverScript = URL(fileURLWithPath: configuration.serverDirectory).appendingPathComponent("api_server.py")
            if !FileManager.default.fileExists(atPath: serverScript.path) {
                errors.append("The API server runtime was not found at the configured server directory.")
            }

            if configuration.mlxPort == configuration.apiPort {
                warnings.append("MLX and API ports are the same. Use different ports to avoid startup collisions.")
            }

            if configuration.serverMode && configuration.apiPort == 3000 {
                warnings.append("Public network access is enabled on the default API port. Validate firewall and port-forwarding settings before sharing access.")
            }

            return Result(errors: errors, warnings: warnings)
        }
    }

                initial.inferenceRuntime = .mlx
                initial.runtimeModelPath = legacyModelPath
                initial.runtimeModelID = URL(fileURLWithPath: legacyModelPath).lastPathComponent
            }
            configuration = initial
        }
        let diagnostics = Self.evaluateAppleFoundationRuntimeDiagnostics()
        appleFoundationDiagnostics = diagnostics
        appleFoundationAvailability = diagnostics.availability
        configuration = Self.normalizedConfiguration(
            configuration,
            appleFoundationAvailability: appleFoundationAvailability
        )
        persistConfiguration(configuration)
        migrateDraftModelsDirectoryIfNeeded()
        prepareBundledServerDirectory()
        Task { @MainActor [weak self] in
            await self?.terminateOrphanedServers()
            await self?.ensureDependenciesOnStartup()
        }
    }

    var overallStatus: LocalServerStatus {
        let statuses = [mlx.status, api.status]
        if statuses.contains(.error) { return .error }
        if statuses.allSatisfy({ $0 == .running }) { return .running }
        if statuses.contains(.starting) || statuses.contains(.stopping) { return .starting }
        return .stopped
    }

    var isRunning: Bool {
        overallStatus == .running
    }

    /// Saved settings differ from the ones the running servers started with.
    var needsRestart: Bool {
        guard overallStatus == .running, let startedConfiguration else { return false }
        return startedConfiguration != configuration
    }

    var hasUsableModel: Bool {
        let path = configuration.activeModelPath
        guard !path.isEmpty else { return false }
        guard configuration.inferenceRuntime == .mlx else { return true }
        return FileManager.default.fileExists(atPath: URL(fileURLWithPath: path).appendingPathComponent("config.json").path)
    }

    var speculativeDecodingStatus: String {
        guard configuration.speculativeDecoding else { return "Speculative decoding is disabled" }
        guard draftModelPath != nil else { return "No matching draft model found; using standard decoding" }
        return "Speculative decoding ready"
    }

    var inferenceServiceLabel: String {
        switch configuration.inferenceRuntime {
        case .mlx:
            return "MLX Inference"
        case .llamaCpp:
            return "llama.cpp Inference"
        case .appleFoundation:
            return "Apple Foundation Inference"
        }
    }

    var inferenceStatusDetail: String {
        switch configuration.inferenceRuntime {
        case .mlx:
            return "Managed by this app on port \(configuration.mlxPort)."
        case .llamaCpp where managesLlamaServer:
            return "llama.cpp \(Self.bundledLlamaVersion.map { "(\($0)) " } ?? "")managed by this app on port \(configuration.mlxPort)."
        case .llamaCpp:

            case .running:
                return "Connected to external llama-server on port \(configuration.mlxPort)."
            case .starting:
                return "Checking external llama-server health on port \(configuration.mlxPort)..."
            case .error:
                return Self.llamaModelProblem(for: configuration)
                    ?? "Could not reach external llama-server on port \(configuration.mlxPort)."
            case .stopping, .stopped:
                return Self.llamaModelProblem(for: configuration)
                    ?? "Waiting for external llama-server on port \(configuration.mlxPort)."
            }
        case .appleFoundation:
            if appleFoundationAvailability.isAvailable {
                return "Apple Foundation runtime readiness checks passed."
            }
            return appleFoundationAvailability.unavailableReason
        }
    }

    func startAll() {
        Task { @MainActor in
            startupMessage = ""
            refreshAppleFoundationAvailability()
            normalizeRuntimeSelectionIfNeeded(
                warningPrefix: "Apple Foundation runtime is unavailable on this host"
            )
            guard validateStartup() else { return }
            startedConfiguration = configuration
            // Settings saved before the URL followed the port may still disagree.
            ServerConfig.baseURL = ServerConfig.url(ServerConfig.baseURL, followingAPIPort: configuration.apiPort)
            guard await ensureDependencies() else {
                // Said where Start was pressed, not only in Settings: otherwise Start seems to do nothing.
                startupMessage = dependencySetupStatus
                setStatus(.error, for: "api")
                return
            }
            await terminateOrphanedServers()
            await startOrAdoptInferenceRuntime()
            await startOrAdoptAPI()
        }
    }

    func stopAll() {
        // Stops whichever inference server this app started (MLX or llama.cpp);
        // an external one is only marked stopped.
        stop(service: "mlx")
        stop(service: "api")
    }

    /// Stops this app's servers and waits for them to exit, killing any still running
    /// after `timeout`. Used on quit, so an update's relaunch doesn't find the old
    /// server still holding its port and the model's memory.
    func stopAllAndWait(timeout: TimeInterval = 5) async {
        stopAll()
        let deadline = Date().addingTimeInterval(timeout)
        while processes.values.contains(where: \.isRunning), Date() < deadline {
            try? await Task.sleep(for: .milliseconds(100))
        }
        for process in processes.values where process.isRunning {
            kill(process.processIdentifier, SIGKILL)
        }
    }

    func restartAll() {
        stopAll()
        Task { @MainActor in
            // Start is skipped while a service is still stopping, so wait for the old processes to exit.
            let deadline = Date().addingTimeInterval(10)
            while (mlx.status == .stopping || api.status == .stopping), Date() < deadline {
                try? await Task.sleep(for: .milliseconds(200))
            }
            startAll()
        }
    }

    /// Versions of Python and the packages the local server runs on, read from
    /// the Python this app uses (bundled in releases, the server's venv otherwise).
    func runtimeVersions() async -> [String: String] {
        guard let managedPython = managedPythonURL else { return [:] }
        let script = """
        import importlib.metadata as metadata, json, platform
        versions = {'python': platform.python_version()}
        for name in ('mlx', 'mlx-vlm', 'mlx-lm', 'fastapi', 'onnxruntime', 'mcp'):
            try:
                versions[name] = metadata.version(name)
            except metadata.PackageNotFoundError:
                pass
        print(json.dumps(versions))
        """
        guard let result = await Self.runProcess(managedPython, arguments: ["-c", script], environment: pythonEnvironment),
              result.status == 0,
              let line = result.output.split(whereSeparator: \.isNewline).last,
              let versions = try? JSONDecoder().decode([String: String].self, from: Data(line.utf8)) else { return [:] }
        return versions
    }

    private var cachedSupportedModelTypes: Set<String>?

    /// The `model_type` values the bundled MLX runtime can load: its model packages
    /// plus the aliases it maps onto them. Nil when the runtime isn't available, so
    /// callers can skip the check rather than reject everything.
    func supportedModelTypes() async -> Set<String>? {
        if let cachedSupportedModelTypes { return cachedSupportedModelTypes }
        guard let managedPython = managedPythonURL else { return nil }
        let script = """
        import json, pkgutil
        import mlx_vlm.models as models
        from mlx_vlm.utils import MODEL_REMAPPING
        names = [name for _, name, is_package in pkgutil.iter_modules(models.__path__) if is_package]
        print(json.dumps(names + list(MODEL_REMAPPING)))
        """
        guard let result = await Self.runProcess(managedPython, arguments: ["-c", script], environment: pythonEnvironment),
              result.status == 0,
              let data = result.output.split(separator: "\n").last.map({ Data($0.utf8) }),
              let names = try? JSONDecoder().decode([String].self, from: data),
              !names.isEmpty else { return nil }
        let types = Set(names.map { $0.lowercased() })
        cachedSupportedModelTypes = types
        return types
    }

    /// Runs a short-lived process off the main actor and returns its exit status and stdout.
    /// Importing the Python stack takes seconds, which froze the UI when done inline.
    private nonisolated static func runProcess(
        _ executable: URL, arguments: [String], environment: [String: String] = [:]
    ) async -> (status: Int32, output: String)? {
        await Task.detached(priority: .utility) { () -> (status: Int32, output: String)? in
            let process = Process()
            let output = Pipe()
            process.executableURL = executable
            process.arguments = arguments
            process.environment = ProcessInfo.processInfo.environment.merging(environment, uniquingKeysWith: { _, new in new })
            process.standardOutput = output
            process.standardError = FileHandle.nullDevice
            do {
                try process.run()
            } catch {
                return nil
            }
            // Read to EOF before waiting so a chatty process can't fill the pipe and block.
            let data = output.fileHandleForReading.readDataToEndOfFile()
            process.waitUntilExit()
            return (process.terminationStatus, String(decoding: data, as: UTF8.self))
        }.value
    }

    @discardableResult
    func saveConfiguration(_ newConfiguration: LocalServerConfiguration) -> LocalServerConfiguration {
        refreshAppleFoundationAvailability()
        var normalized = newConfiguration
        normalized = Self.normalizedConfiguration(
            normalized,
            appleFoundationAvailability: appleFoundationAvailability
        )
        if normalized.inferenceRuntime == .mlx {
            let legacyPath = normalized.mlxModel.trimmingCharacters(in: .whitespacesAndNewlines)
            normalized.runtimeModelPath = legacyPath
            normalized.runtimeModelID = URL(fileURLWithPath: legacyPath).lastPathComponent
        } else if normalized.mlxModel.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
            normalized.mlxModel = normalized.runtimeModelPath
        }

        configuration = normalized
        persistConfiguration(normalized)
        return normalized
    }

    func useModel(at path: String) {
        var updated = configuration
        updated.mlxModel = path
        updated.inferenceRuntime = Self.isGGUF(path) ? .llamaCpp : .mlx
        updated.runtimeModelPath = path
        updated.runtimeModelID = URL(fileURLWithPath: path).lastPathComponent
        updated.modelsDirectory = URL(fileURLWithPath: path).deletingLastPathComponent().path
        saveConfiguration(updated)
    }

    /// Makes `path` the active model and loads it. The inference server holds one
    /// model at a time (and its draft model is chosen at launch), so a running
    /// server is restarted; a stopped one is started.
    func switchModel(to path: String) {
        guard path != configuration.activeModelPath || !isRunning else { return }
        useModel(at: path)
        if overallStatus == .stopped {
            startAll()
        } else {
            restartAll()
        }
    }

    func availableModels() -> [LocalModel] {
        let directory = URL(fileURLWithPath: configuration.modelsDirectory)
        guard let entries = try? FileManager.default.contentsOfDirectory(
            at: directory,
            includingPropertiesForKeys: [.isDirectoryKey],
            options: [.skipsHiddenFiles]
        ) else { return [] }

        return entries.compactMap { entry in
            guard (try? entry.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) == true,
                  FileManager.default.fileExists(atPath: entry.appendingPathComponent("config.json").path) else { return nil }
            return LocalModel(path: entry.path, name: entry.lastPathComponent, sizeBytes: directorySize(at: entry))
        }
        .sorted { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }

    private func startMLX() {
        guard canStart(mlx.status) else { return }
        setStatus(.starting, for: "mlx")

        // Only the API server talks to MLX, so it stays on loopback even in network mode.
        // mlx_server.py runs mlx_vlm.server with Vireo's fixes to it (Gemma 4's thinking).
        var arguments = Self.mlxLauncherArguments(serverDirectory: configuration.serverDirectory)
            + ["--model", modelPath, "--port", "\(configuration.mlxPort)", "--host", "127.0.0.1"]
        if configuration.speculativeDecoding, let draftModel {
            arguments += ["--draft-model", draftModel.path, "--draft-kind", draftModel.kind]
        }
        let checkpoints = Self.prefixCacheCheckpoints(physicalMemory: ProcessInfo.processInfo.physicalMemory)
        startProcess(service: "mlx", arguments: arguments, environment: [
            "APC_ENABLED": configuration.apcEnabled ? "1" : "0",
            "APC_NUM_BLOCKS": "\(configuration.apcNumBlocks)",
            "APC_CHECKPOINT_INTERVAL_TOKENS": "\(checkpoints.intervalTokens)",
            "APC_CHECKPOINT_ENTRIES": "\(checkpoints.entries)",
            "MLX_VLM_MAX_TOKENS": "\(configuration.maxOutputTokens)",
        ])
        pollHealth(service: "mlx", url: URL(string: "http://localhost:\(configuration.mlxPort)/health")!, timeout: 300)
    }

    /// How the MLX server is started: through server/mlx_server.py, which applies Vireo's
    /// fixes to mlx_vlm and then runs it, or the module itself when that file is missing
    /// (a server folder from an older build).
    nonisolated static func mlxLauncherArguments(serverDirectory: String) -> [String] {
        let launcher = URL(fileURLWithPath: serverDirectory).appendingPathComponent("mlx_server.py")
        return FileManager.default.fileExists(atPath: launcher.path) ? [launcher.path] : ["-m", "mlx_vlm.server"]
    }

    /// How often mlx_vlm snapshots a prompt for its prefix cache, and how many snapshots it
    /// keeps. Each agent step re-sends the conversation plus a little more, and the runtime
    /// can only resume from the last snapshot at or before where the prompts differ. Its
    /// default (every 2,048 tokens, 2 kept) re-read up to ~2k tokens on every step; every
    /// 256 tokens a follow-up step went from 2.1 s to 0.07 s of prompt reading on an M5 Max
    /// with Gemma 4 31B (2026-11-09). Snapshots cost memory (the cache's own cap still
    /// applies), so smaller Macs snapshot less often.
    nonisolated static func prefixCacheCheckpoints(physicalMemory: UInt64) -> (intervalTokens: Int, entries: Int) {
        let gigabytes = physicalMemory / 1_073_741_824
        if gigabytes >= 64 { return (256, 8) }
        if gigabytes >= 32 { return (512, 6) }
        return (1024, 4)
    }

    // MARK: - llama.cpp (GGUF models)

    nonisolated static func isGGUF(_ path: String) -> Bool {
        path.lowercased().hasSuffix(".gguf")
    }

    /// Why llama.cpp can't start its own server for the selected model, when it
    /// could if a GGUF model were chosen. Nil when llama.cpp isn't available,
    /// since then an external llama-server is the only option.
    nonisolated static func llamaModelProblem(
        for configuration: LocalServerConfiguration, llamaAvailable: Bool = llamaServerURL != nil
    ) -> String? {
        guard configuration.inferenceRuntime == .llamaCpp, llamaAvailable else { return nil }
        let path = configuration.activeModelPath
        if !isGGUF(path) {
            return "llama.cpp runs GGUF models, and the selected model isn't one. Choose a GGUF model, or switch back to MLX for this model."
        }
        if !FileManager.default.fileExists(atPath: path) {
            return "The selected GGUF model wasn't found at \(path). Choose another model, or reconnect the drive it's on."
        }
        return nil
    }

    /// Whether this app starts llama-server itself: a GGUF model is selected and
    /// llama.cpp is available. Otherwise llama.cpp mode uses an external server.
    var managesLlamaServer: Bool {
        Self.managesLlamaServer(for: configuration)
    }

    nonisolated static func managesLlamaServer(for configuration: LocalServerConfiguration) -> Bool {
        let path = configuration.activeModelPath
        return configuration.inferenceRuntime == .llamaCpp && isGGUF(path)
            && FileManager.default.fileExists(atPath: path) && llamaServerURL != nil
    }

    /// llama-server inside the app (see bundle-llama.sh). Builds run with
    /// `swift run` look for `./bundle-llama.sh .build/bundled-llama/dev` output,
    /// then a Homebrew install.
    nonisolated static var llamaServerURL: URL? {
        var candidates: [URL] = []
        if let resources = Bundle.main.resourceURL {
            candidates.append(resources.appendingPathComponent("llama/llama-server"))
        }
        if let executable = Bundle.main.executableURL?.resolvingSymlinksInPath() {
            // .build/<triple>/debug/VireoMac -> .build/bundled-llama/dev
            candidates.append(executable.deletingLastPathComponent().deletingLastPathComponent()
                .deletingLastPathComponent().appendingPathComponent("bundled-llama/dev/llama-server"))
        }
        candidates += ["/opt/homebrew/bin/llama-server", "/usr/local/bin/llama-server"].map { URL(fileURLWithPath: $0) }
        return candidates.first { FileManager.default.isExecutableFile(atPath: $0.path) }
    }

    nonisolated static var bundledLlamaVersion: String? {
        guard let version = Bundle.main.resourceURL?.appendingPathComponent("llama/VERSION"),
              let text = try? String(contentsOf: version, encoding: .utf8) else { return nil }
        return text.trimmingCharacters(in: .whitespacesAndNewlines)
    }

    /// A vision projector ("mmproj") next to a GGUF model lets it read images.
    nonisolated static func projectorPath(forModelAt path: String) -> String? {
        let folder = URL(fileURLWithPath: path).deletingLastPathComponent()
        let files = (try? FileManager.default.contentsOfDirectory(atPath: folder.path)) ?? []
        return files
            .filter { $0.lowercased().contains("mmproj") && $0.lowercased().hasSuffix(".gguf") }
            .sorted()
            .first
            .map { folder.appendingPathComponent($0).path }
    }

    /// A chat template next to a GGUF model ("<model name>.jinja", or "chat_template.jinja"
    /// as in a Hugging Face model folder) replaces the one inside the file. A GGUF keeps the
    /// template from when it was converted: Gemma 4 31B's still had Google's April template,
    /// three fixes behind (tool calls, turn closing, keeping thinking; found 2026-11-09).
    nonisolated static func chatTemplatePath(forModelAt path: String) -> String? {
        let model = URL(fileURLWithPath: path)
        let folder = model.deletingLastPathComponent()
        return [model.deletingPathExtension().lastPathComponent + ".jinja", "chat_template.jinja"]
            .map { folder.appendingPathComponent($0).path }
            .first { FileManager.default.fileExists(atPath: $0) }
    }

    nonisolated static func llamaServerArguments(
        modelPath: String, port: Int, contextLength: Int, projectorPath: String?,
        chatTemplatePath: String? = nil
    ) -> [String] {
        var arguments = [
            "--model", modelPath,
            // Only the API server talks to it, so it stays on loopback in network mode too.
            "--host", "127.0.0.1", "--port", "\(port)",
            "--ctx-size", "\(contextLength)",
            // One slot: every request gets the whole context rather than a quarter of it.
            "--parallel", "1",
            // Every layer on the GPU (Apple Silicon memory is shared), templates for tool calls.
            "--n-gpu-layers", "999", "--jinja",
            "--alias", URL(fileURLWithPath: modelPath).deletingPathExtension().lastPathComponent,
            "--no-webui",
        ]
        if let projectorPath { arguments += ["--mmproj", projectorPath] }
        if let chatTemplatePath { arguments += ["--chat-template-file", chatTemplatePath] }
        return arguments
    }

    private func startOrAdoptLlama() async {
        guard canStart(mlx.status) else { return }
        let healthURL = URL(string: "http://localhost:\(configuration.mlxPort)/health")!
        if await isHealthy(healthURL) {
            adoptedServices.insert("mlx")
            setStatus(.running, for: "mlx")
            appendLog("[manager] Using an inference server on port \(configuration.mlxPort) that Vireo did not start.", for: "mlx")
            startupMessage = "An inference server that Vireo did not start is already running on port \(configuration.mlxPort). It may be serving a different model, and Stop can't shut it down. Quit it, or change the inference port in Settings."
            return
        }
        guard let server = Self.llamaServerURL else {
            setStatus(.error, for: "mlx")
            startupMessage = "GGUF models need llama.cpp, which isn't included in this build of Vireo."
            return
        }
        setStatus(.starting, for: "mlx")
        let arguments = Self.llamaServerArguments(
            modelPath: modelPath, port: configuration.mlxPort,
            contextLength: configuration.llamaContextLength,
            projectorPath: Self.projectorPath(forModelAt: modelPath),
            chatTemplatePath: Self.chatTemplatePath(forModelAt: modelPath)
        )
        startProcess(service: "mlx", executable: server, arguments: arguments, environment: [:])
        // llama-server answers /health with 503 until the model has loaded.
        pollHealth(service: "mlx", url: healthURL, timeout: 300)
    }

    private func inferenceHealthURLs() -> [URL] {
        guard let base = URL(string: "http://localhost:\(configuration.mlxPort)") else { return [] }
        switch configuration.inferenceRuntime {
        case .mlx:
            return [base.appendingPathComponent("health")]
        case .llamaCpp:
            return [
                base.appendingPathComponent("health"),
                base.appendingPathComponent("v1/models"),
            ]
        case .appleFoundation:
            return []
        }
    }

    private func startOrAdoptInferenceRuntime() async {
        switch configuration.inferenceRuntime {
        case .mlx:
            await startOrAdoptMLX()
        case .llamaCpp where managesLlamaServer:
            await startOrAdoptLlama()
        case .llamaCpp:
            guard canStart(mlx.status) else { return }
            let healthURLs = inferenceHealthURLs()
            var healthy = false
            for url in healthURLs {
                if await isHealthy(url) {
                    healthy = true
                    break
                }
            }
            if healthy {
                setStatus(.running, for: "mlx")
                appendLog("[manager] Using existing healthy llama.cpp service on port \(configuration.mlxPort).", for: "mlx")
            } else {
                setStatus(.error, for: "mlx")
                startupMessage = Self.llamaModelProblem(for: configuration)
                    ?? "llama.cpp runtime is selected, but no healthy inference server was found on port \(configuration.mlxPort). Start llama-server first."
            }
        case .appleFoundation:
            if appleFoundationAvailability.isAvailable {
                setStatus(.error, for: "mlx")
                startupMessage = "Apple Foundation runtime is selected, but inference pipeline integration is not enabled in this local server process yet."
            } else {
                setStatus(.error, for: "mlx")
                startupMessage = appleFoundationAvailability.unavailableReason
            }
        }
    }

            appleFoundationAvailability: appleFoundationAvailability

    private func refreshAppleFoundationAvailability() {
        appleFoundationDiagnostics = Self.evaluateAppleFoundationRuntimeDiagnostics()

    private func normalizeRuntimeSelectionIfNeeded(warningPrefix: String) {
        let normalized = Self.normalizedConfiguration(

            appleFoundationAvailability: appleFoundationAvailability

        startupMessage = "\(warningPrefix): \(appleFoundationAvailability.unavailableReason) Falling back to MLX."

    private func persistConfiguration(_ value: LocalServerConfiguration) {
        if let data = try? JSONEncoder().encode(value) {
            UserDefaults.standard.set(data, forKey: configurationKey)
        }
        UserDefaults.standard.set(value.activeModelPath, forKey: "invurse_local_model_path")
    }

    nonisolated static func normalizedRuntime(
        _ requestedRuntime: InferenceRuntime,
        appleFoundationAvailability: AppleFoundationRuntimeAvailability
    ) -> InferenceRuntime {
        guard requestedRuntime == .appleFoundation else { return requestedRuntime }
        return appleFoundationAvailability.isAvailable ? .appleFoundation : .mlx
    }

    nonisolated static func normalizedConfiguration(
        _ configuration: LocalServerConfiguration,
        appleFoundationAvailability: AppleFoundationRuntimeAvailability
    ) -> LocalServerConfiguration {
        var normalized = configuration
        normalized.inferenceRuntime = normalizedRuntime(
            configuration.inferenceRuntime,
            appleFoundationAvailability: appleFoundationAvailability
        )
        return normalized
    }

    nonisolated static func evaluateAppleFoundationRuntimeAvailability() -> AppleFoundationRuntimeAvailability {
        evaluateAppleFoundationRuntimeDiagnostics().availability
    }

    nonisolated static func evaluateAppleFoundationRuntimeDiagnostics() -> AppleFoundationRuntimeDiagnostics {
        guard #available(macOS 15.0, *) else {
            return AppleFoundationRuntimeDiagnostics(
                osSupport: .fail,
                frameworkAvailability: .notChecked,
                entitlement: .notChecked
            )
        }

#if canImport(FoundationModels)
        let entitlementState: RuntimeCheckState = hasAppleFoundationRuntimeEntitlement() ? .pass : .fail
        return AppleFoundationRuntimeDiagnostics(
            osSupport: .pass,
            frameworkAvailability: .pass,
            entitlement: entitlementState
        )
#else
        return AppleFoundationRuntimeDiagnostics(
            osSupport: .pass,
            frameworkAvailability: .fail,
            entitlement: .notChecked
        )
#endif
    }

    private nonisolated static func hasAppleFoundationRuntimeEntitlement() -> Bool {
        guard let task = SecTaskCreateFromSelf(nil) else { return false }
        let entitlementKeys = [
            "com.apple.developer.foundation-models",
            "com.apple.developer.apple-intelligence"
        ]

        for key in entitlementKeys {
            guard let raw = SecTaskCopyValueForEntitlement(task, key as CFString, nil) else { continue }
            let bridged = raw as AnyObject
            if let number = bridged as? NSNumber, number.boolValue {
                return true
            }
            if let text = bridged as? String, !text.isEmpty {
                return true
            }
            if let values = bridged as? [String], !values.isEmpty {
                return true
            }
        }
        return false
    }

    private func prepareBundledServerDirectory() {
        guard let bundledServer = Bundle.main.resourceURL?.appendingPathComponent("server"),
              FileManager.default.fileExists(atPath: bundledServer.appendingPathComponent("api_server.py").path)
        else { return }

        let supportDirectory = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("InvurseChat/server", isDirectory: true)

    private func installServerDependencies() async -> Bool {
        let serverDirectory = URL(fileURLWithPath: configuration.serverDirectory)
        let setupScript = serverDirectory.appendingPathComponent("setup-mlx.sh")
        guard FileManager.default.fileExists(atPath: setupScript.path) else { return false }

        return await withCheckedContinuation { continuation in
            let process = Process()
            process.executableURL = URL(fileURLWithPath: "/bin/bash")
            process.arguments = [setupScript.path]
            process.currentDirectoryURL = serverDirectory
            process.environment = ProcessInfo.processInfo.environment.merging([
                "PATH": "/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin"
            ], uniquingKeysWith: { _, new in new })
            // pip prints far more than a pipe buffer holds; drain it into the API log
            // so the script can't block on a full pipe.
            let output = Pipe()
            process.standardOutput = output
            process.standardError = output
            output.fileHandleForReading.readabilityHandler = { [weak self] handle in
                let data = handle.availableData
                guard !data.isEmpty else {
                    handle.readabilityHandler = nil
                    return
                }
                let text = String(decoding: data, as: UTF8.self)
                Task { @MainActor in self?.appendLog(text, for: "api") }
            }
            process.terminationHandler = { process in
                continuation.resume(returning: process.terminationStatus == 0)
            }
            do {
                try process.run()
            } catch {
                continuation.resume(returning: false)
            }
        }
    }

    private func validateStartup() -> Bool {
        let preflight = StartupPreflight.evaluate(configuration: configuration)
        let errors = preflight.errors
        let warnings = preflight.warnings

        if !errors.isEmpty {
            startupMessage = errors.joined(separator: " ")
            if errors.contains(where: { $0.contains("model") }) { setStatus(.error, for: "mlx") }
            if errors.contains(where: { $0.contains("API server") }) { setStatus(.error, for: "api") }
            return false
        }

        if !warnings.isEmpty {
            startupMessage = warnings.joined(separator: " ")
        } else {
            startupMessage = ""
        }
        return true
    }

    private func ensureDependencies() async -> Bool {
        if await pythonHasServerDependencies() {
            dependencySetupStatus = "Python dependencies ready"
            dependenciesFailed = false
            return true
        }
        // The bundled Python is complete and signed; installing into it would
        // break the app's signature, so a failed check means a damaged install.
        if Self.bundledPythonURL != nil {
            dependencySetupStatus = "Vireo's built-in Python is incomplete. Reinstall Vireo."
            dependenciesFailed = true
            return false
        }
        if let dependencySetupTask {
            return await dependencySetupTask.value
        }

        dependencySetupInProgress = true
        dependencySetupStatus = "Installing Python dependencies..."
        let task = Task { @MainActor [weak self] in
            guard let self else { return false }
            let installed = await self.installServerDependencies()
            var ready = false
            if installed {
                ready = await self.pythonHasServerDependencies()
            }
            self.dependencySetupInProgress = false
            self.dependenciesFailed = !ready
            self.dependencySetupStatus = ready
                ? "Python dependencies ready"
                : "Dependency installation failed. Check your internet connection and try again."
            return ready
        }
        dependencySetupTask = task
        let result = await task.value
        dependencySetupTask = nil
        return result
    }
